   101→def _conn():
   102→    return pymysql.connect(
   103→        host=MYSQL_HOST, port=MYSQL_PORT, db=MYSQL_DB,
   104→        user=MYSQL_USER, password=MYSQL_PASS,

   109→def db_query(sql, params=None):
   110→    conn = _conn()
   111→    try:
   112→        with conn.cursor() as cur:
   113→            cur.execute(sql, params or ())
   114→            return cur.fetchall()
   115→    finally:
   116→        conn.close()
   117→
   118→def db_query_one(sql, params=None):
   119→    conn = _conn()
   120→    try:
   121→        with conn.cursor() as cur:
   122→            cur.execute(sql, params or ())
   123→            return cur.fetchone()
   124→    finally:
   125→        conn.close()
   126→
   127→def db_execute(sql, params=None):
   128→    conn = _conn()
   129→    try:
   130→        with conn.cursor() as cur:
   131→            cur.execute(sql, params or ())
   132→            return cur.lastrowid or cur.rowcount
   133→    finally:
   134→        conn.close()

   138→def _int(val):
   139→    """Safely convert to int (handles Decimal, None, str)."""
   140→    if val is None:
   141→        return 0
   142→    return int(float(val))
   143→
   144→def _str(val):
   145→    """Safely convert to stripped string."""
   146→    if val is None:
   147→        return ''
   148→    return str(val).strip()
   149→
   150→def _is_sob_medida(codigo):
   151→    """Verifica se o produto codigo termina com -X (sob medida)."""
   152→    if not codigo:
   153→        return False
   154→    c = str(codigo).strip().upper()
   155→    return c.endswith('-X')
   156→
   157→def _extract_s_code(codigo_produto):
   158→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   159→    if not codigo_produto:
   160→        return ''
   161→    m = re.search(r'S-\d+', str(codigo_produto))
   162→    return m.group(0) if m else str(codigo_produto).strip()
   163→

   165→# ── Produto name cache (from MariaDB) ──────────────────────────────
   166→
   167→def _get_produto_nome(codigo_produto):
   168→    """Get real product name from in-memory cache (PostgreSQL produto.pronome)."""
   169→    if not codigo_produto:
   170→        return ''
   171→    key = str(codigo_produto).strip()
   172→    return _produto_cache.get(key, '')
   173→
   174→
   175→def _pg_query_one(sql, params=None):
   176→    """Query PostgreSQL and return one row as dict."""
   177→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   178→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   179→    conn.autocommit = True
   180→    try:
   181→        with conn.cursor() as cur:
   182→            cur.execute(sql, params or ())
   183→            cols = [d[0] for d in cur.description] if cur.description else []
   184→            row = cur.fetchone()
   185→            if row:
   186→                return dict(zip(cols, row))
   187→            return None
   188→    finally:
   189→        conn.close()
   190→
   191→
   192→def _load_produto_cache():
   193→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   194→    global _produto_cache
   195→    _produto_cache = {}
   196→    # 1) PostgreSQL: tabela produto (pronome real)
   197→    try:
   198→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   199→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   200→        conn.autocommit = True
   201→        try:
   202→            with conn.cursor() as cur:
   203→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   204→                cols = [d[0] for d in cur.description]
   205→                for row in cur.fetchall():
   206→                    r = dict(zip(cols, row))
   207→                    cod = str(r.get('produto', '')).strip()
   208→                    nome = str(r.get('pronome', '')).strip()
   209→                    if cod and nome:
   210→                        _produto_cache[cod] = nome[:200]
   211→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   212→        finally:
   213→            conn.close()
   214→    except Exception as e:
   215→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   216→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   217→    try:
   218→        rows = db_query(
   219→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   220→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   221→        )
   222→        added = 0
   223→        for r in rows:
   224→            cod = str(r.get('codigo_produto', '')).strip()
   225→            nome = str(r.get('descricao_produto', '')).strip()
   226→            if cod and cod not in _produto_cache and nome:
   227→                _produto_cache[cod] = nome[:200]
   228→                added += 1
   229→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   230→    except Exception as e:
   231→        print('[WARN] Cache MariaDB falhou: %s' % e)
   232→

   234→# ── Lotes ativos (from MariaDB, no ERP) ────────────────────────────
   235→
   236→def _refresh_lotes_ativos():
   237→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""
   238→    global _lotes_ativos
   239→    try:
   240→        rows = db_query(
   241→            "SELECT DISTINCT lote_codigo FROM lotes_producao "
   242→            "WHERE status IN ('liberado','em_producao','pausado')"
   243→        )
   244→        _lotes_ativos = set()
   245→        for r in rows:
   246→            if r.get('lote_codigo'):
   247→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   248→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   249→    except Exception as e:
   250→        print('[WARN] Lotes ativos falhou: %s' % e)
   251→
   252→
   253→def _build_lotes_filter():
   254→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   255→    if _lotes_ativos:
   256→        ph = ','.join(['%s'] * len(_lotes_ativos))
   257→        where = 'WHERE lote_codigo IN (' + ph + ')'
   258→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   259→        params = tuple(_lotes_ativos)
   260→    else:
   261→        where = 'WHERE 1=0'
   262→        join = ' AND 1=0'
   263→        params = ()
   264→    return where, join, params

   774→def _atualizar_departamento_lote(lote_ordem):
   775→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   776→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   777→    if not lote_ordem:
   778→        return
   779→    # Total de departamentos unicos do lote
   780→    total = db_query_one(
   781→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   782→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   783→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   784→        (lote_ordem,)
   785→    )
   786→    total_setores = int(total['t'] or 0) if total else 0
   787→
   788→    # Primeira operacao pendente (nao concluida) por sequencia
   789→    primeira = db_query_one(
   790→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   791→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   792→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   793→        "ORDER BY op.sequencia ASC LIMIT 1",
   794→        (lote_ordem,)
   795→    )
   796→    dept = primeira['departamento'] if primeira else None
   797→    seq = int(primeira['sequencia']) if primeira else 0
   798→
   799→    db_execute(
   800→        "UPDATE lotes_producao "
   801→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   802→        "WHERE ordem = %s",
   803→        (dept, seq, total_setores, lote_ordem)
   804→    )

   807→def _sync_lote_status(of_numero, novo_status):
   808→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   809→    if not of_numero:
   810→        return
   811→    of_row = db_query_one(
   812→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",
   813→        (of_numero,)
   814→    )
   815→    if not of_row or not of_row.get('lote_ordem'):
   816→        return
   817→    lote_ordem = of_row['lote_ordem']
   818→
   819→    if novo_status == 'em_andamento':
   820→        # Muda status do lote para em_producao se ainda estiver liberado
   821→        db_execute(
   822→            "UPDATE lotes_producao SET status = 'em_producao' "
   823→            "WHERE ordem = %s AND status = 'liberado'",
   824→            (lote_ordem,)
   825→        )
   826→    elif novo_status == 'concluido':
   827→        # Verifica se TODAS as ops do lote estao concluidas
   828→        total_pend = db_query_one(
   829→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   830→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   831→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   832→            (lote_ordem,)
   833→        )
   834→        if total_pend and int(total_pend['t']) == 0:
   835→            db_execute(
   836→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   837→                (lote_ordem,)
   838→            )

   841→def _sync_op_status(of_numero, departamento, novo_status):
   842→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   843→    if not of_numero or not departamento:
   844→        return
   845→    try:
   846→        # 1. Atualiza operacoes_producao
   847→        db_execute(
   848→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   849→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   850→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   851→            (novo_status, novo_status, of_numero, departamento)
   852→        )
   853→        # 2. Atualiza kanban_cards
   854→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   855→        nova_etapa = etapa_map.get(novo_status)
   856→        if nova_etapa:
   857→            db_execute(
   858→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   859→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   860→                (nova_etapa, of_numero, departamento)
   861→            )
   862→        # 3. Sincroniza lotes_producao (status + departamento_atual)
   863→        _sync_lote_status(of_numero, novo_status)
   864→        _atualizar_departamento_lote_from_of(of_numero)
   865→    except Exception as e:
   866→        app.logger.warning('Sync op status falhou: %s', e)

   881→@app.route('/api/finalizar', methods=['POST'])
   882→def api_finalizar():
   883→    """Finalizar producao de pecas especificas e sincroniza com operacoes_producao."""
   884→    try:
   885→        data = request.get_json()
   886→        lotcod = data.get('lote')
   887→        pecas = data.get('pecas', [])
   888→        setor = data.get('setor', SETOR)
   889→        qtd_informada = data.get('qtd_produzida', 0)
   890→
   891→        if not lotcod:
   892→            return jsonify({'error': 'Lote obrigatorio'}), 400
   893→
   894→        setor_para_dept = {
   895→            'CORTE': 'CORTE', 'DOBRA': 'DOBRA', 'SOLDA': 'SOLDA',
   896→            'ACABAMENTO': 'ACABAMENTO', 'MONTAGEM': 'MONTAGEM',
   897→            'FUNILARIA': 'FUNILARIA', 'PINTURA': 'PINTURA',
   898→            'EMBALAGEM': 'EMBALAGEM',
   899→        }
   900→
   901→        if pecas and setor:
   902→            # Finalizar pecas em um setor especifico (com suporte a quantidade parcial)
   903→            for pcod in pecas:
   904→                row = db_query_one(
   905→                    "SELECT sp.id, sp.of_numero, sp.qtd_produzida, "
   906→                    "COALESCE(of2.qtde_ordem, 0) as qtde_ordem "
   907→                    "FROM serralheria_producao sp "
   908→                    "LEFT JOIN ordens_fabricacao of2 ON sp.of_numero = of2.of_numero "
   909→                    "WHERE sp.lote = %s AND sp.produto = %s AND sp.setor = %s "
   910→                    "AND sp.status = 'em_producao'",
   911→                    (lotcod, pcod, setor)
   912→                )
   913→                if not row:
   914→                    app.logger.warning('Finalizar: peca %s setor %s nao encontrada em producao', pcod, setor)
   915→                    continue
   916→
   917→                qtd_total = _int(row['qtde_ordem'])
   918→                qtd_ja_feita = _int(row['qtd_produzida'])
   919→                restante = qtd_total - qtd_ja_feita
   920→
   921→                if qtd_total <= 0:
   922→                    # Sem qtde_ordem, finaliza direto
   923→                    db_execute(
   924→                        "UPDATE serralheria_producao SET status='finalizado', data_fim=NOW() "
   925→                        "WHERE id=%s", (row['id'],)
   926→                    )
   927→                    dept = setor_para_dept.get(setor, setor)
   928→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   929→                    continue
   930→
   931→                # Validacao: quantidade nao pode exceder o restante
   932→                if qtd_informada > restante:
   933→                    return jsonify({
   934→                        'error': 'Quantidade maxima permitida: %d (restam %d de %d)' % (restante, restante, qtd_total),
   935→                        'max_qtd': restante
   936→                    }), 400
   937→
   938→                if qtd_informada <= 0:
   939→                    return jsonify({'error': 'Informe uma quantidade maior que zero'}), 400
   940→
   941→                nova_qtd = qtd_ja_feita + qtd_informada
   942→
   943→                if nova_qtd >= qtd_total:
   944→                    # Finalizacao completa
   945→                    db_execute(
   946→                        "UPDATE serralheria_producao "
   947→                        "SET status='finalizado', data_fim=NOW(), qtd_produzida=%s "
   948→                        "WHERE id=%s",
   949→                        (qtd_total, row['id'])
   950→                    )
   951→                    dept = setor_para_dept.get(setor, setor)
   952→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   953→                    app.logger.info('Lote %s peca %s setor %s: FINALIZADA (%d/%d)', lotcod, pcod, setor, nova_qtd, qtd_total)
   954→                else:
   955→                    # Finalizacao parcial: acumula e mantem em_producao
   956→                    db_execute(
   957→                        "UPDATE serralheria_producao SET qtd_produzida=%s WHERE id=%s",
   958→                        (nova_qtd, row['id'])
   959→                    )
   960→                    app.logger.info('Lote %s peca %s setor %s: PARCIAL %d/%d (restam %d)',
   961→                        lotcod, pcod, setor, nova_qtd, qtd_total, qtd_total - nova_qtd)
   962→
   963→            return jsonify({'success': True, 'mensagem': 'Producao atualizada!'})
   964→        elif pecas:
   965→            # Finalizar todas as producoes das pecas selecionadas (todos os setores) — sem qtde parcial
   966→            placeholders = ','.join(['%s'] * len(pecas))
   967→            rows = db_query(
   968→                "SELECT of_numero, setor FROM serralheria_producao "
   969→                "WHERE lote = %s AND produto IN (" + placeholders + ") AND status = 'em_producao'",
   970→                [lotcod] + pecas
   971→            )
   972→            db_execute(
   973→                "UPDATE serralheria_producao "
   974→                "SET status = 'finalizado', data_fim = NOW() "
   975→                "WHERE lote = %s AND produto IN (" + placeholders + ") AND status = 'em_producao'",
   976→                [lotcod] + pecas
   977→            )
   978→            for row in rows:
   979→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
   980→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
   981→            app.logger.info('Lote %s: %d peca(s) finalizada(s) no totem', lotcod, len(pecas))
   982→        else:
   983→            # Finalizar todas as pecas do lote
   984→            rows = db_query(
   985→                "SELECT of_numero, setor FROM serralheria_producao "
   986→                "WHERE lote = %s AND status = 'em_producao'", (lotcod,)
   987→            )
   988→            db_execute(
   989→                "UPDATE serralheria_producao "
   990→                "SET status = 'finalizado', data_fim = NOW() "
   991→                "WHERE lote = %s AND status = 'em_producao'",
   992→                (lotcod,)
   993→            )
   994→            for row in rows:
   995→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
   996→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
   997→            app.logger.info('Lote %s finalizado no totem', lotcod)
   998→
   999→        return jsonify({'success': True, 'mensagem': 'Producao finalizada!'})
