   102→def _conn():
   103→    return pymysql.connect(
   104→        host=MYSQL_HOST, port=MYSQL_PORT, db=MYSQL_DB,
   105→        user=MYSQL_USER, password=MYSQL_PASS,

   110→def db_query(sql, params=None):
   111→    conn = _conn()
   112→    try:
   113→        with conn.cursor() as cur:
   114→            cur.execute(sql, params or ())
   115→            return cur.fetchall()
   116→    finally:
   117→        conn.close()
   118→
   119→def db_query_one(sql, params=None):
   120→    conn = _conn()
   121→    try:
   122→        with conn.cursor() as cur:
   123→            cur.execute(sql, params or ())
   124→            return cur.fetchone()
   125→    finally:
   126→        conn.close()
   127→
   128→def db_execute(sql, params=None):
   129→    conn = _conn()
   130→    try:
   131→        with conn.cursor() as cur:
   132→            cur.execute(sql, params or ())
   133→            return cur.lastrowid or cur.rowcount
   134→    finally:
   135→        conn.close()

   139→def _int(val):
   140→    """Safely convert to int (handles Decimal, None, str)."""
   141→    if val is None:
   142→        return 0
   143→    return int(float(val))
   144→
   145→def _str(val):
   146→    """Safely convert to stripped string."""
   147→    if val is None:
   148→        return ''
   149→    return str(val).strip()

   151→def _is_sob_medida(codigo):
   152→    """Verifica se o produto codigo termina com -X (sob medida)."""
   153→    if not codigo:
   154→        return False
   155→    c = str(codigo).strip().upper()
   156→    return c.endswith('-X')
   157→
   158→def _extract_s_code(codigo_produto):
   159→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   160→    if not codigo_produto:
   161→        return ''
   162→    m = re.search(r'S-\d+', str(codigo_produto))
   163→    return m.group(0) if m else str(codigo_produto).strip()

   168→def _get_produto_nome(codigo_produto):
   169→    """Get real product name from in-memory cache (PostgreSQL produto.pronome)."""
   170→    if not codigo_produto:
   171→        return ''
   172→    key = str(codigo_produto).strip()
   173→    return _produto_cache.get(key, '')

   176→def _pg_query_one(sql, params=None):
   177→    """Query PostgreSQL and return one row as dict."""
   178→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   179→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   180→    conn.autocommit = True
   181→    try:
   182→        with conn.cursor() as cur:
   183→            cur.execute(sql, params or ())
   184→            cols = [d[0] for d in cur.description] if cur.description else []
   185→            row = cur.fetchone()
   186→            if row:
   187→                return dict(zip(cols, row))
   188→            return None
   189→    finally:
   190→        conn.close()
   191→

   193→def _load_produto_cache():
   194→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   195→    global _produto_cache
   196→    _produto_cache = {}
   197→    # 1) PostgreSQL: tabela produto (pronome real)
   198→    try:
   199→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   200→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   201→        conn.autocommit = True
   202→        try:
   203→            with conn.cursor() as cur:
   204→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   205→                cols = [d[0] for d in cur.description]
   206→                for row in cur.fetchall():
   207→                    r = dict(zip(cols, row))
   208→                    cod = str(r.get('produto', '')).strip()
   209→                    nome = str(r.get('pronome', '')).strip()
   210→                    if cod and nome:
   211→                        _produto_cache[cod] = nome[:200]
   212→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   213→        finally:
   214→            conn.close()
   215→    except Exception as e:
   216→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   217→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   218→    try:
   219→        rows = db_query(
   220→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   221→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   222→        )
   223→        added = 0
   224→        for r in rows:
   225→            cod = str(r.get('codigo_produto', '')).strip()
   226→            nome = str(r.get('descricao_produto', '')).strip()
   227→            if cod and cod not in _produto_cache and nome:
   228→                _produto_cache[cod] = nome[:200]
   229→                added += 1
   230→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   231→    except Exception as e:
   232→        print('[WARN] Cache MariaDB falhou: %s' % e)

   237→def _refresh_lotes_ativos():
   238→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""
   239→    global _lotes_ativos

   243→            "WHERE status IN ('liberado','em_producao','pausado')"
   244→        )
   245→        _lotes_ativos = set()
   246→        for r in rows:
   247→            if r.get('lote_codigo'):
   248→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   249→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   250→    except Exception as e:
   251→        print('[WARN] Lotes ativos falhou: %s' % e)

   253→
   254→def _build_lotes_filter():
   255→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   256→    if _lotes_ativos:
   257→        ph = ','.join(['%s'] * len(_lotes_ativos))
   258→        where = 'WHERE lote_codigo IN (' + ph + ')'
   259→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   260→        params = tuple(_lotes_ativos)
   261→    else:
   262→        where = 'WHERE 1=0'
   263→        join = ' AND 1=0'
   264→        params = ()
   265→    return where, join, params

   268→def _get_lotes_do_setor():
   269→    """Retorna set de lote_codigo que pertencem ao setor do totem.
   270→    Usa departamento_atual do lote E producao ativa."""
   271→    if not _lotes_ativos:
   272→        return set()
   273→    try:
   274→        ph = ','.join(['%s'] * len(_lotes_ativos))
   275→
   276→        # 1. Lotes cujo departamento_atual bate com o setor do totem
   277→        rows1 = db_query(
   278→            "SELECT DISTINCT lote_codigo FROM lotes_producao "
   279→            "WHERE lote_codigo IN (" + ph + ") "
   280→            "AND departamento_atual = %s",
   281→            tuple(_lotes_ativos) + (SETOR,)
   282→        )
   283→
   284→        # 2. Lotes com producao ativa neste setor
   285→        rows2 = db_query(
   286→            "SELECT DISTINCT lote FROM serralheria_producao "
   287→            "WHERE lote IN (" + ph + ") "
   288→            "AND setor = %s AND status = 'em_producao'",
   289→            tuple(_lotes_ativos) + (SETOR,)
   290→        )
   291→
   292→        resultado = set()
   293→        for r in rows1:
   294→            if r.get('lote_codigo'):
   295→                resultado.add(str(r['lote_codigo']).strip())
   296→        for r in rows2:
   297→            if r.get('lote'):
   298→                resultado.add(str(r['lote']).strip())
   299→        return resultado
   300→    except Exception as e:
   301→        app.logger.error('Erro _get_lotes_do_setor: %s', e)
   302→        return set()

   817→def _atualizar_departamento_lote(lote_ordem):
   818→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   819→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   820→    if not lote_ordem:
   821→        return
   822→    # Total de departamentos unicos do lote
   823→    total = db_query_one(
   824→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   825→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   826→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   827→        (lote_ordem,)
   828→    )
   829→    total_setores = int(total['t'] or 0) if total else 0
   830→
   831→    # Primeira operacao pendente (nao concluida) por sequencia
   832→    primeira = db_query_one(
   833→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   834→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   835→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   836→        "ORDER BY op.sequencia ASC LIMIT 1",
   837→        (lote_ordem,)
   838→    )

   840→    seq = int(primeira['sequencia']) if primeira else 0
   841→
   842→    db_execute(
   843→        "UPDATE lotes_producao "
   844→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "

   850→def _sync_lote_status(of_numero, novo_status):
   851→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   852→    if not of_numero:
   853→        return
   854→    of_row = db_query_one(
   855→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",
   856→        (of_numero,)
   857→    )
   858→    if not of_row or not of_row.get('lote_ordem'):
   859→        return
   860→    lote_ordem = of_row['lote_ordem']
   861→
   862→    if novo_status == 'em_andamento':
   863→        # Muda status do lote para em_producao se ainda estiver liberado
   864→        db_execute(
   865→            "UPDATE lotes_producao SET status = 'em_producao' "
   866→            "WHERE ordem = %s AND status = 'liberado'",
   867→            (lote_ordem,)
   868→        )
   869→    elif novo_status == 'concluido':
   870→        # Verifica se TODAS as ops do lote estao concluidas
   871→        total_pend = db_query_one(
   872→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   873→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   874→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   875→            (lote_ordem,)
   876→        )
   877→        if total_pend and int(total_pend['t']) == 0:
   878→            db_execute(
   879→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   880→                (lote_ordem,)
   881→            )

   884→def _sync_op_status(of_numero, departamento, novo_status):
   885→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   886→    if not of_numero or not departamento:
   887→        return
   888→    try:
   889→        # 1. Atualiza operacoes_producao
   890→        db_execute(
   891→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   892→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   893→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   894→            (novo_status, novo_status, of_numero, departamento)
   895→        )
   896→        # 2. Atualiza kanban_cards
   897→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   898→        nova_etapa = etapa_map.get(novo_status)
   899→        if nova_etapa:
   900→            db_execute(
   901→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   902→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   903→                (nova_etapa, of_numero, departamento)
   904→            )
   905→        # 3. Sincroniza lotes_producao (status + departamento_atual)
   906→        _sync_lote_status(of_numero, novo_status)
   907→        _atualizar_departamento_lote_from_of(of_numero)
   908→    except Exception as e:
   909→        app.logger.warning('Sync op status falhou: %s', e)

   954→                    (lotcod, pcod, setor)
   955→                )
   956→                if not row:
   957→                    app.logger.warning('Finalizar: peca %s setor %s nao encontrada em producao', pcod, setor)
   958→                    continue
   959→
   960→                qtd_total = _int(row['qtde_ordem'])
   961→                qtd_ja_feita = _int(row['qtd_produzida'])
   962→                restante = qtd_total - qtd_ja_feita
   963→
   964→                if qtd_total <= 0:
   965→                    # Sem qtde_ordem, finaliza direto
   966→                    db_execute(
   967→                        "UPDATE serralheria_producao SET status='finalizado', data_fim=NOW() "
   968→                        "WHERE id=%s", (row['id'],)
   969→                    )
   970→                    dept = setor_para_dept.get(setor, setor)
   971→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   972→                    continue
   973→
   974→                # Validacao: quantidade nao pode exceder o restante
   975→                if qtd_informada > restante:
   976→                    return jsonify({
   977→                        'error': 'Quantidade maxima permitida: %d (restam %d de %d)' % (restante, restante, qtd_total),
   978→                        'max_qtd': restante
   979→                    }), 400
   980→
   981→                if qtd_informada <= 0:
   982→                    return jsonify({'error': 'Informe uma quantidade maior que zero'}), 400
   983→
   984→                nova_qtd = qtd_ja_feita + qtd_informada
   985→
   986→                if nova_qtd >= qtd_total:
   987→                    # Finalizacao completa
   988→                    db_execute(
   989→                        "UPDATE serralheria_producao "
   990→                        "SET status='finalizado', data_fim=NOW(), qtd_produzida=%s "
   991→                        "WHERE id=%s",
   992→                        (qtd_total, row['id'])
   993→                    )
   994→                    dept = setor_para_dept.get(setor, setor)
   995→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   996→                    app.logger.info('Lote %s peca %s setor %s: FINALIZADA (%d/%d)', lotcod, pcod, setor, nova_qtd, qtd_total)
   997→                else:
   998→                    # Finalizacao parcial: acumula e mantem em_producao
   999→                    db_execute(
  1000→                        "UPDATE serralheria_producao SET qtd_produzida=%s WHERE id=%s",
  1001→                        (nova_qtd, row['id'])
  1002→                    )
  1003→                    app.logger.info('Lote %s peca %s setor %s: PARCIAL %d/%d (restam %d)',
  1004→                        lotcod, pcod, setor, nova_qtd, qtd_total, qtd_total - nova_qtd)
  1005→
  1006→            return jsonify({'success': True, 'mensagem': 'Producao atualizada!'})
  1007→        elif pecas:
  1008→            # Finalizar todas as producoes das pecas selecionadas (todos os setores) — sem qtde parcial
  1009→            placeholders = ','.join(['%s'] * len(pecas))
  1010→            rows = db_query(
  1011→                "SELECT of_numero, setor FROM serralheria_producao "
  1012→                "WHERE lote = %s AND produto IN (" + placeholders + ") AND status = 'em_producao'",

  1015→            db_execute(
  1016→                "UPDATE serralheria_producao "

  1019→                [lotcod] + pecas
  1020→            )
  1021→            for row in rows:
  1022→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
  1023→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
  1024→            app.logger.info('Lote %s: %d peca(s) finalizada(s) no totem', lotcod, len(pecas))
  1025→        else:
  1026→            # Finalizar todas as pecas do lote
  1027→            rows = db_query(
  1028→                "SELECT of_numero, setor FROM serralheria_producao "
  1029→                "WHERE lote = %s AND status = 'em_producao'", (lotcod,)
  1030→            )
  1031→            db_execute(
  1032→                "UPDATE serralheria_producao "

  1034→                "WHERE lote = %s AND status = 'em_producao'",
  1035→                (lotcod,)

  1037→            for row in rows:
  1038→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
  1039→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
  1040→            app.logger.info('Lote %s finalizado no totem', lotcod)
  1041→
  1042→        return jsonify({'success': True, 'mensagem': 'Producao finalizada!'})
  1043→    except Exception as e:
