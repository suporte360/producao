    65→@app.template_filter('formatar_tempo')
    66→def formatar_tempo(segundos):
    67→    """Formata segundos em HH:MM:SS"""
    68→    try:
    69→        seg = int(segundos or 0)
    70→        h = seg // 3600
    71→        m = (seg % 3600) // 60
    72→        s = seg % 60
    73→        return f'{h:02d}:{m:02d}:{s:02d}'
    74→    except Exception:
    75→        return '00:00:00'

    77→@app.template_filter('formatar_data_br')
    78→def formatar_data_br(value):
    79→    """Formata datetime/date para dd/mm/yyyy HH:MM"""
    80→    if not value:
    81→        return '-'
    82→    try:
    83→        if hasattr(value, 'strftime'):
    84→            if hasattr(value, 'hour'):
    85→                return value.strftime('%d/%m/%Y %H:%M')
    86→            return value.strftime('%d/%m/%Y')
    87→        return str(value)
    88→    except Exception:
    89→        return str(value)

    91→# =============================================
    92→# Proteção contra força bruta no login
    93→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
    94→# =============================================
    95→
    96→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
    97→MAX_TENTATIVAS  = 5
    98→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
    99→
   100→def _get_ip():
   101→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
   102→
   103→def _checar_bloqueio(ip):
   104→    dados = _login_attempts[ip]
   105→    if time.time() < dados['bloqueado_ate']:
   106→        restante = int(dados['bloqueado_ate'] - time.time())
   107→        return True, restante
   108→    return False, 0
   109→
   110→def _registrar_falha(ip):
   111→    dados = _login_attempts[ip]
   112→    dados['tentativas'] += 1
   113→    if dados['tentativas'] >= MAX_TENTATIVAS:
   114→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   115→        dados['tentativas'] = 0
   116→
   117→def _limpar_tentativas(ip):
   118→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}

   124→def login_required(f):
   125→    @wraps(f)
   126→    def decorated(*args, **kwargs):
   127→        if 'usuario_id' not in session:
   128→            flash('Faça login para continuar.', 'warning')
   129→            return redirect(url_for('login'))
   130→        return f(*args, **kwargs)
   131→    return decorated
   132→
   133→def role_required(*roles):
   134→    def decorator(f):
   135→        @wraps(f)
   136→        def decorated(*args, **kwargs):
   137→            if session.get('usuario_role') not in roles:
   138→                flash('Acesso não autorizado.', 'error')
   139→                return redirect(url_for('dashboard'))
   140→            return f(*args, **kwargs)
   141→        return decorated
   142→    return decorator

   148→@app.route('/login', methods=['GET', 'POST'])
   149→def login():
   150→    if 'usuario_id' in session:
   151→        return redirect(url_for('dashboard'))
   152→
   153→    if request.method == 'POST':
   154→        ip = _get_ip()
   155→
   156→        bloqueado, restante = _checar_bloqueio(ip)
   157→        if bloqueado:
   158→            mins = restante // 60
   159→            secs = restante % 60
   160→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   161→            return render_template('login.html')
   162→
   163→        usuario = request.form.get('usuario', '').strip()
   164→        senha   = request.form.get('senha', '')
   165→
   166→        if not usuario or not senha:
   167→            flash('Informe usuário e senha.', 'error')
   168→            return render_template('login.html')
   169→
   170→        user = db.verificar_usuario(usuario, senha)
   171→
   172→        if user:
   173→            _limpar_tentativas(ip)
   174→            session.permanent = True
   175→            session['usuario_id']          = user['id']
   176→            session['usuario_nome']        = user['nome']
   177→            session['usuario_role']        = user['role']
   178→            session['usuario_departamento']= user.get('nome_departamento', '')
   179→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   180→            return redirect(url_for('dashboard'))
   181→        else:
   182→            _registrar_falha(ip)
   183→            dados   = _login_attempts[ip]
   184→            restantes = MAX_TENTATIVAS - dados['tentativas']
   185→            if restantes > 0:
   186→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   187→            else:
   188→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   189→
   190→    return render_template('login.html')

   294→    departamentos = db.listar_departamentos(apenas_ativos=True)
   295→    todos_cards = db.get_kanban_cards(departamento=None)
   296→
   297→    # Agrupar cards por departamento
   298→    cards_por_setor = {}
   299→    for dept in departamentos:
   300→        cards_por_setor[dept['codigo']] = {
   301→            'info': dept,
   302→            'cards': [],
   303→            'total': 0,
   304→            'em_producao': 0,
   305→            'aguardando': 0,
   306→            'pausados': 0,
   307→            'concluidos': 0,
   308→        }
   309→
   310→    for card in todos_cards:
   311→        dept_code = card.get('departamento', '')
   312→        if dept_code in cards_por_setor:
   313→            cards_por_setor[dept_code]['cards'].append(card)
   314→            cards_por_setor[dept_code]['total'] += 1
   315→            etapa = card.get('etapa', '')
   316→            if etapa == 'em_producao':
   317→                cards_por_setor[dept_code]['em_producao'] += 1
   318→            elif etapa in ('aguardando', 'pendente'):
   319→                cards_por_setor[dept_code]['aguardando'] += 1
   320→            elif etapa == 'pausado':
   321→                cards_por_setor[dept_code]['pausados'] += 1
   322→            elif etapa == 'concluido':
   323→                cards_por_setor[dept_code]['concluidos'] += 1
   324→
   325→    # KPIs gerais
   326→    kpis = {
   327→        'setores_ativos': sum(1 for s in cards_por_setor.values() if s['total'] > 0),
   328→        'total_cards': len(todos_cards),
   329→        'em_producao': sum(1 for c in todos_cards if c.get('etapa') == 'em_producao'),
   330→        'concluidos_hoje': db.get_kanban_concluidos_hoje(),
   331→    }
   332→
   333→    return render_template('kanban/setores.html',
   334→                           cards_por_setor=cards_por_setor,
   335→                           kpis=kpis,
   336→                           usuario_role=session.get('usuario_role'))

   378→def pcp_importar_erp():
   379→    pg_ok   = False
   380→    pg_erro = ''
   381→    lotes_erp = []
   382→    lotes_ja_importados = set()
   383→
   384→    try:
   385→        pg = DatabasePostgreSQL()
   386→        # Busca lotes do ERP com produto final e maior ordem
   387→        lotes_raw = pg.get_lotes_agrupados_abertos()
   388→        # Para cada lote busca o produto final
   389→        for lote in lotes_raw:
   390→            prod = pg.get_produto_pronto_por_lote(lote['lote_codigo'])
   391→            lote['nome_produto']        = prod['nome_produto']   if prod else None
   392→            lote['produto_final_codigo']= prod['codigo_produto'] if prod else lote.get('produto_final_codigo')
   393→            lote['maior_ordem']         = prod['quantidade']     if prod else None
   394→            # busca a maior ordem numerica do lote
   395→            ordens = pg.get_ordens_por_lote(lote['lote_codigo'])
   396→            if ordens:
   397→                lote['maior_ordem'] = max(int(o['ordem']) for o in ordens if o.get('ordem') and str(o['ordem']).isdigit())
   398→        lotes_erp = lotes_raw
   399→        pg_ok = True
   400→    except Exception as e:
   401→        pg_erro = str(e)
   402→
   403→    # Lotes já importados no MariaDB
   404→    ja = db.query("SELECT DISTINCT lote_codigo FROM lotes_producao WHERE lote_codigo IS NOT NULL")
   405→    lotes_ja_importados = {r['lote_codigo'] for r in ja}
   406→
   407→    return render_template('pcp/importar_erp.html',
   408→                           pg_ok=pg_ok, pg_erro=pg_erro,
   409→                           lotes_erp=lotes_erp,
   410→                           lotes_ja_importados=lotes_ja_importados,
   411→                           now=datetime.now(),
   412→                           usuario_nome=session['usuario_nome'])

   415→@app.route('/pcp/importar_erp', methods=['POST'])
   416→@login_required
   417→@role_required('admin', 'gerente', 'pcp', 'diretor')
   418→def pcp_executar_importacao():
   419→    data  = request.json or {}
   420→    lotes = data.get('lotes', [])
   421→    if not lotes:
   422→        return jsonify({'status':'error','message':'Nenhum lote informado'}), 400
   423→
   424→    importados = 0
   425→    atualizados = 0
   426→    erros = []
   427→
   428→    try:
   429→        pg = DatabasePostgreSQL()
   430→    except Exception as e:
   431→        return jsonify({'status':'error','message': f'Erro PostgreSQL: {e}'}), 500
   432→
   433→    for item in lotes:
   434→        try:
   435→            lote_cod = str(item.get('lote_codigo',''))
   436→            ordem_num = item.get('ordem')
   437→            if not lote_cod:
   438→                continue
   439→
   440→            # Verifica se já existe
   441→            existente = db.query_one(
   442→                "SELECT ordem FROM lotes_producao WHERE lote_codigo=%s", (lote_cod,))
   443→            ja_existia = existente is not None
   444→
   445→            # Importa lote principal
   446→            db.importar_lote_erp({
   447→                'ordem':             int(ordem_num) if str(ordem_num).isdigit() else 0,
   448→                'lote_codigo':       lote_cod,
   449→                'codigo_produto':    item.get('produto',''),
   450→                'descricao_produto': item.get('produto',''),  # agora envia lote_descricao do template
   451→                'qtde_ordem':        float(item.get('qtd', 0) or 0),
   452→                'unidade_medida':    'PC',
   453→                'status_erp':        'A',
   454→                'data_previsao_erp': item.get('previsao') or None,
   455→                'data_abertura_erp': item.get('abertura') or None,
   456→                'planejador':        session.get('usuario_nome'),
   457→            })
   458→
   459→            lote_ordem_local = int(ordem_num) if str(ordem_num).isdigit() else 0
   460→
   461→            # Busca ordens do lote no ERP e importa OFs
   462→            ordens_erp = pg.get_ordens_por_lote(lote_cod)
   463→            for of in ordens_erp:
   464→                of_num = of.get('ordem')
   465→                if not of_num:
   466→                    continue
   467→                db.importar_of({
   468→                    'lote_ordem':       lote_ordem_local,
   469→                    'of_numero':        str(of_num),
   470→                    'codigo_produto':   of.get('produto',''),
   471→                    'descricao_produto':item.get('produto',''),
   472→                    'qtde_ordem':       float(of.get('quantidade') or 0),
   473→                    'unidade_medida':   'PC',
   474→                    'tipo':             'filho' if of.get('nivel_producao','') != '1' else 'pai',
   475→                    'data_previsao':    of.get('data_previsao') or None,
   476→                })
   477→
   478→                # Busca processos (operações) de cada OF
   479→                processos = pg.get_processos_ordem(str(of_num))
   480→                for proc in processos:
   481→                    dept_original = (proc.get('nome_departamento') or '').upper().strip()
   482→                    if not dept_original:
   483→                        continue
   484→                    # Mapeia nomes do ERP (fases) para nomes do sistema local
   485→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   486→                    db.importar_operacao({
   487→                        'of_numero':          str(of_num),
   488→                        'lote_ordem':         lote_ordem_local,
   489→                        'sequencia':          proc.get('numero_sequencial', 0),
   490→                        'fase':               proc.get('fase',''),
   491→                        'descricao_operacao': proc.get('descricao_processo') or proc.get('operacao',''),
   492→                        'departamento':       dept,
   493→                        'codigo_barras':      None,
   494→                    })
   495→
   496→                # Sincroniza kanban
   497→                try:
   498→                    db.sincronizar_kanban_com_operacoes(lote_ordem_local)
   499→                except Exception:
   500→                    pass
   501→
   502→            db.add_log(session['usuario_id'], 'importar_erp',
   503→                       f'Lote {lote_cod} importado do ERP', 'lotes_producao', lote_ordem_local)
   504→
   505→            if ja_existia:
   506→                atualizados += 1
   507→            else:
   508→                importados += 1
   509→
   510→        except Exception as e:
   511→            erros.append(f"{item.get('lote_codigo')}: {e}")
   512→
   513→    return jsonify({
   514→        'status':     'success',
   515→        'importados':  importados,

   825→def _baixar_estoque_por_of(ordem):
   826→    """Busca materiais da OF no ERP e da baixa automatica no estoque da fabrica."""
   827→    try:
   828→        pg = DatabasePostgreSQL()
   829→        materiais = pg.get_requisicoes_ordem(ordem)
   830→        if not materiais:
   831→            return
   832→        for mat in materiais:
   833→            codigo = str(mat.get('produto') or mat.get('reqproduto') or '')
   834→            if not codigo:
   835→                continue
   836→            qtd = float(mat.get('quantidade') or mat.get('rqoquanti') or 0)
   837→            if qtd <= 0:
   838→                continue
   839→            item = db.get_estoque_por_codigo(codigo, tipo='fabrica')
   840→            if item and item['qtd_atual'] > 0:
   841→                db.registrar_baixa_estoque(codigo, qtd, ordem, 'liberacao_of', tipo='fabrica')
   842→                print(f"[ESTOQUE] Baixa fabrica: {codigo} -{qtd} (OP #{ordem})")
   843→    except Exception as e:
   844→        print(f"[ESTOQUE] Erro ao baixar estoque da OP #{ordem}: {e}")
