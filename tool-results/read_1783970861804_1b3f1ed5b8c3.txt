   116→   116→    """Safely convert to stripped string."""
   117→   117→    if val is None:
   118→   118→        return ''
   119→   119→    return str(val).strip()

   121→   121→def _is_sob_medida(codigo):
   122→   122→    """Verifica se o produto codigo termina com -X (sob medida)."""
   123→   123→    if not codigo:
   124→   124→        return False
   125→   125→    c = str(codigo).strip().upper()
   126→   126→    return c.endswith('-X')

   128→   128→def _extract_s_code(codigo_produto):
   129→   129→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   130→   130→    if not codigo_produto:
   131→   131→        return ''
   132→   132→    m = re.search(r'S-\d+', str(codigo_produto))
   133→   133→    return m.group(0) if m else str(codigo_produto).strip()

   146→   146→def _pg_query_one(sql, params=None):
   147→   147→    """Query PostgreSQL and return one row as dict."""
   148→   148→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   149→   149→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   150→   150→    conn.autocommit = True
   151→   151→    try:
   152→   152→        with conn.cursor() as cur:
   153→   153→            cur.execute(sql, params or ())
   154→   154→            cols = [d[0] for d in cur.description] if cur.description else []
   155→   155→            row = cur.fetchone()
   156→   156→            if row:
   157→   157→                return dict(zip(cols, row))
   158→   158→            return None
   159→   159→    finally:
   160→   160→        conn.close()

   163→   163→def _load_produto_cache():
   164→   164→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   165→   165→    global _produto_cache
   166→   166→    _produto_cache = {}
   167→   167→    # 1) PostgreSQL: tabela produto (pronome real)
   168→   168→    try:
   169→   169→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   170→   170→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   171→   171→        conn.autocommit = True
   172→   172→        try:
   173→   173→            with conn.cursor() as cur:
   174→   174→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   175→   175→                cols = [d[0] for d in cur.description]
   176→   176→                for row in cur.fetchall():
   177→   177→                    r = dict(zip(cols, row))
   178→   178→                    cod = str(r.get('produto', '')).strip()
   179→   179→                    nome = str(r.get('pronome', '')).strip()
   180→   180→                    if cod and nome:
   181→   181→                        _produto_cache[cod] = nome[:200]
   182→   182→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   183→   183→        finally:
   184→   184→            conn.close()
   185→   185→    except Exception as e:
   186→   186→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   187→   187→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   188→   188→    try:
   189→   189→        rows = db_query(
   190→   190→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   191→   191→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   192→   192→        )
   193→   193→        added = 0
   194→   194→        for r in rows:
   195→   195→            cod = str(r.get('codigo_produto', '')).strip()
   196→   196→            nome = str(r.get('descricao_produto', '')).strip()
   197→   197→            if cod and cod not in _produto_cache and nome:
   198→   198→                _produto_cache[cod] = nome[:200]
   199→   199→                added += 1
   200→   200→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   201→   201→    except Exception as e:
   202→   202→        print('[WARN] Cache MariaDB falhou: %s' % e)

   224→   224→def _build_lotes_filter():
   225→   225→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   226→   226→    if _lotes_ativos:
   227→   227→        ph = ','.join(['%s'] * len(_lotes_ativos))
   228→   228→        where = 'WHERE lote_codigo IN (' + ph + ')'
   229→   229→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   230→   230→        params = tuple(_lotes_ativos)
   231→   231→    else:
   232→   232→        where = 'WHERE 1=0'
   233→   233→        join = ' AND 1=0'
   234→   234→        params = ()
   235→   235→    return where, join, params

   783→   783→def _atualizar_departamento_lote(lote_ordem):
   784→   784→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   785→   785→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   786→   786→    if not lote_ordem:
   787→   787→        return
   788→   788→    # Total de departamentos unicos do lote
   789→   789→    total = db_query_one(
   790→   790→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   791→   791→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   792→   792→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   793→   793→        (lote_ordem,)
   794→   794→    )
   795→   795→    total_setores = int(total['t'] or 0) if total else 0
   796→   796→
   797→   797→    # Primeira operacao pendente (nao concluida) por sequencia
   798→   798→    primeira = db_query_one(
   799→   799→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   800→   800→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   801→   801→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   802→   802→        "ORDER BY op.sequencia ASC LIMIT 1",
   803→   803→        (lote_ordem,)
   804→   804→    )
   805→   805→    dept = primeira['departamento'] if primeira else None
   806→   806→    seq = int(primeira['sequencia']) if primeira else 0
   807→   807→
   808→   808→    db_execute(
   809→   809→        "UPDATE lotes_producao "
   810→   810→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   811→   811→        "WHERE ordem = %s",
   812→   812→        (dept, seq, total_setores, lote_ordem)
   813→   813→    )

   923→   923→                    continue
