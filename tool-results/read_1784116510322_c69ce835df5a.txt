   138→   138→def _int(val):
   139→   139→    """Safely convert to int (handles Decimal, None, str)."""
   140→   140→    if val is None:
   141→   141→        return 0
   142→   142→    return int(float(val))

   144→   144→def _str(val):
   145→   145→    """Safely convert to stripped string."""
   146→   146→    if val is None:
   147→   147→        return ''
   148→   148→    return str(val).strip()

   150→   150→def _is_sob_medida(codigo):
   151→   151→    """Verifica se o produto codigo termina com -X (sob medida)."""
   152→   152→    if not codigo:
   153→   153→        return False
   154→   154→    c = str(codigo).strip().upper()
   155→   155→    return c.endswith('-X')
   156→   156→
   157→   157→def _extract_s_code(codigo_produto):
   158→   158→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   159→   159→    if not codigo_produto:
   160→   160→        return ''
   161→   161→    m = re.search(r'S-\d+', str(codigo_produto))
   162→   162→    return m.group(0) if m else str(codigo_produto).strip()

   175→   175→def _pg_query_one(sql, params=None):
   176→   176→    """Query PostgreSQL and return one row as dict."""
   177→   177→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   178→   178→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   179→   179→    conn.autocommit = True
   180→   180→    try:
   181→   181→        with conn.cursor() as cur:
   182→   182→            cur.execute(sql, params or ())
   183→   183→            cols = [d[0] for d in cur.description] if cur.description else []
   184→   184→            row = cur.fetchone()
   185→   185→            if row:
   186→   186→                return dict(zip(cols, row))
   187→   187→            return None
   188→   188→    finally:
   189→   189→        conn.close()

   192→   192→def _load_produto_cache():
   193→   193→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   194→   194→    global _produto_cache
   195→   195→    _produto_cache = {}
   196→   196→    # 1) PostgreSQL: tabela produto (pronome real)
   197→   197→    try:
   198→   198→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   199→   199→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   200→   200→        conn.autocommit = True
   201→   201→        try:
   202→   202→            with conn.cursor() as cur:
   203→   203→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   204→   204→                cols = [d[0] for d in cur.description]
   205→   205→                for row in cur.fetchall():
   206→   206→                    r = dict(zip(cols, row))
   207→   207→                    cod = str(r.get('produto', '')).strip()
   208→   208→                    nome = str(r.get('pronome', '')).strip()
   209→   209→                    if cod and nome:
   210→   210→                        _produto_cache[cod] = nome[:200]
   211→   211→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   212→   212→        finally:
   213→   213→            conn.close()
   214→   214→    except Exception as e:
   215→   215→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   216→   216→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   217→   217→    try:
   218→   218→        rows = db_query(
   219→   219→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   220→   220→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   221→   221→        )
   222→   222→        added = 0
   223→   223→        for r in rows:
   224→   224→            cod = str(r.get('codigo_produto', '')).strip()
   225→   225→            nome = str(r.get('descricao_produto', '')).strip()
   226→   226→            if cod and cod not in _produto_cache and nome:
   227→   227→                _produto_cache[cod] = nome[:200]
   228→   228→                added += 1
   229→   229→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   230→   230→    except Exception as e:
   231→   231→        print('[WARN] Cache MariaDB falhou: %s' % e)
   232→   232→

   236→   236→def _refresh_lotes_ativos():
   237→   237→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""

   244→   244→        _lotes_ativos = set()
   245→   245→        for r in rows:
   246→   246→            if r.get('lote_codigo'):
   247→   247→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   248→   248→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   249→   249→    except Exception as e:
   250→   250→        print('[WARN] Lotes ativos falhou: %s' % e)

   253→   253→def _build_lotes_filter():
   254→   254→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   255→   255→    if _lotes_ativos:
   256→   256→        ph = ','.join(['%s'] * len(_lotes_ativos))
   257→   257→        where = 'WHERE lote_codigo IN (' + ph + ')'
   258→   258→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   259→   259→        params = tuple(_lotes_ativos)
   260→   260→    else:
   261→   261→        where = 'WHERE 1=0'
   262→   262→        join = ' AND 1=0'
   263→   263→        params = ()
   264→   264→    return where, join, params

   775→   775→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   776→   776→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   777→   777→    if not lote_ordem:

   781→   781→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   782→   782→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   783→   783→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   784→   784→        (lote_ordem,)

   789→   789→    primeira = db_query_one(
   790→   790→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   791→   791→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "

   794→   794→        (lote_ordem,)
   795→   795→    )

   797→   797→    seq = int(primeira['sequencia']) if primeira else 0
   798→   798→
   799→   799→    db_execute(
   800→   800→        "UPDATE lotes_producao "
   801→   801→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   802→   802→        "WHERE ordem = %s",
   803→   803→        (dept, seq, total_setores, lote_ordem)
