   151→   151→def _is_sob_medida(codigo):

   158→   158→def _extract_s_code(codigo_produto):
   159→   159→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   160→   160→    if not codigo_produto:
   161→   161→        return ''
   162→   162→    m = re.search(r'S-\d+', str(codigo_produto))
   163→   163→    return m.group(0) if m else str(codigo_produto).strip()

   176→   176→def _pg_query_one(sql, params=None):
   177→   177→    """Query PostgreSQL and return one row as dict."""
   178→   178→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   179→   179→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   180→   180→    conn.autocommit = True
   181→   181→    try:
   182→   182→        with conn.cursor() as cur:
   183→   183→            cur.execute(sql, params or ())
   184→   184→            cols = [d[0] for d in cur.description] if cur.description else []
   185→   185→            row = cur.fetchone()
   186→   186→            if row:
   187→   187→                return dict(zip(cols, row))
   188→   188→            return None
   189→   189→    finally:
   190→   190→        conn.close()

   193→   193→def _load_produto_cache():
   194→   194→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   195→   195→    global _produto_cache
   196→   196→    _produto_cache = {}
   197→   197→    # 1) PostgreSQL: tabela produto (pronome real)
   198→   198→    try:
   199→   199→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   200→   200→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   201→   201→        conn.autocommit = True
   202→   202→        try:
   203→   203→            with conn.cursor() as cur:
   204→   204→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   205→   205→                cols = [d[0] for d in cur.description]
   206→   206→                for row in cur.fetchall():
   207→   207→                    r = dict(zip(cols, row))
   208→   208→                    cod = str(r.get('produto', '')).strip()
   209→   209→                    nome = str(r.get('pronome', '')).strip()
   210→   210→                    if cod and nome:
   211→   211→                        _produto_cache[cod] = nome[:200]
   212→   212→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   213→   213→        finally:
   214→   214→            conn.close()
   215→   215→    except Exception as e:
   216→   216→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   217→   217→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   218→   218→    try:
   219→   219→        rows = db_query(
   220→   220→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   221→   221→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   222→   222→        )
   223→   223→        added = 0
   224→   224→        for r in rows:
   225→   225→            cod = str(r.get('codigo_produto', '')).strip()
   226→   226→            nome = str(r.get('descricao_produto', '')).strip()
   227→   227→            if cod and cod not in _produto_cache and nome:
   228→   228→                _produto_cache[cod] = nome[:200]
   229→   229→                added += 1
   230→   230→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   231→   231→    except Exception as e:
   232→   232→        print('[WARN] Cache MariaDB falhou: %s' % e)

   254→   254→def _build_lotes_filter():
   255→   255→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   256→   256→    if _lotes_ativos:
   257→   257→        ph = ','.join(['%s'] * len(_lotes_ativos))
   258→   258→        where = 'WHERE lote_codigo IN (' + ph + ')'
   259→   259→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   260→   260→        params = tuple(_lotes_ativos)
   261→   261→    else:
   262→   262→        where = 'WHERE 1=0'
   263→   263→        join = ' AND 1=0'
   264→   264→        params = ()
   265→   265→    return where, join, params

   872→   872→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   873→   873→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   874→   874→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   875→   875→            (lote_ordem,)
   876→   876→        )
   877→   877→        if total_pend and int(total_pend['t']) == 0:
   878→   878→            db_execute(
   879→   879→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   880→   880→                (lote_ordem,)
