   113→   113→    """Safely convert to stripped string."""

   118→   118→def _is_sob_medida(codigo):
   119→   119→    """Verifica se o produto codigo termina com -X (sob medida)."""
   120→   120→    if not codigo:
   121→   121→        return False
   122→   122→    c = str(codigo).strip().upper()
   123→   123→    return c.endswith('-X')

   125→   125→def _extract_s_code(codigo_produto):
   126→   126→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   127→   127→    if not codigo_produto:
   128→   128→        return ''
   129→   129→    m = re.search(r'S-\d+', str(codigo_produto))
   130→   130→    return m.group(0) if m else str(codigo_produto).strip()

   143→   143→def _pg_query_one(sql, params=None):
   144→   144→    """Query PostgreSQL and return one row as dict."""
   145→   145→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   146→   146→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   147→   147→    conn.autocommit = True
   148→   148→    try:
   149→   149→        with conn.cursor() as cur:
   150→   150→            cur.execute(sql, params or ())
   151→   151→            cols = [d[0] for d in cur.description] if cur.description else []
   152→   152→            row = cur.fetchone()
   153→   153→            if row:
   154→   154→                return dict(zip(cols, row))
   155→   155→            return None
   156→   156→    finally:
   157→   157→        conn.close()

   160→   160→def _load_produto_cache():
   161→   161→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   162→   162→    global _produto_cache
   163→   163→    _produto_cache = {}
   164→   164→    # 1) PostgreSQL: tabela produto (pronome real)
   165→   165→    try:
   166→   166→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   167→   167→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   168→   168→        conn.autocommit = True
   169→   169→        try:
   170→   170→            with conn.cursor() as cur:
   171→   171→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   172→   172→                cols = [d[0] for d in cur.description]
   173→   173→                for row in cur.fetchall():
   174→   174→                    r = dict(zip(cols, row))
   175→   175→                    cod = str(r.get('produto', '')).strip()
   176→   176→                    nome = str(r.get('pronome', '')).strip()
   177→   177→                    if cod and nome:
   178→   178→                        _produto_cache[cod] = nome[:200]
   179→   179→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   180→   180→        finally:
   181→   181→            conn.close()
   182→   182→    except Exception as e:
   183→   183→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   184→   184→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   185→   185→    try:
   186→   186→        rows = db_query(
   187→   187→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   188→   188→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   189→   189→        )
   190→   190→        added = 0
   191→   191→        for r in rows:
   192→   192→            cod = str(r.get('codigo_produto', '')).strip()
   193→   193→            nome = str(r.get('descricao_produto', '')).strip()
   194→   194→            if cod and cod not in _produto_cache and nome:
   195→   195→                _produto_cache[cod] = nome[:200]
   196→   196→                added += 1
   197→   197→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   198→   198→    except Exception as e:
   199→   199→        print('[WARN] Cache MariaDB falhou: %s' % e)

   215→   215→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   216→   216→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))

   218→   218→        print('[WARN] Lotes ativos falhou: %s' % e)

   221→   221→def _build_lotes_filter():
   222→   222→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   223→   223→    if _lotes_ativos:
   224→   224→        ph = ','.join(['%s'] * len(_lotes_ativos))
   225→   225→        where = 'WHERE lote_codigo IN (' + ph + ')'
   226→   226→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   227→   227→        params = tuple(_lotes_ativos)
   228→   228→    else:
   229→   229→        where = 'WHERE 1=0'
   230→   230→        join = ' AND 1=0'
   231→   231→        params = ()
   232→   232→    return where, join, params

   742→   742→def _sync_op_status(of_numero, departamento, novo_status):
   743→   743→    """Sincroniza operacoes_producao e kanban_cards quando totem inicia/finaliza."""
   744→   744→    if not of_numero or not departamento:
   745→   745→        return
   746→   746→    try:
   747→   747→        # Atualiza operacoes_producao
   748→   748→        db_execute(
   749→   749→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   750→   750→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   751→   751→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   752→   752→            (novo_status, novo_status, of_numero, departamento)
   753→   753→        )
   754→   754→        # Atualiza kanban_cards
   755→   755→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   756→   756→        nova_etapa = etapa_map.get(novo_status)

   759→   759→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   760→   760→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   761→   761→                (nova_etapa, of_numero, departamento)
   762→   762→            )
   763→   763→    except Exception as e:
   764→   764→        app.logger.warning('Sync op status falhou: %s', e)
