   141→   141→    def get_lotes(self, status=None, departamento=None, prioridade=None, pagina=1, por_pagina=30):
   142→   142→        offset = (pagina - 1) * por_pagina
   143→   143→        sql = """
   144→   144→            SELECT l.*, 
   145→   145→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem) AS total_operacoes,
   146→   146→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem AND op.status = 'concluido') AS operacoes_concluidas
   147→   147→            FROM lotes_producao l
   148→   148→            WHERE 1=1
   149→   149→        """
   150→   150→        params = []
   151→   151→        if status:
   152→   152→            if isinstance(status, list):
   153→   153→                placeholders = ','.join(['%s'] * len(status))
   154→   154→                sql += f" AND l.status IN ({placeholders})"
   155→   155→                params.extend(status)
   156→   156→            else:
   157→   157→                sql += " AND l.status = %s"
   158→   158→                params.append(status)

   160→   160→            sql += " AND l.departamento_atual = %s"
   161→   161→            params.append(departamento)
   162→   162→        if prioridade:
   163→   163→            sql += " AND l.prioridade = %s"
   164→   164→            params.append(prioridade)
   165→   165→        sql += """
   166→   166→            ORDER BY
   167→   167→                FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'),
   168→   168→                l.data_previsao_erp ASC
   169→   169→            LIMIT %s OFFSET %s
   170→   170→        """
   171→   171→        params.extend([por_pagina, offset])

   184→   184→    def get_lotes_por_departamento(self, departamento, status=None):
   185→   185→        """
   186→   186→        Retorna 1 linha por OP/LOTE para o departamento, com a próxima operação pendente.
   187→   187→        Evita duplicação: pega só a operação mais antiga pendente/em_andamento do setor.
   188→   188→        """
   189→   189→        sql = """
   190→   190→            SELECT l.*,
   191→   191→                   op.id            AS op_id,
   192→   192→                   op.status        AS op_status,
   193→   193→                   op.data_inicio   AS data_inicio,
   194→   194→                   op.descricao_operacao,
   195→   195→                   op.qtde_produzida,
   196→   196→                   op.sequencia,
   197→   197→                   op.of_numero,
   198→   198→                   (SELECT COUNT(*) FROM operacoes_producao op2
   199→   199→                    WHERE op2.lote_ordem = l.ordem AND op2.departamento = %s
   200→   200→                      AND op2.status != 'concluido') AS ops_pendentes
   201→   201→            FROM lotes_producao l
   202→   202→            INNER JOIN operacoes_producao op ON op.lote_ordem = l.ordem
   203→   203→            WHERE op.departamento = %s
   204→   204→              AND op.status IN ('pendente', 'em_andamento', 'pausado')
   205→   205→              AND l.status NOT IN ('cancelado', 'finalizado')
   206→   206→              AND op.id = (
   207→   207→                  SELECT MIN(op3.id)
   208→   208→                  FROM operacoes_producao op3
   209→   209→                  WHERE op3.lote_ordem = l.ordem
   210→   210→                    AND op3.departamento = %s
   211→   211→                    AND op3.status IN ('pendente', 'em_andamento', 'pausado')
   212→   212→              )
   213→   213→        """
   214→   214→        params = [departamento, departamento, departamento]
   215→   215→        if status:
   216→   216→            sql += " AND l.status = %s"
   217→   217→            params.append(status)
   218→   218→        sql += " ORDER BY FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'), l.data_previsao_erp ASC"
   219→   219→        return self.query(sql, params)

   390→   390→        """Atualiza o card do kanban correspondente à operação.
   391→   391→        Se o card não existir, cria automaticamente."""
   392→   392→        try:
   393→   393→            # Busca a operação e o lote
   394→   394→            op = self.query_one("""
   395→   395→                SELECT op.*, l.prioridade
   396→   396→                FROM operacoes_producao op
   397→   397→                LEFT JOIN lotes_producao l ON op.lote_ordem = l.ordem
   398→   398→                WHERE op.id = %s
   399→   399→            """, (operacao_id,))
   400→   400→            if not op:
   401→   401→                return
   402→   402→
   403→   403→            # Verifica se já existe card para esta operação
   404→   404→            existente = self.query_one(
   405→   405→                "SELECT id FROM kanban_cards WHERE operacao_id = %s",
   406→   406→                (operacao_id,)
   407→   407→            )
   408→   408→
   409→   409→            if existente:
   410→   410→                # Atualiza card existente
   411→   411→                self.execute("""
   412→   412→                    UPDATE kanban_cards
   413→   413→                    SET etapa = %s,
   414→   414→                        operador_id = COALESCE(%s, operador_id),
   415→   415→                        atualizado_em = NOW()
   416→   416→                    WHERE operacao_id = %s
   417→   417→                """, (etapa, operador_id, operacao_id))
   418→   418→            else:
   419→   419→                # Cria card novo (para operações importadas antes do fix)
   420→   420→                self.execute("""
   421→   421→                    INSERT INTO kanban_cards
   422→   422→                        (lote_ordem, of_numero, operacao_id, etapa, departamento, prioridade, operador_id)
   423→   423→                    VALUES (%s, %s, %s, %s, %s, %s, %s)
   424→   424→                """, (
   425→   425→                    op['lote_ordem'],
   426→   426→                    op.get('of_numero'),
   427→   427→                    operacao_id,
   428→   428→                    etapa,
   429→   429→                    op.get('departamento'),
   430→   430→                    op.get('prioridade') or 'media',
   431→   431→                    operador_id
   432→   432→                ))
   433→   433→        except Exception as e:
   434→   434→            print(f"Erro ao sincronizar kanban: {e}")

   436→   436→    def finalizar_operacao(self, operacao_id, usuario_id, qtde_produzida=None, qtde_refugada=0, observacoes=None, tipo='total'):
   437→   437→        """
   438→   438→        tipo='total'   → marca como concluido e fecha o lote se todas as ops concluídas
   439→   439→        tipo='parcial' → registra a quantidade produzida até agora e reinicia a operação
   440→   440→                         (status volta para 'pendente' com qtde_produzida acumulada)
   441→   441→                         Assim o próximo setor já pode ver as peças disponíveis.
   442→   442→        """
   443→   443→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   444→   444→        if not op:
   445→   445→            return False
   446→   446→
   447→   447→        if tipo == 'parcial':
   448→   448→            # Acumula a quantidade e volta para pendente (disponível para próximo setor)
   449→   449→            self.execute("""
   450→   450→                UPDATE operacoes_producao
   451→   451→                SET status = 'pendente',
   452→   452→                    data_inicio = NULL,
   453→   453→                    qtde_produzida = COALESCE(qtde_produzida, 0) + COALESCE(%s, 0),
   454→   454→                    qtde_refugada  = COALESCE(qtde_refugada, 0)  + COALESCE(%s, 0),
   455→   455→                    observacoes    = COALESCE(%s, observacoes)
   456→   456→                WHERE id = %s
   457→   457→            """, (qtde_produzida, qtde_refugada, observacoes, operacao_id))
   458→   458→            # Atualiza status do lote para 'importado' (disponível) se estava em_producao
   459→   459→            self.execute("""
   460→   460→                UPDATE lotes_producao SET status = 'importado'
   461→   461→                WHERE ordem = %s AND status = 'em_producao'
   462→   462→            """, (op['lote_ordem'],))
   463→   463→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'parcial', qtde_produzida=qtde_produzida)
   464→   464→            # Sincroniza kanban: card volta para 'aguardando' (próximo setor pega)
   465→   465→            self._sincronizar_kanban_operacao(operacao_id, 'aguardando')
   466→   466→        else:
   467→   467→            self.execute("""
   468→   468→                UPDATE operacoes_producao
   469→   469→                SET status = 'concluido', data_fim = NOW(), usuario_fim_id = %s,
   470→   470→                    qtde_produzida = COALESCE(%s, qtde_produzida),
   471→   471→                    qtde_refugada = COALESCE(%s, 0),
   472→   472→                    observacoes = COALESCE(%s, observacoes)
   473→   473→                WHERE id = %s
   474→   474→            """, (usuario_id, qtde_produzida, qtde_refugada, observacoes, operacao_id))
   475→   475→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'fim', qtde_produzida=qtde_produzida)
   476→   476→            # Sincroniza kanban: card vai para 'concluido'
   477→   477→            self._sincronizar_kanban_operacao(operacao_id, 'concluido', usuario_id)
   478→   478→            # Verifica se todas as operações do lote foram concluídas
   479→   479→            self._verificar_conclusao_lote(op['lote_ordem'])
   480→   480→        return True

   844→   844→    def get_dashboard_gerente(self):
   845→   845→        """Retorna todos os dados do dashboard gerencial em poucas queries."""
   846→   846→        try:
   847→   847→            # 1. KPIs principais
   848→   848→            kpis = self.query_one("""
   849→   849→                SELECT
   850→   850→                    COUNT(*) AS total_lotes,
   851→   851→                    SUM(status = 'importado')     AS aguardando_liberacao,
   852→   852→                    SUM(status = 'liberado')      AS aguardando_inicio,
   853→   853→                    SUM(status = 'em_producao')   AS em_producao,
   854→   854→                    SUM(status = 'pausado')       AS pausados,
   855→   855→                    SUM(status = 'finalizado' AND DATE(data_fim_producao) = CURDATE()) AS concluidos_hoje,
   856→   856→                    SUM(status = 'finalizado' AND MONTH(data_fim_producao) = MONTH(CURDATE())
   857→   857→                                                AND YEAR(data_fim_producao)  = YEAR(CURDATE())) AS concluidos_mes,
   858→   858→                    SUM(prioridade = 'urgente' AND status NOT IN ('finalizado','cancelado')) AS urgentes,
   859→   859→                    SUM(prioridade = 'alta'     AND status NOT IN ('finalizado','cancelado')) AS altas,
   860→   860→                    SUM(data_previsao_erp IS NOT NULL
   861→   861→                        AND data_previsao_erp < CURDATE()
   862→   862→                        AND status NOT IN ('finalizado','cancelado')) AS atrasados
   863→   863→                FROM lotes_producao
   864→   864→            """)
   865→   865→
   866→   866→            # 2. Produção por setor (quantidade de operações por status)
   867→   867→            por_setor = self.query("""
   868→   868→                SELECT
   869→   869→                    d.codigo AS departamento,
   870→   870→                    d.nome   AS nome_departamento,
   871→   871→                    d.cor_hex,
   872→   872→                    d.ordem_fluxo,
   873→   873→                    COUNT(op.id) AS total_ops,
   874→   874→                    SUM(op.status = 'pendente')     AS pendentes,
   875→   875→                    SUM(op.status = 'em_andamento') AS em_andamento,
   876→   876→                    SUM(op.status = 'pausado')      AS pausados,
   877→   877→                    SUM(op.status = 'concluido' AND DATE(op.data_fim) = CURDATE()) AS concluidas_hoje
   878→   878→                FROM departamentos d
   879→   879→                LEFT JOIN operacoes_producao op ON op.departamento = d.codigo
   880→   880→                WHERE d.ativo = 1 AND d.ordem_fluxo > 0
   881→   881→                GROUP BY d.codigo, d.nome, d.cor_hex, d.ordem_fluxo
   882→   882→                ORDER BY d.ordem_fluxo
   883→   883→            """)
   884→   884→
   885→   885→            # 3. Produção dos últimos 7 dias (finalizados por dia)
   886→   886→            por_dia = self.query("""
   887→   887→                SELECT
   888→   888→                    DATE(data_fim_producao) AS dia,
   889→   889→                    COUNT(*) AS total,
   890→   890→                    SUM(qtde_ordem) AS pecas
   891→   891→                FROM lotes_producao
   892→   892→                WHERE status = 'finalizado'
   893→   893→                  AND data_fim_producao >= DATE_SUB(CURDATE(), INTERVAL 6 DAY)
   894→   894→                GROUP BY DATE(data_fim_producao)
   895→   895→                ORDER BY dia
   896→   896→            """)
   897→   897→
   898→   898→            # 4. Próximas OPs a iniciar (liberadas + aguardando, com prioridade)
   899→   899→            proximas = self.query("""
   900→   900→                SELECT ordem, lote_codigo, descricao_produto, qtde_ordem,
   901→   901→                       prioridade, data_previsao_erp, status
   902→   902→                FROM lotes_producao
   903→   903→                WHERE status IN ('liberado', 'importado')
   904→   904→                ORDER BY FIELD(prioridade, 'urgente','alta','media','baixa'),
   905→   905→                         data_previsao_erp ASC
   906→   906→                LIMIT 8
   907→   907→            """)
   908→   908→
   909→   909→            # 5. Em produção agora
   910→   910→            em_producao = self.query("""
   911→   911→                SELECT ordem, lote_codigo, descricao_produto, qtde_ordem,
   912→   912→                       prioridade, data_inicio_producao, departamento_atual
   913→   913→                FROM lotes_producao
   914→   914→                WHERE status = 'em_producao'
   915→   915→                ORDER BY data_inicio_producao ASC
   916→   916→                LIMIT 8
   917→   917→            """)
   918→   918→
   919→   919→            # 6. TOP atrasados (alertas)
   920→   920→            atrasados = self.query("""
   921→   921→                SELECT ordem, lote_codigo, descricao_produto, qtde_ordem,
   922→   922→                       prioridade, data_previsao_erp, status, departamento_atual
   923→   923→                FROM lotes_producao
   924→   924→                WHERE data_previsao_erp IS NOT NULL
   925→   925→                  AND data_previsao_erp < CURDATE()
   926→   926→                  AND status NOT IN ('finalizado','cancelado')
   927→   927→                ORDER BY FIELD(prioridade, 'urgente','alta','media','baixa'),
   928→   928→                         data_previsao_erp ASC
   929→   929→                LIMIT 10
   930→   930→            """)
   931→   931→
   932→   932→            return {
   933→   933→                'kpis': {
   934→   934→                    'total_lotes':          int(kpis['total_lotes']          or 0),
   935→   935→                    'aguardando_liberacao': int(kpis['aguardando_liberacao'] or 0),
   936→   936→                    'aguardando_inicio':    int(kpis['aguardando_inicio']    or 0),
   937→   937→                    'em_producao':          int(kpis['em_producao']          or 0),
   938→   938→                    'pausados':             int(kpis['pausados']             or 0),
   939→   939→                    'concluidos_hoje':      int(kpis['concluidos_hoje']      or 0),
   940→   940→                    'concluidos_mes':       int(kpis['concluidos_mes']       or 0),
   941→   941→                    'urgentes':             int(kpis['urgentes']             or 0),
   942→   942→                    'altas':                int(kpis['altas']                or 0),
   943→   943→                    'atrasados':            int(kpis['atrasados']            or 0),

   946→   946→                'por_dia':        por_dia,

   955→   955→                         'pausados':0,'concluidos_hoje':0,'concluidos_mes':0,'urgentes':0,'altas':0,'atrasados':0},
