    28→def _load_produto_cache():
    29→    """Carrega nomes reais dos produtos do PostgreSQL."""
    30→    global _produto_cache
    31→    try:
    32→        pg = DatabasePostgreSQL()
    33→        rows = pg.query(
    34→            "SELECT produto, pronome FROM public.produto "
    35→            "WHERE pronome IS NOT NULL AND pronome != ''"
    36→        )
    37→        _produto_cache = {}
    38→        for r in rows:
    39→            cod = str(r.get('produto', '')).strip()
    40→            nome = str(r.get('pronome', '')).strip()
    41→            if cod:
    42→                _produto_cache[cod] = nome[:200]
    43→        print('[OK] Cache de produtos carregado: %d produtos do ERP' % len(_produto_cache))
    44→    except Exception as e:
    45→        print('[WARN] Cache de produtos falhou: %s' % e)

    47→def _get_nome_produto(codigo):
    48→    """Retorna nome real do produto pelo codigo."""
    49→    if not codigo:
    50→        return ''
    51→    return _produto_cache.get(str(codigo).strip(), '')

    53→def _is_sob_medida(codigo):
    54→    """Verifica se o produto codigo termina com -X (sob medida)."""
    55→    if not codigo:
    56→        return False
    57→    c = str(codigo).strip().upper()
    58→    # S-0440-X, CJ-0174-X, qualquer codigo terminado em -X
    59→    return c.endswith('-X')

    61→def _resolver_nome_lote(lote):
    62→    """Resolve nome real para descricao_produto de um lote/OF."""
    63→    desc = lote.get('descricao_produto', '') or ''
    64→    cod = lote.get('codigo_produto', '') or ''
    65→    if not desc or desc.upper().startswith('PRODUCAO DE'):
    66→        nome = _get_nome_produto(cod)
    67→        if nome:
    68→            return nome
    69→    return desc

   120→@app.template_filter('formatar_tempo')
   121→def formatar_tempo(segundos):
   122→    """Formata segundos em HH:MM:SS"""
   123→    try:
   124→        seg = int(segundos or 0)
   125→        h = seg // 3600
   126→        m = (seg % 3600) // 60
   127→        s = seg % 60
   128→        return f'{h:02d}:{m:02d}:{s:02d}'
   129→    except Exception:
   130→        return '00:00:00'

   132→@app.template_filter('formatar_data_br')
   133→def formatar_data_br(value):
   134→    """Formata datetime/date para dd/mm/yyyy HH:MM"""
   135→    if not value:
   136→        return '-'
   137→    try:
   138→        if hasattr(value, 'strftime'):
   139→            if hasattr(value, 'hour'):
   140→                return value.strftime('%d/%m/%Y %H:%M')
   141→            return value.strftime('%d/%m/%Y')
   142→        return str(value)
   143→    except Exception:
   144→        return str(value)

   146→# =============================================
   147→# Proteção contra força bruta no login
   148→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
   149→# =============================================
   150→
   151→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
   152→MAX_TENTATIVAS  = 5
   153→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
   154→
   155→def _get_ip():
   156→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
   157→
   158→def _checar_bloqueio(ip):
   159→    dados = _login_attempts[ip]
   160→    if time.time() < dados['bloqueado_ate']:
   161→        restante = int(dados['bloqueado_ate'] - time.time())
   162→        return True, restante
   163→    return False, 0
   164→
   165→def _registrar_falha(ip):
   166→    dados = _login_attempts[ip]
   167→    dados['tentativas'] += 1
   168→    if dados['tentativas'] >= MAX_TENTATIVAS:
   169→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   170→        dados['tentativas'] = 0
   171→
   172→def _limpar_tentativas(ip):
   173→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}
   174→

   179→def login_required(f):
   180→    @wraps(f)
   181→    def decorated(*args, **kwargs):
   182→        if 'usuario_id' not in session:
   183→            flash('Faça login para continuar.', 'warning')
   184→            return redirect(url_for('login'))
   185→        return f(*args, **kwargs)
   186→    return decorated
   187→
   188→def role_required(*roles):
   189→    def decorator(f):
   190→        @wraps(f)
   191→        def decorated(*args, **kwargs):
   192→            if session.get('usuario_role') not in roles:
   193→                flash('Acesso não autorizado.', 'error')
   194→                return redirect(url_for('dashboard'))
   195→            return f(*args, **kwargs)
   196→        return decorated
   197→    return decorator

   203→@app.route('/login', methods=['GET', 'POST'])
   204→def login():
   205→    if 'usuario_id' in session:
   206→        return redirect(url_for('dashboard'))
   207→
   208→    if request.method == 'POST':
   209→        ip = _get_ip()
   210→
   211→        bloqueado, restante = _checar_bloqueio(ip)
   212→        if bloqueado:
   213→            mins = restante // 60
   214→            secs = restante % 60
   215→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   216→            return render_template('login.html')
   217→
   218→        usuario = request.form.get('usuario', '').strip()
   219→        senha   = request.form.get('senha', '')
   220→
   221→        if not usuario or not senha:
   222→            flash('Informe usuário e senha.', 'error')
   223→            return render_template('login.html')
   224→
   225→        user = db.verificar_usuario(usuario, senha)
   226→
   227→        if user:
   228→            _limpar_tentativas(ip)
   229→            session.permanent = True
   230→            session['usuario_id']          = user['id']
   231→            session['usuario_nome']        = user['nome']
   232→            session['usuario_role']        = user['role']
   233→            session['usuario_departamento']= user.get('nome_departamento', '')
   234→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   235→            return redirect(url_for('dashboard'))
   236→        else:
   237→            _registrar_falha(ip)
   238→            dados   = _login_attempts[ip]
   239→            restantes = MAX_TENTATIVAS - dados['tentativas']
   240→            if restantes > 0:
   241→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   242→            else:
   243→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   244→
   245→    return render_template('login.html')

   380→    # Agrupar cards por departamento
   381→    cards_por_setor = {}
   382→    for dept in departamentos:
   383→        cards_por_setor[dept['codigo']] = {
   384→            'info': dept,
   385→            'cards': [],
   386→            'total': 0,
   387→            'em_producao': 0,
   388→            'aguardando': 0,
   389→            'pausados': 0,
   390→            'concluidos': 0,
   391→        }

   393→    for card in todos_cards:
   394→        dept_code = card.get('departamento', '')
   395→        if dept_code in cards_por_setor:
   396→            cards_por_setor[dept_code]['cards'].append(card)
   397→            cards_por_setor[dept_code]['total'] += 1
   398→            etapa = card.get('etapa', '')
   399→            if etapa == 'em_producao':
   400→                cards_por_setor[dept_code]['em_producao'] += 1
   401→            elif etapa in ('aguardando', 'pendente'):
   402→                cards_por_setor[dept_code]['aguardando'] += 1
   403→            elif etapa == 'pausado':
   404→                cards_por_setor[dept_code]['pausados'] += 1
   405→            elif etapa == 'concluido':
   406→                cards_por_setor[dept_code]['concluidos'] += 1
   407→
   408→    # KPIs gerais
   409→    kpis = {
   410→        'setores_ativos': sum(1 for s in cards_por_setor.values() if s['total'] > 0),
   411→        'total_cards': len(todos_cards),
   412→        'em_producao': sum(1 for c in todos_cards if c.get('etapa') == 'em_producao'),
   413→        'concluidos_hoje': db.get_kanban_concluidos_hoje(),
   414→    }
   415→
   416→    return render_template('kanban/setores.html',
   417→                           cards_por_setor=cards_por_setor,

   466→

   471→        # Para cada lote busca o produto final

   474→            lote['nome_produto'] = prod.get('nome_produto') if prod else None
   475→            # produto_final_codigo ja vem correto do ordem3, so sobrescreve se pronome foi encontrado

   477→                lote['produto_final_codigo'] = prod['codigo_produto']
   478→            lote['maior_ordem'] = prod.get('quantidade') if prod else None
   479→            # busca a maior ordem numerica do lote
   480→            ordens = pg.get_ordens_por_lote(lote['lote_codigo'])
   481→            if ordens:
   482→                lote['maior_ordem'] = max(int(o['ordem']) for o in ordens if o.get('ordem') and str(o['ordem']).isdigit())
   483→            lote['sob_medida'] = _is_sob_medida(lote.get('produto_final_codigo')) or _is_sob_medida(lote.get('lote_codigo'))

   485→        pg_ok = True
   486→    except Exception as e:
   487→        pg_erro = str(e)
   488→
   489→    # Lotes já importados no MariaDB
   490→    ja = db.query("SELECT DISTINCT lote_codigo FROM lotes_producao WHERE lote_codigo IS NOT NULL")
   491→    lotes_ja_importados = {r['lote_codigo'] for r in ja}

   494→                           pg_ok=pg_ok, pg_erro=pg_erro,

   529→            ja_existia = existente is not None
   530→
   531→            # Importa lote principal
   532→            db.importar_lote_erp({
   533→                'ordem':             int(ordem_num) if str(ordem_num).isdigit() else 0,
   534→                'lote_codigo':       lote_cod,
   535→                'codigo_produto':    item.get('produto',''),
   536→                'descricao_produto': item.get('nome','') or item.get('produto',''),
   537→                'qtde_ordem':        float(item.get('qtd', 0) or 0),
   538→                'unidade_medida':    'PC',
   539→                'status_erp':        'A',
   540→                'data_previsao_erp': item.get('previsao') or None,
   541→                'data_abertura_erp': item.get('abertura') or None,
   542→                'planejador':        session.get('usuario_nome'),
   543→            })
   544→
   545→            lote_ordem_local = int(ordem_num) if str(ordem_num).isdigit() else 0
   546→
   547→            # Busca ordens do lote no ERP e importa OFs
   548→            ordens_erp = pg.get_ordens_por_lote(lote_cod)
   549→            for of in ordens_erp:
   550→                of_num = of.get('ordem')
   551→                if not of_num:
   552→                    continue
   553→                db.importar_of({
   554→                    'lote_ordem':       lote_ordem_local,
   555→                    'of_numero':        str(of_num),
   556→                    'codigo_produto':   of.get('produto',''),
   557→                    'descricao_produto': item.get('nome','') or item.get('produto',''),
   558→                    'qtde_ordem':       float(of.get('quantidade') or 0),
   559→                    'unidade_medida':   'PC',
   560→                    'tipo':             'filho' if of.get('nivel_producao','') != '1' else 'pai',
   561→                    'data_previsao':    of.get('data_previsao') or None,
   562→                })
   563→
   564→                # Busca processos (operações) de cada OF
   565→                processos = pg.get_processos_ordem(str(of_num))
   566→                for proc in processos:
   567→                    dept_original = (proc.get('nome_departamento') or '').upper().strip()
   568→                    if not dept_original:
   569→                        continue
   570→                    # Mapeia nomes do ERP (fases) para nomes do sistema local
   571→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   572→                    db.importar_operacao({
   573→                        'of_numero':          str(of_num),
   574→                        'lote_ordem':         lote_ordem_local,
   575→                        'sequencia':          proc.get('numero_sequencial', 0),
   576→                        'fase':               proc.get('fase',''),
   577→                        'descricao_operacao': proc.get('descricao_processo') or proc.get('operacao',''),
   578→                        'departamento':       dept,
   579→                        'codigo_barras':      None,
   580→                    })
   581→
   582→                # Sincroniza kanban
   583→                try:
   584→                    db.sincronizar_kanban_com_operacoes(lote_ordem_local)
   585→                except Exception:
   586→                    pass
   587→
   588→                # Atualiza departamento_atual baseado na primeira op pendente
   589→                try:
   590→                    db.atualizar_departamento_atual(lote_ordem_local)
   591→                except Exception:
   592→                    pass
   593→
   594→            db.add_log(session['usuario_id'], 'importar_erp',
   595→                       f'Lote {lote_cod} importado do ERP', 'lotes_producao', lote_ordem_local)
   596→
   597→            if ja_existia:
   598→                atualizados += 1
   599→            else:
   600→                importados += 1
   601→
   602→        except Exception as e:
   603→            erros.append(f"{item.get('lote_codigo')}: {e}")
   604→
   605→    return jsonify({
   606→        'status':     'success',
   607→        'importados':  importados,
   608→        'atualizados': atualizados,
   609→        'erros':       erros,
   610→    })

   957→def _baixar_estoque_por_of(ordem):
   958→    """Busca materiais da OF no ERP e da baixa automatica no estoque da fabrica."""
   959→    try:
   960→        pg = DatabasePostgreSQL()
   961→        materiais = pg.get_requisicoes_ordem(ordem)
   962→        if not materiais:
   963→            return
   964→        for mat in materiais:
   965→            codigo = str(mat.get('produto') or mat.get('reqproduto') or '')
   966→            if not codigo:
   967→                continue
   968→            qtd = float(mat.get('quantidade') or mat.get('rqoquanti') or 0)
   969→            if qtd <= 0:
   970→                continue
   971→            item = db.get_estoque_por_codigo(codigo, tipo='fabrica')
   972→            if item and item['qtd_atual'] > 0:
   973→                db.registrar_baixa_estoque(codigo, qtd, ordem, 'liberacao_of', tipo='fabrica')
   974→                print(f"[ESTOQUE] Baixa fabrica: {codigo} -{qtd} (OP #{ordem})")
   975→    except Exception as e:
   976→        print(f"[ESTOQUE] Erro ao baixar estoque da OP #{ordem}: {e}")

  1057→@login_required

  1059→def api_materiais_pendentes():

  1061→    lotes = db.get_lotes_liberados_pendentes()
  1062→    if not lotes:
  1063→        return jsonify({'status': 'success', 'ofs': []})
  1064→    ordens = [l['ordem'] for l in lotes]

  1067→        materiais_erp = pg.get_requisicoes_multiplas_ordens(ordens)
  1068→    except Exception:
  1069→        materiais_erp = []
  1070→
  1071→    # Pega todos os codigo_produto das OFs ativas para excluir peças fabricadas do BOM
  1072→    codigos_fabricados = db.get_codigos_produtos_ofs_ativas()
  1073→
  1074→    mats_por_of = {}
  1075→    for m in materiais_erp:
  1076→        of_num = m.get('ordem')
  1077→        if of_num not in mats_por_of:
  1078→            mats_por_of[of_num] = []
  1079→        mats_por_of[of_num].append(m)
  1080→    todos_codigos = list(set(str(m.get('produto') or '') for m in materiais_erp if m.get('produto')))
  1081→    estoque_map = {}
  1082→    for cod in todos_codigos:
  1083→        item = db.get_estoque_por_codigo(cod, tipo='almoxarifado')
  1084→        estoque_map[cod] = item
  1085→    resultado = []
  1086→    for lote in lotes:
  1087→        mats = mats_por_of.get(lote['ordem'], [])
  1088→        mats_com = []
  1089→        for m in mats:
  1090→            cod = str(m.get('produto') or '')
  1091→            # Pula peças que são fabricadas em outras OFs (ex: alça, suporte, etc.)
  1092→            if cod in codigos_fabricados:
  1093→                continue
  1094→            est = estoque_map.get(cod)
  1095→            mats_com.append({
  1096→                'codigo': cod,
  1097→                'descricao': m.get('descricao') or '',
  1098→                'unidade': m.get('unidade') or 'UN',
  1099→                'qtd_necessaria': float(m.get('quantidade') or 0),
  1100→                'tem_no_estoque': est is not None and est['qtd_atual'] > 0,
  1101→                'qtd_disponivel': float(est['qtd_atual']) if est else 0
  1102→            })
  1103→        # So inclui a OF se tiver materiais brutos
  1104→        if mats_com:
  1105→            resultado.append({
  1106→                'ordem': lote['ordem'],
  1107→                'produto': lote.get('descricao_produto') or lote.get('lote_descricao') or '',
  1108→                'quantidade': lote.get('quantidade'),
  1109→                'prioridade': lote.get('prioridade'),
  1110→                'departamento': lote.get('departamento'),
  1111→                'materiais': mats_com
  1112→            })
  1113→    return jsonify({'status': 'success', 'ofs': resultado})
