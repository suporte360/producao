    12→
    13→class DatabaseMySQL:
    14→    _initialized = False
    15→    _instance = None
    16→
    17→    def __new__(cls):
    18→        if cls._instance is None:
    19→            cls._instance = super(DatabaseMySQL, cls).__new__(cls)
    20→        return cls._instance
    21→
    22→    def __init__(self):
    23→        if self._initialized:
    24→            return
    25→        self._initialized = True
    26→        self.config = {
    27→            'host': Config.MYSQL_HOST,
    28→            'port': Config.MYSQL_PORT,
    29→            'user': Config.MYSQL_USERNAME,
    30→            'password': Config.MYSQL_PASSWORD,
    31→            'database': Config.MYSQL_DATABASE,
    32→            'charset': 'utf8mb4',
    33→            'cursorclass': DictCursor,
    34→            'autocommit': False,
    35→            'connect_timeout': 10,
    36→        }
    37→
    38→    @contextmanager
    39→    def get_connection(self):
    40→        connection = pymysql.connect(**self.config)
    41→        try:
    42→            yield connection
    43→            connection.commit()
    44→        except Exception as e:
    45→            connection.rollback()
    46→            raise e
    47→        finally:
    48→            connection.close()
    49→
    50→    def query(self, sql, params=None):
    51→        with self.get_connection() as conn:
    52→            with conn.cursor() as cursor:
    53→                cursor.execute(sql, params or ())
    54→                return cursor.fetchall()
    55→
    56→    def query_one(self, sql, params=None):
    57→        with self.get_connection() as conn:
    58→            with conn.cursor() as cursor:
    59→                cursor.execute(sql, params or ())
    60→                return cursor.fetchone()
    61→
    62→    def execute(self, sql, params=None):
    63→        with self.get_connection() as conn:
    64→            with conn.cursor() as cursor:
    65→                cursor.execute(sql, params or ())
    66→                return cursor.lastrowid or cursor.rowcount

    73→        sql = "SELECT * FROM usuarios WHERE usuario = %s AND ativo = 1"
    74→        user = self.query_one(sql, (usuario,))
    75→        if user:
    76→            if user['senha'].startswith('scrypt:') or user['senha'].startswith('pbkdf2:'):
    77→                if check_password_hash(user['senha'], senha):
    78→                    self.execute("UPDATE usuarios SET ultimo_acesso = NOW() WHERE id = %s", (user['id'],))
    79→                    return user
    80→            elif user['senha'] == senha:
    81→                self.execute("UPDATE usuarios SET ultimo_acesso = NOW() WHERE id = %s", (user['id'],))
    82→                return user
    83→        return None

   141→    def get_lotes(self, status=None, departamento=None, prioridade=None, pagina=1, por_pagina=30):
   142→        offset = (pagina - 1) * por_pagina
   143→        sql = """
   144→            SELECT l.*, 
   145→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem) AS total_operacoes,
   146→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem AND op.status = 'concluido') AS operacoes_concluidas
   147→            FROM lotes_producao l
   148→            WHERE 1=1
   149→        """
   150→        params = []
   151→        if status:
   152→            if isinstance(status, list):
   153→                placeholders = ','.join(['%s'] * len(status))
   154→                sql += f" AND l.status IN ({placeholders})"
   155→                params.extend(status)
   156→            else:
   157→                sql += " AND l.status = %s"
   158→                params.append(status)
   159→        if departamento:
   160→            sql += " AND l.departamento_atual = %s"
   161→            params.append(departamento)
   162→        if prioridade:
   163→            sql += " AND l.prioridade = %s"
   164→            params.append(prioridade)
   165→        sql += """
   166→            ORDER BY
   167→                FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'),
   168→                l.data_previsao_erp ASC
   169→            LIMIT %s OFFSET %s
   170→        """
   171→        params.extend([por_pagina, offset])
   172→        return self.query(sql, params)

   184→    def get_lotes_por_departamento(self, departamento, status=None):
   185→        """
   186→        Retorna 1 linha por OP/LOTE para o departamento, com a próxima operação pendente.
   187→        Evita duplicação: pega só a operação mais antiga pendente/em_andamento do setor.
   188→        """
   189→        sql = """
   190→            SELECT l.*,
   191→                   op.id            AS op_id,
   192→                   op.status        AS op_status,
   193→                   op.data_inicio   AS data_inicio,
   194→                   op.descricao_operacao,
   195→                   op.qtde_produzida,
   196→                   op.sequencia,
   197→                   op.of_numero,
   198→                   (SELECT COUNT(*) FROM operacoes_producao op2
   199→                    WHERE op2.lote_ordem = l.ordem AND op2.departamento = %s
   200→                      AND op2.status != 'concluido') AS ops_pendentes
   201→            FROM lotes_producao l
   202→            INNER JOIN operacoes_producao op ON op.lote_ordem = l.ordem
   203→            WHERE op.departamento = %s
   204→              AND op.status IN ('pendente', 'em_andamento', 'pausado')
   205→              AND l.status NOT IN ('cancelado', 'finalizado')
   206→              AND op.id = (
   207→                  SELECT MIN(op3.id)
   208→                  FROM operacoes_producao op3
   209→                  WHERE op3.lote_ordem = l.ordem
   210→                    AND op3.departamento = %s
   211→                    AND op3.status IN ('pendente', 'em_andamento', 'pausado')
   212→              )
   213→        """
   214→        params = [departamento, departamento, departamento]
   215→        if status:
   216→            sql += " AND l.status = %s"
   217→            params.append(status)
   218→        sql += " ORDER BY FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'), l.data_previsao_erp ASC"
   219→        return self.query(sql, params)

   262→    def atualizar_departamento_atual(self, lote_ordem):
   263→        """Calcula e atualiza o departamento_atual, setor_atual_seq e total_setores
   264→        de um lote baseado na primeira operacao pendente (nao concluida)."""
   265→        # Busca OFs do lote
   266→        ofs = self.query(
   267→            "SELECT of_numero FROM ordens_fabricacao WHERE lote_ordem = %s",
   268→            (lote_ordem,)
   269→        )
   270→        if not ofs:
   271→            return
   272→
   273→        of_nums = [o['of_numero'] for o in ofs]
   274→        if not of_nums:
   275→            return
   276→
   277→        ph = ','.join(['%s'] * len(of_nums))
   278→
   279→        # Total de departamentos unicos (setores) do lote
   280→        total = self.query_one(
   281→            "SELECT COUNT(DISTINCT departamento) as t FROM operacoes_producao "
   282→            "WHERE of_numero IN (" + ph + ") AND departamento IS NOT NULL AND departamento != ''",
   283→            tuple(of_nums)
   284→        )
   285→        total_setores = int(total['t'] or 0) if total else 0
   286→
   287→        # Primeira operacao pendente (nao concluida) ordenada por sequencia
   288→        primeira = self.query_one(
   289→            "SELECT departamento, sequencia FROM operacoes_producao "
   290→            "WHERE of_numero IN (" + ph + ") "
   291→            "AND status != 'concluido' "
   292→            "ORDER BY sequencia ASC LIMIT 1",
   293→            tuple(of_nums)
   294→        )
   295→
   296→        dept = primeira['departamento'] if primeira else None
   297→        seq = int(primeira['sequencia']) if primeira else 0
   298→
   299→        self.execute(
   300→            "UPDATE lotes_producao "
   301→            "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   302→            "WHERE ordem = %s",
   303→            (dept, seq, total_setores, lote_ordem)
   304→        )

   399→    def iniciar_operacao(self, operacao_id, usuario_id):
   400→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   401→        if not op:
   402→            return False
   403→        self.execute("""
   404→            UPDATE operacoes_producao 
   405→            SET status = 'em_andamento', data_inicio = NOW(), usuario_inicio_id = %s
   406→            WHERE id = %s AND status IN ('pendente', 'pausado')
   407→        """, (usuario_id, operacao_id))
   408→        self.atualizar_status_lote(op['lote_ordem'], 'em_producao', usuario_id)
   409→        self.registrar_apontamento_tempo(op['lote_ordem'], op['of_numero'], operacao_id, op['departamento'], usuario_id, 'inicio')
   410→        # Sincroniza kanban: card vai para 'em_producao'
   411→        self._sincronizar_kanban_operacao(operacao_id, 'em_producao', usuario_id)
   412→        return True

   414→    def pausar_operacao(self, operacao_id, usuario_id, motivo='Pausa'):

   416→        if not op:
   417→            return False
   418→        self.execute("""
   419→            UPDATE operacoes_producao SET status = 'pausado' WHERE id = %s AND status = 'em_andamento'
   420→        """, (operacao_id,))
   421→        self.registrar_apontamento_tempo(op['lote_ordem'], op['of_numero'], operacao_id, op['departamento'], usuario_id, 'pausa', motivo)
   422→        # Sincroniza kanban: card vai para 'pausado'
   423→        self._sincronizar_kanban_operacao(operacao_id, 'pausado', usuario_id)

   438→    def _sincronizar_kanban_operacao(self, operacao_id, etapa, operador_id=None):
   439→        """Atualiza o card do kanban correspondente à operação.
   440→        Se o card não existir, cria automaticamente."""
   441→        try:
   442→            # Busca a operação e o lote
   443→            op = self.query_one("""
   444→                SELECT op.*, l.prioridade
   445→                FROM operacoes_producao op
   446→                LEFT JOIN lotes_producao l ON op.lote_ordem = l.ordem
   447→                WHERE op.id = %s
   448→            """, (operacao_id,))
   449→            if not op:
   450→                return
   451→
   452→            # Verifica se já existe card para esta operação
   453→            existente = self.query_one(
   454→                "SELECT id FROM kanban_cards WHERE operacao_id = %s",
   455→                (operacao_id,)
   456→            )
   457→
   458→            if existente:
   459→                # Atualiza card existente
   460→                self.execute("""
   461→                    UPDATE kanban_cards
   462→                    SET etapa = %s,
   463→                        operador_id = COALESCE(%s, operador_id),
   464→                        atualizado_em = NOW()
   465→                    WHERE operacao_id = %s
   466→                """, (etapa, operador_id, operacao_id))
   467→            else:
   468→                # Cria card novo (para operações importadas antes do fix)
   469→                self.execute("""
   470→                    INSERT INTO kanban_cards
   471→                        (lote_ordem, of_numero, operacao_id, etapa, departamento, prioridade, operador_id)
   472→                    VALUES (%s, %s, %s, %s, %s, %s, %s)
   473→                """, (
   474→                    op['lote_ordem'],
   475→                    op.get('of_numero'),
   476→                    operacao_id,
   477→                    etapa,
   478→                    op.get('departamento'),
   479→                    op.get('prioridade') or 'media',
   480→                    operador_id
   481→                ))
   482→        except Exception as e:
   483→            print(f"Erro ao sincronizar kanban: {e}")

   485→    def finalizar_operacao(self, operacao_id, usuario_id, qtde_produzida=None, qtde_refugada=0, observacoes=None, tipo='total'):
   486→        """
   487→        tipo='total'   → marca como concluido e fecha o lote se todas as ops concluídas
   488→        tipo='parcial' → registra a quantidade produzida até agora e reinicia a operação
   489→                         (status volta para 'pendente' com qtde_produzida acumulada)
   490→                         Assim o próximo setor já pode ver as peças disponíveis.
   491→        """
   492→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   493→        if not op:
   494→            return False
   495→
   496→        if tipo == 'parcial':
   497→            # Acumula a quantidade e volta para pendente (disponível para próximo setor)
   498→            self.execute("""
   499→                UPDATE operacoes_producao
   500→                SET status = 'pendente',
   501→                    data_inicio = NULL,
   502→                    qtde_produzida = COALESCE(qtde_produzida, 0) + COALESCE(%s, 0),
   503→                    qtde_refugada  = COALESCE(qtde_refugada, 0)  + COALESCE(%s, 0),
   504→                    observacoes    = COALESCE(%s, observacoes)
   505→                WHERE id = %s
   506→            """, (qtde_produzida, qtde_refugada, observacoes, operacao_id))
   507→            # Atualiza status do lote para 'importado' (disponível) se estava em_producao
   508→            self.execute("""
   509→                UPDATE lotes_producao SET status = 'importado'
   510→                WHERE ordem = %s AND status = 'em_producao'
   511→            """, (op['lote_ordem'],))
   512→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'parcial', qtde_produzida=qtde_produzida)
   513→            # Sincroniza kanban: card volta para 'aguardando' (próximo setor pega)
   514→            self._sincronizar_kanban_operacao(operacao_id, 'aguardando')
   515→        else:
   516→            self.execute("""
   517→                UPDATE operacoes_producao
   518→                SET status = 'concluido', data_fim = NOW(), usuario_fim_id = %s,
   519→                    qtde_produzida = COALESCE(%s, qtde_produzida),
   520→                    qtde_refugada = COALESCE(%s, 0),
   521→                    observacoes = COALESCE(%s, observacoes)
   522→                WHERE id = %s
   523→            """, (usuario_id, qtde_produzida, qtde_refugada, observacoes, operacao_id))
   524→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'fim', qtde_produzida=qtde_produzida)
   525→            # Sincroniza kanban: card vai para 'concluido'
   526→            self._sincronizar_kanban_operacao(operacao_id, 'concluido', usuario_id)
   527→            # Verifica se todas as operações do lote foram concluídas
   528→            self._verificar_conclusao_lote(op['lote_ordem'])
   529→        return True

   531→    def _verificar_conclusao_lote(self, lote_ordem):
   532→        total = self.query_one("SELECT COUNT(*) as t FROM operacoes_producao WHERE lote_ordem = %s", (lote_ordem,))['t']
   533→        concluidas = self.query_one("SELECT COUNT(*) as t FROM operacoes_producao WHERE lote_ordem = %s AND status = 'concluido'", (lote_ordem,))['t']
   534→        if total > 0 and total == concluidas:
   535→            self.execute("UPDATE lotes_producao SET status = 'finalizado', data_fim_producao = NOW() WHERE ordem = %s", (lote_ordem,))
   536→        # Sempre recalcula departamento_atual ao finalizar uma operacao
   537→        try:
   538→            self.atualizar_departamento_atual(lote_ordem)
   539→        except Exception:
   540→            pass

   735→                elif op['status'] == 'concluido':

   738→                    etapa = 'pausado'
   739→                self.add_kanban_card(
   740→                    lote_ordem=lote_ordem,
   741→                    etapa=etapa,

  1038→    def get_estoque_interno(self, status_filtro=None, tipo=None):
  1039→        """Retorna itens do estoque. status_filtro: 'baixo'|'normal'|None  tipo: 'almoxarifado'|'fabrica'|None"""
