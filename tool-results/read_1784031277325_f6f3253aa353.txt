   479→   479→function tempoDecorrido(isoStr) {
   480→   480→  if (!isoStr || isoStr === 'None') return '';
   481→   481→  try {
   482→   482→    // Aceita formato "YYYY-MM-DD HH:MM:SS" ou "YYYY-MM-DDTHH:MM:SS"
   483→   483→    var clean = isoStr.replace('T', ' ');
   484→   484→    var parts = clean.substring(0, 19).split(/[- :]/);
   485→   485→    if (parts.length < 3) return '';
   486→   486→    var y = parseInt(parts[0], 10), mo = parseInt(parts[1], 10) - 1, d = parseInt(parts[2], 10);
   487→   487→    var h = parseInt(parts[3] || 0, 10), mi = parseInt(parts[4] || 0, 10), s = parseInt(parts[5] || 0, 10);
   488→   488→    if (isNaN(y) || isNaN(mo) || isNaN(d)) return '';
   489→   489→    var inicio = new Date(y, mo, d, h, mi, s);
   490→   490→    if (isNaN(inicio.getTime())) return '';
   491→   491→    var diff = Math.max(0, Math.floor((Date.now() - inicio.getTime()) / 1000));
   492→   492→    var hh = String(Math.floor(diff / 3600)).padStart(2, '0');
   493→   493→    var mm = String(Math.floor((diff % 3600) / 60)).padStart(2, '0');
   494→   494→    var ss = String(diff % 60).padStart(2, '0');
   495→   495→    return hh + ':' + mm + ':' + ss;
   496→   496→  } catch (e) { return ''; }
   497→   497→}

   500→   500→function minutosDecorridos(isoStr) {
   501→   501→  if (!isoStr || isoStr === 'None') return -1;
   502→   502→  try {
   503→   503→    var parts = isoStr.replace('T', ' ').substring(0, 19).split(/[- :]/);
   504→   504→    if (parts.length < 3) return -1;
   505→   505→    var inicio = new Date(parseInt(parts[0],10), parseInt(parts[1],10)-1, parseInt(parts[2],10), parseInt(parts[3]||0,10), parseInt(parts[4]||0,10), parseInt(parts[5]||0,10));
   506→   506→    if (isNaN(inicio.getTime())) return -1;
   507→   507→    return Math.max(0, (Date.now() - inicio.getTime()) / 60000);
   508→   508→  } catch (e) { return -1; }
   509→   509→}

   533→   533→function calcProgressoTempo(card) {
   534→   534→  /* Prioridade: tempo estimado > operações concluídas > 0 */
   535→   535→  if (card.etapa === 'concluido') return 100;
   536→   536→  if (card.etapa !== 'em_producao') return 0;
   537→   537→
   538→   538→  var dtInicio = card.data_inicio_fmt || null;
   539→   539→  var minDecorridos = minutosDecorridos(dtInicio);
   540→   540→
   541→   541→  /* Busca tempo medio para este produto+setor */
   542→   542→  var peca = (card.cod_peca || '').trim();
   543→   543→  var setor = (card.departamento || '').trim().toUpperCase();
   544→   544→  var chave = peca + '|' + setor;
   545→   545→  var tempoEstimado = _temposMedios[chave];
   546→   546→
   547→   547→  /* Se tem tempo medio historico, calcula % por tempo */
   548→   548→  if (tempoEstimado > 0 && minDecorridos >= 0) {
   549→   549→    var pct = Math.min(99, Math.round((minDecorridos / tempoEstimado) * 100));
   550→   550→    return Math.max(1, pct);
   551→   551→  }
   552→   552→  /* Se tem data inicio mas sem historico, mostra 10% (sinalizando que comecou) */
   553→   553→  if (minDecorridos >= 0) return 10;
   554→   554→  /* Fallback: progresso por operacoes */
   555→   555→  return card.progresso_ops || 0;
   556→   556→}
