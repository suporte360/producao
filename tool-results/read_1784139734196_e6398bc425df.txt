   390→    return d.toLocaleDateString('pt-BR',{day:'2-digit',month:'2-digit',year:'numeric'});

   404→function progBar(feitas, total) {
   405→  if (!total) return '<span style="color:var(--muted);font-size:.65rem">\u2014</span>';
   406→  var pct = Math.min(100, Math.round((feitas||0)/total*100));
   407→  var cor = pct>=100?'#22c55e':pct>=50?'#f59e0b':'#3b82f6';
   408→  return '<div class="op-progress"><div class="op-progress-bar"><div class="op-progress-fill" style="width:' + pct + '%;background:' + cor + '"></div></div><span style="font-size:.55rem;color:var(--muted)">' + pct + '%</span></div>';
   409→}

   411→/* ─── Auto-scroll suave (fractional accumulator) ── */
   412→var scrollCol = null;
   413→var scrollAccum = 0;
   414→var scrollSpeed = 0.4;
   415→var scrollDir = 1;
   416→
   417→function autoScrollTick() {
   418→  if (!scrollCol) return;
   419→  var maxScroll = scrollCol.scrollHeight - scrollCol.clientHeight;
   420→  if (maxScroll <= 0) return;
   421→  scrollAccum += scrollSpeed * scrollDir;
   422→  if (Math.abs(scrollAccum) >= 1) {
   423→    scrollCol.scrollTop += Math.round(scrollAccum);
   424→    scrollAccum = 0;
   425→  }
   426→  if (scrollCol.scrollTop >= maxScroll - 2) { scrollDir = -1; }
   427→  else if (scrollCol.scrollTop <= 0) { scrollDir = 1; }
   428→}

   430→/* ─── Render Lotes (cards agrupados com OFs horizontais) ── */
   431→function renderLotes(lotes) {
   432→  var panel = document.getElementById('painelLotes');
   433→  document.getElementById('cntLotes').textContent = lotes.length;
   434→  if (!lotes.length) {
   435→    panel.innerHTML = '<div class="empty-panel"><i class="fa-solid fa-circle-check" style="color:#14532d"></i><span>Nenhuma OP aguardando separacao</span></div>';
   436→    return;
   437→  }
   438→  var html = '';
   439→  for (var i = 0; i < lotes.length; i++) {
   440→    var l = lotes[i];
   441→    var sepStatus = l.separacao_status || 'pendente';
   442→    var prio = (l.prioridade||'media').toLowerCase();
   443→    var urgente = prio === 'urgente' || prio === 'alta';
   444→    var atrasada = (l.dias_atraso||0) > 0;
   445→    var cls = 'op-card';
   446→    if (urgente) cls += ' urgente';
   447→    if (atrasada) cls += ' atrasada';
   448→    if (sepStatus === 'separando') cls += ' separando';
   449→    else if (sepStatus === 'separado') cls += ' separado';
   450→    else if (sepStatus === 'entregue') cls += ' entregue';
   451→
   452→    /* OFs horizontais - mostra todas, com cor por status */
   453→    var ofs = l.ordens_do_lote || [];
   454→    var ofsHtml = '';
   455→    var sepCount = 0;
   456→    var entCount = 0;
   457→    for (var k = 0; k < ofs.length; k++) {
   458→      var o = ofs[k];
   459→      if (o.separacao_status === 'separado') { sepCount++; }
   460→      if (o.separacao_status === 'entregue') { entCount++; }
   461→      if (ofsHtml) ofsHtml += '<span class="lote-of-sep">-</span>';
   462→      var ofCls = 'lote-of-item';
   463→      var ofPrio = (o.prioridade||'media').toLowerCase();
   464→      var ofAtraso = (o.dias_atraso||0) > 0;
   465→      if (ofPrio === 'urgente') ofCls += ' urgente-of';
   466→      else if (ofAtraso) ofCls += ' atraso';
   467→      if (o.separacao_status === 'separando') ofCls += ' separando';
   468→      else if (o.separacao_status === 'separado') ofCls += ' separado';
   469→      else if (o.separacao_status === 'entregue') ofCls += ' entregue';
   470→      ofsHtml += '<span class="' + ofCls + '">#' + o.ordem + '</span>';
   471→    }
   472→    var sepInfo = '';
   473→    if (sepCount > 0) {
   474→      sepInfo = '<span class="lote-sep-info" style="color:#fbbf24">' + sepCount + ' OF' + (sepCount > 1 ? 's' : '') + ' separada' + (sepCount > 1 ? 's' : '') + '</span>';
   475→    }
   476→    if (entCount > 0) {
   477→      sepInfo += '<span class="lote-sep-info" style="color:#64748b">' + entCount + ' entregue' + (entCount > 1 ? 's' : '') + '</span>';
   478→    }
   479→
   480→    /* componentes */
   481→    var matsHtml = '';
   482→    var comps = l.componentes || [];
   483→    var totalComps = l.total_componentes || comps.length;
   484→    if (comps.length > 0) {
   485→      matsHtml = '<div class="op-mats"><div class="op-mats-title"><i class="fa-solid fa-screwdriver-wrench" style="margin-right:4px"></i>' + totalComps + ' componente' + (totalComps !== 1 ? 's' : '') + ' para separar</div>';
   486→      for (var j = 0; j < comps.length; j++) {
   487→        var c = comps[j];
   488→        matsHtml += '<div class="op-mat-row">' +
   489→          '<span class="mat-cod">' + c.codigo + '</span>' +
   490→          '<span class="mat-desc">' + c.descricao + '</span>' +
   491→          '<span class="mat-qtd">' + c.qtd_necessaria + '</span>' +
   492→          '<span class="mat-und">' + c.unidade + '</span>' +
   493→        '</div>';
   494→      }
   495→      matsHtml += '</div>';
   496→    } else {
   497→      matsHtml = '<div class="op-comp-count" style="margin-top:4px"><i class="fa-solid fa-info-circle" style="margin-right:3px"></i>Sem componentes brutos (so pecas fabricadas)</div>';
   498→    }
   499→
   500→    html += '<div class="' + cls + '">' +
   501→      '<div class="lote-header">' +
   502→        '<div class="lote-id">' +
   503→          '<span class="lote-lp">LP ' + (l.lote_codigo || '---') + '</span>' +
   504→          '<span class="lote-of-label">OF</span>' +
   505→          '<span class="lote-ofs">' + ofsHtml + '</span>' +
   506→          sepInfo +
   507→        '</div>' +
   508→        '<div class="lote-badges">' +
   509→          badgePrio(l.prioridade) +
   510→          badgeSep(sepStatus) +
   511→        '</div>' +
   512→      '</div>' +
   513→      '<div class="op-info">' +
   514→        '<div class="op-prod">' + (l.descricao_produto || '\u2014').substring(0, 50) + '</div>' +
   515→        '<div class="op-meta">' +
   516→          '<span class="op-qtd"><i class="fa-solid fa-boxes-stacked" style="font-size:.6rem;margin-right:2px"></i>' + Math.round(l.qtde_ordem || 0) + ' pc</span>' +
   517→          (l.departamento_atual ? '<span class="op-dept">' + l.departamento_atual + '</span>' : '') +
   518→          (atrasada ? '<span class="op-qtd" style="color:#f87171"><i class="fa-solid fa-circle-exclamation" style="font-size:.6rem;margin-right:2px"></i>' + l.dias_atraso + 'd atraso</span>' : '') +
   519→          (l.data_previsao_erp ? '<span class="op-qtd"><i class="fa-regular fa-calendar" style="font-size:.6rem;margin-right:2px"></i>' + fmtData(l.data_previsao_erp) + '</span>' : '') +
   520→        '</div>' +
   521→        matsHtml +
   522→      '</div>' +
   523→    '</div>';
   524→  }
   525→  panel.innerHTML = html;
   526→}
