    29→def _load_produto_cache():
    30→    """Carrega nomes reais dos produtos do PostgreSQL."""
    31→    global _produto_cache
    32→    try:
    33→        pg = DatabasePostgreSQL()
    34→        rows = pg.query(
    35→            "SELECT produto, pronome FROM public.produto "
    36→            "WHERE pronome IS NOT NULL AND pronome != ''"
    37→        )
    38→        _produto_cache = {}
    39→        for r in rows:
    40→            cod = str(r.get('produto', '')).strip()
    41→            nome = str(r.get('pronome', '')).strip()
    42→            if cod:
    43→                _produto_cache[cod] = nome[:200]
    44→        print('[OK] Cache de produtos carregado: %d produtos do ERP' % len(_produto_cache))
    45→    except Exception as e:
    46→        print('[WARN] Cache de produtos falhou: %s' % e)

    48→def _get_nome_produto(codigo):
    49→    """Retorna nome real do produto pelo codigo."""
    50→    if not codigo:
    51→        return ''
    52→    return _produto_cache.get(str(codigo).strip(), '')

    54→def _is_sob_medida(codigo):
    55→    """Verifica se o produto codigo termina com -X (sob medida)."""
    56→    if not codigo:
    57→        return False
    58→    c = str(codigo).strip().upper()
    59→    # S-0440-X, CJ-0174-X, qualquer codigo terminado em -X
    60→    return c.endswith('-X')

    62→def _resolver_nome_lote(lote):
    63→    """Resolve nome real para descricao_produto de um lote/OF."""
    64→    desc = lote.get('descricao_produto', '') or ''
    65→    cod = lote.get('codigo_produto', '') or ''
    66→    if not desc or desc.upper().startswith('PRODUCAO DE'):
    67→        nome = _get_nome_produto(cod)
    68→        if nome:
    69→            return nome
    70→    return desc

   246→# =============================================
   247→# Proteção contra força bruta no login
   248→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
   249→# =============================================
   250→
   251→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
   252→MAX_TENTATIVAS  = 5
   253→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
   254→
   255→def _get_ip():
   256→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
   257→
   258→def _checar_bloqueio(ip):
   259→    dados = _login_attempts[ip]
   260→    if time.time() < dados['bloqueado_ate']:
   261→        restante = int(dados['bloqueado_ate'] - time.time())
   262→        return True, restante
   263→    return False, 0
   264→
   265→def _registrar_falha(ip):
   266→    dados = _login_attempts[ip]
   267→    dados['tentativas'] += 1
   268→    if dados['tentativas'] >= MAX_TENTATIVAS:
   269→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   270→        dados['tentativas'] = 0
   271→
   272→def _limpar_tentativas(ip):
   273→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}
   274→

   279→def login_required(f):
   280→    @wraps(f)
   281→    def decorated(*args, **kwargs):
   282→        if 'usuario_id' not in session:
   283→            flash('Faça login para continuar.', 'warning')
   284→            return redirect(url_for('login'))
   285→        return f(*args, **kwargs)
   286→    return decorated
   287→
   288→def role_required(*roles):
   289→    def decorator(f):
   290→        @wraps(f)
   291→        def decorated(*args, **kwargs):
   292→            if session.get('usuario_role') not in roles:
   293→                flash('Acesso não autorizado.', 'error')
   294→                return redirect(url_for('dashboard'))
   295→            return f(*args, **kwargs)
   296→        return decorated
   297→    return decorator

   303→@app.route('/login', methods=['GET', 'POST'])
   304→def login():
   305→    if 'usuario_id' in session:
   306→        return redirect(url_for('dashboard'))
   307→
   308→    if request.method == 'POST':
   309→        ip = _get_ip()
   310→
   311→        bloqueado, restante = _checar_bloqueio(ip)
   312→        if bloqueado:
   313→            mins = restante // 60
   314→            secs = restante % 60
   315→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   316→            return render_template('login.html')
   317→
   318→        usuario = request.form.get('usuario', '').strip()
   319→        senha   = request.form.get('senha', '')
   320→
   321→        if not usuario or not senha:
   322→            flash('Informe usuário e senha.', 'error')
   323→            return render_template('login.html')
   324→
   325→        user = db.verificar_usuario(usuario, senha)
   326→
   327→        if user:
   328→            _limpar_tentativas(ip)
   329→            session.permanent = True
   330→            session['usuario_id']          = user['id']
   331→            session['usuario_nome']        = user['nome']
   332→            session['usuario_role']        = user['role']
   333→            session['usuario_departamento']= user.get('nome_departamento', '')
   334→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   335→            return redirect(url_for('dashboard'))
   336→        else:
   337→            _registrar_falha(ip)
   338→            dados   = _login_attempts[ip]
   339→            restantes = MAX_TENTATIVAS - dados['tentativas']
   340→            if restantes > 0:
   341→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   342→            else:
   343→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   344→
   345→    return render_template('login.html')

   470→@app.route('/kanban/setores')
   471→@login_required
   472→@role_required('admin', 'gerente', 'diretor')
   473→def kanban_setores():
   474→    """Kanban visual organizado por setor/departamento."""
   475→    departamentos = db.listar_departamentos(apenas_ativos=True)
   476→    todos_cards = db.get_kanban_cards(departamento=None)
   477→
   478→    # Agrupar cards por departamento
   479→    cards_por_setor = {}
   480→    for dept in departamentos:
   481→        cards_por_setor[dept['codigo']] = {
   482→            'info': dept,
   483→            'cards': [],
   484→            'total': 0,
   485→            'em_producao': 0,
   486→            'aguardando': 0,
   487→            'pausados': 0,
   488→            'concluidos': 0,
   489→        }
   490→
   491→    for card in todos_cards:
   492→        dept_code = card.get('departamento', '')
   493→        if dept_code in cards_por_setor:
   494→            cards_por_setor[dept_code]['cards'].append(card)
   495→            cards_por_setor[dept_code]['total'] += 1
   496→            etapa = card.get('etapa', '')
   497→            if etapa == 'em_producao':
   498→                cards_por_setor[dept_code]['em_producao'] += 1
   499→            elif etapa in ('aguardando', 'pendente'):
   500→                cards_por_setor[dept_code]['aguardando'] += 1
   501→            elif etapa == 'pausado':
   502→                cards_por_setor[dept_code]['pausados'] += 1
   503→            elif etapa == 'concluido':
   504→                cards_por_setor[dept_code]['concluidos'] += 1
   505→
   506→    # KPIs gerais
   507→    kpis = {
   508→        'setores_ativos': sum(1 for s in cards_por_setor.values() if s['total'] > 0),
   509→        'total_cards': len(todos_cards),
   510→        'em_producao': sum(1 for c in todos_cards if c.get('etapa') == 'em_producao'),
   511→        'concluidos_hoje': db.get_kanban_concluidos_hoje(),
   512→    }
   513→
   514→    return render_template('kanban/setores.html',
   515→                           cards_por_setor=cards_por_setor,
   516→                           kpis=kpis,
   517→                           usuario_role=session.get('usuario_role'))

   564→

   566→        pg = DatabasePostgreSQL()

   589→    lotes_ja_importados = {r['lote_codigo'] for r in ja}

   592→                           pg_ok=pg_ok, pg_erro=pg_erro,

   599→@app.route('/pcp/importar_erp', methods=['POST'])
   600→@login_required
   601→@role_required('admin', 'gerente', 'pcp', 'diretor')
   602→def pcp_executar_importacao():
   603→    data  = request.json or {}
   604→    lotes = data.get('lotes', [])
   605→    if not lotes:
   606→        return jsonify({'status':'error','message':'Nenhum lote informado'}), 400
   607→
   608→    importados = 0
   609→    atualizados = 0
   610→    erros = []
   611→
   612→    try:
   613→        pg = DatabasePostgreSQL()
   614→    except Exception as e:
   615→        return jsonify({'status':'error','message': f'Erro PostgreSQL: {e}'}), 500
   616→
   617→    for item in lotes:
   618→        try:
   619→            lote_cod = str(item.get('lote_codigo',''))
   620→            ordem_num = item.get('ordem')
   621→            if not lote_cod:
   622→                continue
   623→
   624→            # Verifica se já existe
   625→            existente = db.query_one(
   626→                "SELECT ordem FROM lotes_producao WHERE lote_codigo=%s", (lote_cod,))
   627→            ja_existia = existente is not None
   628→
   629→            # Importa lote principal
   630→            db.importar_lote_erp({
   631→                'ordem':             int(ordem_num) if str(ordem_num).isdigit() else 0,
   632→                'lote_codigo':       lote_cod,
   633→                'codigo_produto':    item.get('produto',''),
   634→                'descricao_produto': item.get('nome','') or item.get('produto',''),
   635→                'qtde_ordem':        float(item.get('qtd', 0) or 0),
   636→                'unidade_medida':    'PC',
   637→                'status_erp':        'A',
   638→                'data_previsao_erp': item.get('previsao') or None,
   639→                'data_abertura_erp': item.get('abertura') or None,
   640→                'planejador':        session.get('usuario_nome'),
   641→                'observacoes':       item.get('observacoes') or '',
   642→            })
   643→
   644→            lote_ordem_local = int(ordem_num) if str(ordem_num).isdigit() else 0
   645→
   646→            # Busca ordens do lote no ERP e importa OFs
   647→            ordens_erp = pg.get_ordens_por_lote(lote_cod)
   648→            for of in ordens_erp:
   649→                of_num = of.get('ordem')
   650→                if not of_num:
   651→                    continue
   652→                db.importar_of({
   653→                    'lote_ordem':       lote_ordem_local,
   654→                    'of_numero':        str(of_num),
   655→                    'codigo_produto':   of.get('produto',''),
   656→                    'descricao_produto': item.get('nome','') or item.get('produto',''),
   657→                    'qtde_ordem':       float(of.get('quantidade') or 0),
   658→                    'unidade_medida':   'PC',
   659→                    'tipo':             'filho' if of.get('nivel_producao','') != '1' else 'pai',
   660→                    'data_previsao':    of.get('data_previsao') or None,
   661→                })
   662→
   663→                # Busca processos (operações) de cada OF
   664→                processos = pg.get_processos_ordem(str(of_num))
   665→                for proc in processos:
   666→                    dept_original = (proc.get('nome_departamento') or '').upper().strip()
   667→                    if not dept_original:
   668→                        continue
   669→                    # Limpa parênteses e sujeiras do nome do ERP (ex: "CORTE()" → "CORTE")
   670→                    dept_original = re.sub(r'\s*\([^)]*\)\s*$', '', dept_original).strip()
   671→                    # Mapeia nomes do ERP (fases) para nomes do sistema local
   672→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   673→                    # Correcao automatica: se a operacao envolve dobra, forca DOBRA
   674→                    desc_op = (proc.get('descricao_processo') or proc.get('operacao','')).upper()
   675→                    if 'DOBRAR' in desc_op and dept not in ('DOBRA',):
   676→                        dept = 'DOBRA'
   677→                    db.importar_operacao({
   678→                        'of_numero':          str(of_num),
   679→                        'lote_ordem':         lote_ordem_local,
   680→                        'sequencia':          proc.get('numero_sequencial', 0),
   681→                        'fase':               proc.get('fase',''),
   682→                        'descricao_operacao': proc.get('descricao_processo') or proc.get('operacao',''),
   683→                        'departamento':       dept,
   684→                        'codigo_barras':      None,
   685→                    })
   686→
   687→                # Sincroniza kanban
   688→                try:
   689→                    db.sincronizar_kanban_com_operacoes(lote_ordem_local)
   690→                except Exception:
   691→                    pass
   692→
   693→                # Atualiza departamento_atual baseado na primeira op pendente
   694→                try:
   695→                    db.atualizar_departamento_atual(lote_ordem_local)
   696→                except Exception:
   697→                    pass
   698→
   699→            db.add_log(session['usuario_id'], 'importar_erp',
   700→                       f'Lote {lote_cod} importado do ERP', 'lotes_producao', lote_ordem_local)
   701→
   702→            if ja_existia:
   703→                atualizados += 1
   704→            else:
   705→                importados += 1
   706→
   707→        except Exception as e:
   708→            erros.append(f"{item.get('lote_codigo')}: {e}")
   709→
   710→    return jsonify({
   711→        'status':     'success',
   712→        'importados':  importados,
   713→        'atualizados': atualizados,
   714→        'erros':       erros,
   715→    })

  1048→def _formatar_minutos(minutos):
  1049→    """Formata minutos para string legivel: '1h 23min' ou '45min'."""
  1050→    m = float(minutos)
  1051→    if m <= 0:
  1052→        return '—'
  1053→    h = int(m // 60)
  1054→    mi = int(round(m % 60))
  1055→    if h > 0 and mi > 0:
  1056→        return str(h) + 'h ' + str(mi) + 'min'
  1057→    if h > 0:
  1058→        return str(h) + 'h'
  1059→    return str(mi) + 'min'
