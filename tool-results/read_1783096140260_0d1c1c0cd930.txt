   447→function esc(s) {
   448→  if (!s) return '';
   449→  var d = document.createElement('div');
   450→  d.appendChild(document.createTextNode(String(s)));
   451→  return d.innerHTML;

   454→/* ── Animate Number ── */
   455→function animateValue(el, end) {
   456→  var startVal = parseInt(el.textContent) || 0;
   457→  if (startVal === end) return;
   458→  var dur = 600;
   459→  var st = null;
   460→  function step(ts) {
   461→    if (!st) st = ts;
   462→    var p = Math.min((ts - st) / dur, 1);
   463→    var eased = 1 - Math.pow(1 - p, 3);
   464→    el.textContent = Math.round(startVal + (end - startVal) * eased);
   465→    if (p < 1) requestAnimationFrame(step);
   466→    else el.style.animation = 'countPulse .3s ease';
   467→  }
   468→  el.style.animation = '';
   469→  requestAnimationFrame(step);
   470→}

   502→/* ── Update Pipeline with active workers ── */
   503→function updatePipeline(ativos) {
   504→  /* Clear all nodes first */
   505→  var i;
   506→  for (i = 0; i < 5; i++) {
   507→    document.getElementById('node-' + i).classList.remove('active');
   508→    document.getElementById('nw-' + i).textContent = '';
   509→  }
   510→
   511→  /* Assign active workers to sectors cyclically */
   512→  if (ativos.length === 0) return;
   513→
   514→  /* Group by unique worker (one entry per worker) */
   515→  var seen = {};
   516→  var workers = [];
   517→  for (i = 0; i < ativos.length; i++) {
   518→    var nome = ativos[i].usuario_nome;
   519→    if (!seen[nome]) {
   520→      seen[nome] = true;
   521→      workers.push(ativos[i]);
   522→    }
   523→  }
   524→
   525→  for (i = 0; i < workers.length && i < 5; i++) {
   526→    var node = document.getElementById('node-' + i);
   527→    node.classList.add('active');
   528→    document.getElementById('nw-' + i).textContent = workers[i].usuario_nome;
   529→  }
   530→
   531→  /* If there are active workers, keep all pipes active */
   532→  var arrows = document.querySelectorAll('.p-arrow');
   533→  var hasActive = workers.length > 0;
   534→  for (i = 0; i < arrows.length; i++) {
   535→    if (hasActive) {
   536→      arrows[i].classList.remove('inactive');
   537→    } else {
   538→      arrows[i].classList.add('inactive');
   539→    }
   540→  }
   541→}

   543→/* ── Render Lotes Marquee on Dashboard ── */
   544→function renderLotesDash(lotes) {
   545→  var track = document.getElementById('marqueeTrack');
   546→  var countEl = document.getElementById('lotesDashCount');
   547→  countEl.textContent = lotes.length;
   548→
   549→  if (lotes.length === 0) {

   551→    track.style.animation = 'none';
   552→    return;
   553→  }
   554→
   555→  /* Build marquee items */
   556→  var items = '';
   557→  for (var i = 0; i < lotes.length; i++) {
   558→    var l = lotes[i];
   559→    var isProd = l.status_producao === 'em_producao';
   560→    var cls = isProd ? ' prod' : '';
   561→    var badgeCls = isProd ? 'prod' : 'wait';
   562→    var badgeText = isProd ? 'PRODUZINDO' : 'AGUARDANDO';
   563→
   564→    var prodText = '';
   565→    if (l.produto_nome) {
   566→      prodText = esc(l.produto_nome);
   567→    } else if (l.produto_final) {
   568→      prodText = esc(l.produto_final);
   569→    }
   570→
   571→    var shortDesc = '';
   572→    if (l.descricao) {
   573→      shortDesc = l.descricao.length > 35 ? l.descricao.substring(0, 35) + '...' : l.descricao;

   575→
   576→    items += '<div class="marquee-item' + cls + '">'
   577→      + '<span class="mi-lote">' + esc(l.lotcod) + '</span>'
   578→      + '<span class="mi-sep">|</span>'
   579→      + (prodText ? '<span class="mi-produto">' + prodText + '</span><span class="mi-sep">|</span>' : '')
   580→      + (shortDesc ? '<span class="mi-desc">' + esc(shortDesc) + '</span><span class="mi-sep">|</span>' : '')

   584→      + '</div>';
   585→  }
   586→
   587→  /* Duplicate items for seamless loop */
   588→  track.innerHTML = items + items;
   589→
   590→  /* Adjust speed based on number of lotes: more lotes = faster scroll */
   591→  var duration = Math.max(22, lotes.length * 9);

   593→  track.style.animation = 'marqueeScroll ' + duration + 's linear infinite';

   699→      if (l.descricao) {
   700→        descInline = l.descricao.length > 45 ? l.descricao.substring(0, 45) + '...' : l.descricao;
   701→      }
   702→
   703→      /* Product code + name */
   704→      var prodInfo = '';
   705→      if (l.produto_final) {
   706→        prodInfo = '<span style="color:#42a5f5;font-weight:700">' + esc(l.produto_final) + '</span>';
   707→      }
   708→      if (l.produto_nome) {
   709→        prodInfo += (prodInfo ? ' ' : '') + '<span style="color:#ccc">' + esc(l.produto_nome) + '</span>';
   710→      }
   711→
   712→      /* Convert ERP quantity: 6.000 -> 6 pecas */
   713→      var qtdPecas = 0;
   714→      if (l.qtd_produto_final) {
   715→        qtdPecas = Math.round(parseFloat(l.qtd_produto_final));
   716→      }
   717→
   718→      html += '<div class="lote-card' + borderClass + '" onclick="verLote(\'' + esc(l.lotcod) + '\')">'
   719→        + '<div class="lote-left">'
   720→        + '<div class="lote-header"><span class="lote-codigo">LOTE ' + esc(l.lotcod) + '</span>'
   721→        + (descInline ? '<span class="lote-desc-inline">' + esc(descInline) + '</span>' : '')
   722→        + badge + '</div>'
   723→        + (prodInfo ? '<div class="lote-prod-row">' + prodInfo + '</div>' : '')
   724→        + '</div>'
   725→        + '<div class="lote-qty-wrap"><div class="lote-qty">' + qtdPecas + '</div><div class="lote-qty-label">pecas</div></div>'
   726→        + '</div>';
   727→    }
   728→    document.getElementById('lotesList').innerHTML = html;
   729→  }).catch(function(e) {
   730→    console.error('Lotes error:', e);
   731→    document.getElementById('lotesList').innerHTML = '<div class="loading" style="color:#ef5350">Erro ao carregar lotes</div>';
   732→  });
   733→}

   735→/* ── Ver Lote -> Screen 3 (Pecas) ── */
   736→function verLote(lotcod) {
   737→  currentLote = lotcod;
   738→  currentPecas = [];
   739→  currentPecasAtivas = {};
   740→  selectedSetor = null;
   741→  selectedPecas = [];
   742→  document.getElementById('pecasTitle').textContent = 'PECAS - LOTE ' + lotcod;
   743→  document.getElementById('pecasContent').innerHTML = '<div class="loading"><div class="spinner"></div><br>Carregando pecas...</div>';
   744→  document.getElementById('pecasActions').innerHTML = '';
   745→  showScreen(3);
   746→
   747→  fetch('/api/lotes/' + lotcod + '/pecas').then(function(r) { return r.json(); }).then(function(data) {

   751→    }
   752→    currentPecas = data.pecas || [];
   753→    currentPecasAtivas = data.pecas_ativas || {};
   754→    renderPecas(data);
   755→  }).catch(function(e) {
   756→    console.error('Pecas error:', e);
   757→    document.getElementById('pecasContent').innerHTML = '<div class="loading" style="color:#ef5350">Erro ao carregar</div>';
   758→  });

   761→/* ── Render Pecas Screen ── */
   762→function renderPecas(data) {
   763→  var html = '';
   764→  var pecas = currentPecas;
   765→  var hasAtivas = false;
   766→
   767→  /* Produto final (S-) com qtd convertida */
   768→  if (data.produto_final) {
   769→    var qtdFinal = Math.round(parseFloat(data.produto_final.quantidade));
   770→    html += '<div class="prod-final">'
   771→      + '<div><div class="pf-label">Produto Final (referencia)</div>'
   772→      + '<div class="pf-codigo">' + esc(data.produto_final.codigo) + '</div></div>'
   773→      + '<div style="text-align:right"><div style="font-size:12px;color:#888">Qtd</div>'
   774→      + '<div style="font-size:20px;font-weight:700;color:#42a5f5">' + qtdFinal + '</div></div>'
   775→      + '</div>';
   776→  }
   777→
   778→  if (data.descricao) {
   779→    html += '<div style="padding:4px 4px 12px;font-size:13px;color:#888">' + esc(data.descricao) + '</div>';
   780→  }
   781→
   782→  /* Contar pecas disponiveis (nao em producao) */
   783→  var disponiveis = 0;
   784→  for (var k = 0; k < pecas.length; k++) {
   785→    if (!currentPecasAtivas[pecas[k].codigo]) disponiveis++;
   786→  }
   787→
   788→  html += '<div class="section-title">Pecas para Fabricar (' + disponiveis + ' de ' + pecas.length + ')</div>';
   789→
   790→  /* Pecas em producao primeiro */
   791→  for (var i = 0; i < pecas.length; i++) {
   792→    var p = pecas[i];
   793→    var tipo = p.codigo.substring(0, 2);
   794→    var tipoLabel = tipo === 'CJ' ? 'Conjunto' : 'Peca';
   795→    var qtd = Math.round(parseFloat(p.quantidade));
   796→    var isActive = currentPecasAtivas[p.codigo];
   797→    var cardClass = 'peca-card';
   798→    var clickAction = '';
   799→    var checkIcon = '';
   800→
   801→    if (isActive) {
   802→      cardClass += ' em_producao';
   803→      checkIcon = '<div class="pecas-check-icon">&#10003;</div>';
   804→      hasAtivas = true;
   805→    } else if (selectedSetor) {
   806→      clickAction = ' onclick="togglePecaSelection(\'' + esc(p.codigo) + '\')"';
   807→      var isSelected = selectedPecas.indexOf(p.codigo) >= 0;
   808→      if (isSelected) {
   809→        cardClass += ' selected';
   810→        checkIcon = '<div class="pecas-check-icon">&#10003;</div>';
   811→      } else {
   812→        checkIcon = '<div class="pecas-check-icon"></div>';
   813→      }
   814→    } else {
   815→      clickAction = ' onclick="abrirSetor()"';
   816→      checkIcon = '<div class="pecas-check-icon"></div>';
   817→    }
   818→
   819→    var pNome = p.produto_nome ? esc(p.produto_nome) : '';
   820→
   821→    html += '<div class="' + cardClass + '"' + clickAction + '>'
   822→      + '<div style="display:flex;align-items:center">'
   823→      + checkIcon
   824→      + '<div class="peca-left">'
   825→      + '<div class="peca-codigo">' + esc(p.codigo) + '</div>'
   826→      + (pNome ? '<div class="peca-nome">' + pNome + '</div>' : '')
   827→      + '<div class="peca-sub">' + tipoLabel + ' | OF ' + p.ordem + '</div>'
   828→      + (isActive ? '<div class="peca-badge">' + esc(isActive.setor) + ' - ' + esc(isActive.usuario) + '</div>' : '')
   829→      + '</div></div>'
   830→      + '<div class="peca-qtd-wrap"><div class="peca-qtd">' + qtd + '</div><div class="peca-qtd-label">pecas</div></div>'
   831→      + '</div>';
   832→  }
   833→
   834→  if (pecas.length === 0) {
   835→    html += '<div class="loading" style="color:#f59e0b">Nenhuma peca para fabricar neste lote</div>';
   836→  }
   837→
   838→  document.getElementById('pecasContent').innerHTML = html;
   839→
   840→  /* Acoes */
   841→  var actions = '<button class="btn-back" onclick="showScreen(2)">&#8592; Voltar aos Lotes</button>';
   842→
   843→  /* Se tem pecas em producao, mostrar botao finalizar */
   844→  if (hasAtivas) {
   845→    actions = '<button class="btn-touch btn-red" onclick="confirmarFinalizar(\'' + esc(currentLote) + '\')">FINALIZAR PRODUCAO</button>'
   846→      + actions;
   847→  }
   848→
   849→  /* Se setor selecionado e pecas selecionadas, mostrar iniciar */
   850→  if (selectedSetor && selectedPecas.length > 0) {
   851→    actions = '<button class="btn-touch btn-green" onclick="iniciarProducaoSetor()">INICIAR PRODUCAO (' + selectedPecas.length + ')</button>'
   852→      + actions;
   853→  } else if (selectedSetor && selectedPecas.length === 0) {
   854→    actions = '<div style="text-align:center;padding:6px;color:#f59e0b;font-size:13px">Selecione as pecas para ' + esc(selectedSetor) + '</div>'
   855→      + actions;
   856→  }
   857→
   858→  document.getElementById('pecasActions').innerHTML = actions;
   859→}

   882→/* ── Toggle Peca Selection ── */
   883→function togglePecaSelection(codigo) {
   884→  var idx = selectedPecas.indexOf(codigo);
   885→  if (idx >= 0) {
   886→    selectedPecas.splice(idx, 1);
   887→  } else {
   888→    selectedPecas.push(codigo);
   889→  }
   890→  /* Update UI without full re-render */
   891→  var cards = document.querySelectorAll('.peca-card');
   892→  for (var i = 0; i < cards.length; i++) {
   893→    var card = cards[i];
   894→    var codeEl = card.querySelector('.peca-codigo');
   895→    if (!codeEl) continue;
   896→    var code = codeEl.textContent;
   897→    var checkIcon = card.querySelector('.pecas-check-icon');
   898→    if (selectedPecas.indexOf(code) >= 0) {
   899→      card.classList.add('selected');
   900→      if (checkIcon) checkIcon.innerHTML = '&#10003;';
   901→    } else if (!card.classList.contains('em_producao')) {
   902→      card.classList.remove('selected');
   903→      if (checkIcon) checkIcon.innerHTML = '';
   904→    }
   905→  }
   906→  /* Update actions */
   907→  var actionsEl = document.getElementById('pecasActions');
   908→  var actions = '';
   909→  if (selectedPecas.length > 0) {
   910→    actions = '<button class="btn-touch btn-green" onclick="iniciarProducaoSetor()">INICIAR PRODUCAO (' + selectedPecas.length + ')</button>';
   911→  } else {
   912→    actions = '<div style="text-align:center;padding:6px;color:#f59e0b;font-size:13px">Selecione as pecas para ' + esc(selectedSetor) + '</div>';
   913→  }
   914→  actions += '<button class="btn-back" onclick="showScreen(2)">&#8592; Voltar aos Lotes</button>';
   915→  actionsEl.innerHTML = actions;
   916→}

   929→  fetch('/api/usuarios').then(function(r) { return r.json(); }).then(function(data) {

   947→/* ── Selecionar Usuario -> Iniciar -> RETURN TO DASHBOARD ── */
   948→function selecionarUsuario(id, nome) {
   949→  /* Build pecas array from selectedPecas */
   950→  var pecas = [];
   951→  for (var i = 0; i < currentPecas.length; i++) {
   952→    if (selectedPecas.indexOf(currentPecas[i].codigo) >= 0) {
   953→      pecas.push({ ordem: currentPecas[i].ordem, codigo: currentPecas[i].codigo });
   954→    }
   955→  }
   956→  fetch('/api/iniciar', {
   957→    method: 'POST',
   958→    headers: { 'Content-Type': 'application/json' },
   959→    body: JSON.stringify({ lote: currentLote, usuario_id: id, usuario_nome: nome, setor: selectedSetor, pecas: pecas })
   960→  }).then(function(r) { return r.json(); }).then(function(data) {
   961→    if (data.success) {
   962→      /* Return to dashboard as requested */
   963→      showScreen(1);
   964→    } else {
   965→      alert('Erro: ' + (data.error || 'Erro desconhecido'));
   966→    }
   967→  }).catch(function(e) { alert('Erro de conexao'); });
   968→}
