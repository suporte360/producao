    59→    59→@app.template_filter('formatar_tempo')
    60→    60→def formatar_tempo(segundos):
    61→    61→    """Formata segundos em HH:MM:SS"""
    62→    62→    try:
    63→    63→        seg = int(segundos or 0)
    64→    64→        h = seg // 3600
    65→    65→        m = (seg % 3600) // 60
    66→    66→        s = seg % 60
    67→    67→        return f'{h:02d}:{m:02d}:{s:02d}'
    68→    68→    except Exception:
    69→    69→        return '00:00:00'

    71→    71→@app.template_filter('formatar_data_br')
    72→    72→def formatar_data_br(value):
    73→    73→    """Formata datetime/date para dd/mm/yyyy HH:MM"""
    74→    74→    if not value:
    75→    75→        return '-'
    76→    76→    try:
    77→    77→        if hasattr(value, 'strftime'):
    78→    78→            if hasattr(value, 'hour'):
    79→    79→                return value.strftime('%d/%m/%Y %H:%M')
    80→    80→            return value.strftime('%d/%m/%Y')
    81→    81→        return str(value)
    82→    82→    except Exception:
    83→    83→        return str(value)

    86→    86→# Proteção contra força bruta no login
    87→    87→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
    88→    88→# =============================================
    89→    89→
    90→    90→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
    91→    91→MAX_TENTATIVAS  = 5
    92→    92→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
    93→    93→
    94→    94→def _get_ip():
    95→    95→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
    96→    96→
    97→    97→def _checar_bloqueio(ip):
    98→    98→    dados = _login_attempts[ip]
    99→    99→    if time.time() < dados['bloqueado_ate']:
   100→   100→        restante = int(dados['bloqueado_ate'] - time.time())
   101→   101→        return True, restante
   102→   102→    return False, 0
   103→   103→
   104→   104→def _registrar_falha(ip):
   105→   105→    dados = _login_attempts[ip]
   106→   106→    dados['tentativas'] += 1
   107→   107→    if dados['tentativas'] >= MAX_TENTATIVAS:
   108→   108→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   109→   109→        dados['tentativas'] = 0
   110→   110→
   111→   111→def _limpar_tentativas(ip):
   112→   112→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}

   118→   118→def login_required(f):
   119→   119→    @wraps(f)
   120→   120→    def decorated(*args, **kwargs):
   121→   121→        if 'usuario_id' not in session:
   122→   122→            flash('Faça login para continuar.', 'warning')
   123→   123→            return redirect(url_for('login'))
   124→   124→        return f(*args, **kwargs)
   125→   125→    return decorated
   126→   126→
   127→   127→def role_required(*roles):
   128→   128→    def decorator(f):
   129→   129→        @wraps(f)
   130→   130→        def decorated(*args, **kwargs):
   131→   131→            if session.get('usuario_role') not in roles:
   132→   132→                flash('Acesso não autorizado.', 'error')
   133→   133→                return redirect(url_for('dashboard'))
   134→   134→            return f(*args, **kwargs)
   135→   135→        return decorated
   136→   136→    return decorator

   142→   142→@app.route('/login', methods=['GET', 'POST'])
   143→   143→def login():
   144→   144→    if 'usuario_id' in session:
   145→   145→        return redirect(url_for('dashboard'))
   146→   146→
   147→   147→    if request.method == 'POST':
   148→   148→        ip = _get_ip()
   149→   149→
   150→   150→        bloqueado, restante = _checar_bloqueio(ip)
   151→   151→        if bloqueado:
   152→   152→            mins = restante // 60
   153→   153→            secs = restante % 60
   154→   154→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   155→   155→            return render_template('login.html')
   156→   156→
   157→   157→        usuario = request.form.get('usuario', '').strip()
   158→   158→        senha   = request.form.get('senha', '')
   159→   159→
   160→   160→        if not usuario or not senha:
   161→   161→            flash('Informe usuário e senha.', 'error')
   162→   162→            return render_template('login.html')
   163→   163→
   164→   164→        user = db.verificar_usuario(usuario, senha)
   165→   165→
   166→   166→        if user:
   167→   167→            _limpar_tentativas(ip)
   168→   168→            session.permanent = True
   169→   169→            session['usuario_id']          = user['id']
   170→   170→            session['usuario_nome']        = user['nome']
   171→   171→            session['usuario_role']        = user['role']
   172→   172→            session['usuario_departamento']= user.get('nome_departamento', '')
   173→   173→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   174→   174→            return redirect(url_for('dashboard'))
   175→   175→        else:
   176→   176→            _registrar_falha(ip)
   177→   177→            dados   = _login_attempts[ip]
   178→   178→            restantes = MAX_TENTATIVAS - dados['tentativas']
   179→   179→            if restantes > 0:
   180→   180→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   181→   181→            else:
   182→   182→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   183→   183→
   184→   184→    return render_template('login.html')

   715→   715→def _baixar_estoque_por_of(ordem):
   716→   716→    """Busca materiais da OF no ERP e dá baixa automática no estoque interno."""
   717→   717→    try:
   718→   718→        pg = DatabasePostgreSQL()
   719→   719→        materiais = pg.get_requisicoes_ordem(ordem)
   720→   720→        if not materiais:
   721→   721→            return
   722→   722→        for mat in materiais:
   723→   723→            codigo = str(mat.get('produto') or mat.get('reqproduto') or '')
   724→   724→            if not codigo:
   725→   725→                continue
   726→   726→            qtd = float(mat.get('quantidade') or mat.get('rqoquanti') or 0)
   727→   727→            if qtd <= 0:
   728→   728→                continue
   729→   729→            # Verifica se o material existe no estoque interno
   730→   730→            item = db.get_estoque_por_codigo(codigo)
   731→   731→            if item and item['qtd_atual'] > 0:
   732→   732→                db.registrar_baixa_estoque(codigo, qtd, ordem, 'liberacao_of')
   733→   733→                print(f"[ESTOQUE] Baixa automática: {codigo} -{qtd} (OP #{ordem})")
   734→   734→    except Exception as e:
   735→   735→        print(f"[ESTOQUE] Erro ao baixar estoque da OP #{ordem}: {e}")
