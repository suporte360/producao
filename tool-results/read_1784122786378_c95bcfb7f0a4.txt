   110→def _int(val):
   111→    """Safely convert to int (handles Decimal, None, str)."""
   112→    if val is None:
   113→        return 0
   114→    return int(float(val))

   116→def _str(val):
   117→    """Safely convert to stripped string."""
   118→    if val is None:
   119→        return ''
   120→    return str(val).strip()

   122→def _is_sob_medida(codigo):
   123→    """Verifica se o produto codigo termina com -X (sob medida)."""
   124→    if not codigo:
   125→        return False
   126→    c = str(codigo).strip().upper()
   127→    return c.endswith('-X')
   128→
   129→def _extract_s_code(codigo_produto):
   130→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   131→    if not codigo_produto:
   132→        return ''
   133→    m = re.search(r'S-\d+', str(codigo_produto))
   134→    return m.group(0) if m else str(codigo_produto).strip()

   147→def _pg_query_one(sql, params=None):
   148→    """Query PostgreSQL and return one row as dict."""
   149→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   150→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   151→    conn.autocommit = True
   152→    try:
   153→        with conn.cursor() as cur:
   154→            cur.execute(sql, params or ())
   155→            cols = [d[0] for d in cur.description] if cur.description else []
   156→            row = cur.fetchone()
   157→            if row:
   158→                return dict(zip(cols, row))
   159→            return None
   160→    finally:
   161→        conn.close()

   164→def _load_produto_cache():
   165→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   166→    global _produto_cache
   167→    _produto_cache = {}
   168→    # 1) PostgreSQL: tabela produto (pronome real)
   169→    try:
   170→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   171→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   172→        conn.autocommit = True
   173→        try:
   174→            with conn.cursor() as cur:
   175→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   176→                cols = [d[0] for d in cur.description]
   177→                for row in cur.fetchall():
   178→                    r = dict(zip(cols, row))
   179→                    cod = str(r.get('produto', '')).strip()
   180→                    nome = str(r.get('pronome', '')).strip()
   181→                    if cod and nome:
   182→                        _produto_cache[cod] = nome[:200]
   183→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   184→        finally:
   185→            conn.close()
   186→    except Exception as e:
   187→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   188→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   189→    try:
   190→        rows = db_query(
   191→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   192→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   193→        )
   194→        added = 0
   195→        for r in rows:
   196→            cod = str(r.get('codigo_produto', '')).strip()
   197→            nome = str(r.get('descricao_produto', '')).strip()
   198→            if cod and cod not in _produto_cache and nome:
   199→                _produto_cache[cod] = nome[:200]
   200→                added += 1
   201→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   202→    except Exception as e:
   203→        print('[WARN] Cache MariaDB falhou: %s' % e)

   208→def _refresh_lotes_ativos():

   216→        _lotes_ativos = set()
   217→        for r in rows:
   218→            if r.get('lote_codigo'):
   219→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   220→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   221→    except Exception as e:
   222→        print('[WARN] Lotes ativos falhou: %s' % e)

   224→
   225→def _build_lotes_filter():
   226→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   227→    if _lotes_ativos:
   228→        ph = ','.join(['%s'] * len(_lotes_ativos))
   229→        where = 'WHERE lote_codigo IN (' + ph + ')'
   230→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   231→        params = tuple(_lotes_ativos)
   232→    else:
   233→        where = 'WHERE 1=0'
   234→        join = ' AND 1=0'
   235→        params = ()
   236→    return where, join, params

   240→    """Retorna set de lote_codigo que pertencem ao(s) setor(es) do totem.
   241→    Usa departamento_atual do lote E operacoes pendentes."""
   242→    if not _lotes_ativos:

   244→    try:
   245→        ph = ','.join(['%s'] * len(_lotes_ativos))
   246→        dept_ph = ','.join(['%s'] * len(SETORES_SERRALHERIA))

   249→        rows1 = db_query(
   250→            "SELECT DISTINCT lote_codigo FROM lotes_producao "
   251→            "WHERE lote_codigo IN (" + ph + ") "
   252→            "AND departamento_atual IN (" + dept_ph + ")",
   253→            tuple(_lotes_ativos) + tuple(SETORES_SERRALHERIA)

   256→        # 2. Lotes com producao ativa nos setores do totem
   257→        rows2 = db_query(
   258→            "SELECT DISTINCT lote FROM serralheria_producao "
   259→            "WHERE lote IN (" + ph + ") "
   260→            "AND setor IN (" + dept_ph + ") AND status = 'em_producao'",
   261→            tuple(_lotes_ativos) + tuple(SETORES_SERRALHERIA)
   262→        )
   263→
   264→        resultado = set()
   265→        for r in rows1:
   266→            if r.get('lote_codigo'):
   267→                resultado.add(str(r['lote_codigo']).strip())
   268→        for r in rows2:
   269→            if r.get('lote'):
   270→                resultado.add(str(r['lote']).strip())
   271→        return resultado
   272→    except Exception as e:
   273→        app.logger.error('Erro _get_lotes_do_setor: %s', e)
   274→        return set()

   827→def _atualizar_departamento_lote(lote_ordem):
   828→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   829→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   830→    if not lote_ordem:
   831→        return
   832→    # Total de departamentos unicos do lote
   833→    total = db_query_one(
   834→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   835→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   836→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   837→        (lote_ordem,)
   838→    )
   839→    total_setores = int(total['t'] or 0) if total else 0
   840→
   841→    # Primeira operacao pendente (nao concluida) por sequencia
   842→    primeira = db_query_one(
   843→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   844→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   845→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   846→        "ORDER BY op.sequencia ASC LIMIT 1",
   847→        (lote_ordem,)
   848→    )
   849→    dept = primeira['departamento'] if primeira else None
   850→    seq = int(primeira['sequencia']) if primeira else 0
   851→
   852→    db_execute(
   853→        "UPDATE lotes_producao "
   854→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   855→        "WHERE ordem = %s",
   856→        (dept, seq, total_setores, lote_ordem)
   857→    )

   860→def _sync_lote_status(of_numero, novo_status):
   861→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   862→    if not of_numero:
   863→        return
   864→    of_row = db_query_one(
   865→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",
   866→        (of_numero,)
   867→    )
   868→    if not of_row or not of_row.get('lote_ordem'):
   869→        return
   870→    lote_ordem = of_row['lote_ordem']
   871→
   872→    if novo_status == 'em_andamento':
   873→        # Muda status do lote para em_producao se ainda estiver liberado
   874→        db_execute(
   875→            "UPDATE lotes_producao SET status = 'em_producao' "
   876→            "WHERE ordem = %s AND status = 'liberado'",
   877→            (lote_ordem,)
   878→        )
   879→    elif novo_status == 'concluido':
   880→        # Verifica se TODAS as ops do lote estao concluidas
   881→        total_pend = db_query_one(
   882→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   883→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   884→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   885→            (lote_ordem,)
   886→        )
   887→        if total_pend and int(total_pend['t']) == 0:
   888→            db_execute(
   889→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   890→                (lote_ordem,)
   891→            )

   894→def _sync_op_status(of_numero, departamento, novo_status):
   895→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   896→    if not of_numero or not departamento:
   897→        return
   898→    try:
   899→        # 1. Atualiza operacoes_producao
   900→        db_execute(
   901→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   902→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   903→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   904→            (novo_status, novo_status, of_numero, departamento)
   905→        )
   906→        # 2. Atualiza kanban_cards
   907→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   908→        nova_etapa = etapa_map.get(novo_status)
   909→        if nova_etapa:
   910→            db_execute(
   911→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   912→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   913→                (nova_etapa, of_numero, departamento)
   914→            )
   915→        # 3. Sincroniza lotes_producao (status + departamento_atual)
   916→        _sync_lote_status(of_numero, novo_status)
   917→        _atualizar_departamento_lote_from_of(of_numero)
   918→    except Exception as e:
   919→        app.logger.warning('Sync op status falhou: %s', e)

   957→                    "SELECT sp.id, sp.of_numero, sp.qtd_produzida, "
   958→                    "COALESCE(of2.qtde_ordem, 0) as qtde_ordem "
   959→                    "FROM serralheria_producao sp "
   960→                    "LEFT JOIN ordens_fabricacao of2 ON sp.of_numero = of2.of_numero "
   961→                    "WHERE sp.lote = %s AND sp.produto = %s AND sp.setor = %s "
   962→                    "AND sp.status = 'em_producao'",
   963→                    (lotcod, pcod, setor)
   964→                )
   965→                if not row:
   966→                    app.logger.warning('Finalizar: peca %s setor %s nao encontrada em producao', pcod, setor)
   967→                    continue
   968→
   969→                qtd_total = _int(row['qtde_ordem'])
   970→                qtd_ja_feita = _int(row['qtd_produzida'])
   971→                restante = qtd_total - qtd_ja_feita
   972→
   973→                if qtd_total <= 0:
   974→                    # Sem qtde_ordem, finaliza direto
   975→                    db_execute(
   976→                        "UPDATE serralheria_producao SET status='finalizado', data_fim=NOW() "
   977→                        "WHERE id=%s", (row['id'],)
   978→                    )
   979→                    dept = setor_para_dept.get(setor, setor)
   980→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   981→                    continue
   982→
   983→                # Validacao: quantidade nao pode exceder o restante
   984→                if qtd_informada > restante:
   985→                    return jsonify({
   986→                        'error': 'Quantidade maxima permitida: %d (restam %d de %d)' % (restante, restante, qtd_total),
   987→                        'max_qtd': restante
   988→                    }), 400
   989→
   990→                if qtd_informada <= 0:
   991→                    return jsonify({'error': 'Informe uma quantidade maior que zero'}), 400
   992→
   993→                nova_qtd = qtd_ja_feita + qtd_informada
   994→
   995→                if nova_qtd >= qtd_total:
   996→                    # Finalizacao completa
   997→                    db_execute(
   998→                        "UPDATE serralheria_producao "
   999→                        "SET status='finalizado', data_fim=NOW(), qtd_produzida=%s "
  1000→                        "WHERE id=%s",
  1001→                        (qtd_total, row['id'])
  1002→                    )
  1003→                    dept = setor_para_dept.get(setor, setor)
  1004→                    _sync_op_status(row['of_numero'], dept, 'concluido')
  1005→                    app.logger.info('Lote %s peca %s setor %s: FINALIZADA (%d/%d)', lotcod, pcod, setor, nova_qtd, qtd_total)
  1006→                else:
  1007→                    # Finalizacao parcial: acumula e mantem em_producao
  1008→                    db_execute(
  1009→                        "UPDATE serralheria_producao SET qtd_produzida=%s WHERE id=%s",
  1010→                        (nova_qtd, row['id'])
  1011→                    )
  1012→                    app.logger.info('Lote %s peca %s setor %s: PARCIAL %d/%d (restam %d)',
  1013→                        lotcod, pcod, setor, nova_qtd, qtd_total, qtd_total - nova_qtd)
  1014→
  1015→            return jsonify({'success': True, 'mensagem': 'Producao atualizada!'})
  1016→        elif pecas:
  1017→            # Finalizar todas as producoes das pecas selecionadas (todos os setores) — sem qtde parcial
  1018→            placeholders = ','.join(['%s'] * len(pecas))
  1019→            rows = db_query(
  1020→                f"SELECT of_numero, setor FROM serralheria_producao "
  1021→                f"WHERE lote = %s AND produto IN ({placeholders}) AND status = 'em_producao'",
  1022→                [lotcod] + pecas
  1023→            )
  1024→            db_execute(
  1025→                f"UPDATE serralheria_producao "
  1026→                f"SET status = 'finalizado', data_fim = NOW() "
  1027→                f"WHERE lote = %s AND produto IN ({placeholders}) AND status = 'em_producao'",
  1028→                [lotcod] + pecas
  1029→            )
  1030→            for row in rows:
  1031→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
  1032→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
  1033→            app.logger.info('Lote %s: %d peca(s) finalizada(s) no totem', lotcod, len(pecas))
  1034→        else:
  1035→            # Finalizar todas as pecas do lote
  1036→            rows = db_query(
  1037→                "SELECT of_numero, setor FROM serralheria_producao "
  1038→                "WHERE lote = %s AND status = 'em_producao'", (lotcod,)
  1039→            )
  1040→            db_execute(
  1041→                "UPDATE serralheria_producao "
  1042→                "SET status = 'finalizado', data_fim = NOW() "
  1043→                "WHERE lote = %s AND status = 'em_producao'",
  1044→                (lotcod,)
  1045→            )
  1046→            for row in rows:
  1047→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
  1048→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
  1049→            app.logger.info('Lote %s finalizado no totem', lotcod)
  1050→
  1051→        return jsonify({'success': True, 'mensagem': 'Producao finalizada!'})
  1052→    except Exception as e:
  1053→        app.logger.error('Erro finalizar: %s', e)
  1054→        return jsonify({'error': str(e)}), 500
