    29→def _load_produto_cache():
    30→    """Carrega nomes reais dos produtos do PostgreSQL."""
    31→    global _produto_cache
    32→    try:
    33→        pg = DatabasePostgreSQL()
    34→        rows = pg.query(
    35→            "SELECT produto, pronome FROM public.produto "
    36→            "WHERE pronome IS NOT NULL AND pronome != ''"
    37→        )
    38→        _produto_cache = {}
    39→        for r in rows:
    40→            cod = str(r.get('produto', '')).strip()
    41→            nome = str(r.get('pronome', '')).strip()
    42→            if cod:
    43→                _produto_cache[cod] = nome[:200]
    44→        print('[OK] Cache de produtos carregado: %d produtos do ERP' % len(_produto_cache))
    45→    except Exception as e:
    46→        print('[WARN] Cache de produtos falhou: %s' % e)

    48→def _get_nome_produto(codigo):
    49→    """Retorna nome real do produto pelo codigo."""
    50→    if not codigo:
    51→        return ''
    52→    return _produto_cache.get(str(codigo).strip(), '')

    54→def _is_sob_medida(codigo):
    55→    """Verifica se o produto codigo termina com -X (sob medida)."""
    56→    if not codigo:
    57→        return False
    58→    c = str(codigo).strip().upper()
    59→    # S-0440-X, CJ-0174-X, qualquer codigo terminado em -X
    60→    return c.endswith('-X')

    62→def _resolver_nome_lote(lote):
    63→    """Resolve nome real para descricao_produto de um lote/OF."""
    64→    desc = lote.get('descricao_produto', '') or ''
    65→    cod = lote.get('codigo_produto', '') or ''
    66→    if not desc or desc.upper().startswith('PRODUCAO DE'):
    67→        nome = _get_nome_produto(cod)
    68→        if nome:
    69→            return nome
    70→    return desc

   253→# =============================================
   254→# Proteção contra força bruta no login
   255→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
   256→# =============================================
   257→
   258→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
   259→MAX_TENTATIVAS  = 5
   260→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
   261→
   262→def _get_ip():
   263→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
   264→
   265→def _checar_bloqueio(ip):
   266→    dados = _login_attempts[ip]
   267→    if time.time() < dados['bloqueado_ate']:
   268→        restante = int(dados['bloqueado_ate'] - time.time())
   269→        return True, restante
   270→    return False, 0
   271→
   272→def _registrar_falha(ip):
   273→    dados = _login_attempts[ip]
   274→    dados['tentativas'] += 1
   275→    if dados['tentativas'] >= MAX_TENTATIVAS:
   276→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   277→        dados['tentativas'] = 0
   278→
   279→def _limpar_tentativas(ip):
   280→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}
   281→

   286→def login_required(f):
   287→    @wraps(f)
   288→    def decorated(*args, **kwargs):
   289→        if 'usuario_id' not in session:
   290→            flash('Faça login para continuar.', 'warning')
   291→            return redirect(url_for('login'))
   292→        return f(*args, **kwargs)
   293→    return decorated
   294→
   295→def role_required(*roles):
   296→    def decorator(f):
   297→        @wraps(f)
   298→        def decorated(*args, **kwargs):
   299→            if session.get('usuario_role') not in roles:
   300→                flash('Acesso não autorizado.', 'error')
   301→                return redirect(url_for('dashboard'))
   302→            return f(*args, **kwargs)
   303→        return decorated
   304→    return decorator

   310→@app.route('/login', methods=['GET', 'POST'])
   311→def login():
   312→    if 'usuario_id' in session:
   313→        return redirect(url_for('dashboard'))
   314→
   315→    if request.method == 'POST':
   316→        ip = _get_ip()
   317→
   318→        bloqueado, restante = _checar_bloqueio(ip)
   319→        if bloqueado:
   320→            mins = restante // 60
   321→            secs = restante % 60
   322→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   323→            return render_template('login.html')
   324→
   325→        usuario = request.form.get('usuario', '').strip()
   326→        senha   = request.form.get('senha', '')
   327→
   328→        if not usuario or not senha:
   329→            flash('Informe usuário e senha.', 'error')
   330→            return render_template('login.html')
   331→
   332→        user = db.verificar_usuario(usuario, senha)
   333→
   334→        if user:
   335→            _limpar_tentativas(ip)
   336→            session.permanent = True
   337→            session['usuario_id']          = user['id']
   338→            session['usuario_nome']        = user['nome']
   339→            session['usuario_role']        = user['role']
   340→            session['usuario_departamento']= user.get('nome_departamento', '')
   341→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   342→            return redirect(url_for('dashboard'))
   343→        else:
   344→            _registrar_falha(ip)
   345→            dados   = _login_attempts[ip]
   346→            restantes = MAX_TENTATIVAS - dados['tentativas']
   347→            if restantes > 0:
   348→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   349→            else:
   350→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   351→
   352→    return render_template('login.html')

   477→@app.route('/kanban/setores')
   478→@login_required
   479→@role_required('admin', 'gerente', 'diretor')
   480→def kanban_setores():
   481→    """Kanban visual organizado por setor/departamento."""
   482→    departamentos = db.listar_departamentos(apenas_ativos=True)
   483→    todos_cards = db.get_kanban_cards(departamento=None)
   484→
   485→    # Agrupar cards por departamento
   486→    cards_por_setor = {}
   487→    for dept in departamentos:
   488→        cards_por_setor[dept['codigo']] = {
   489→            'info': dept,
   490→            'cards': [],
   491→            'total': 0,
   492→            'em_producao': 0,
   493→            'aguardando': 0,
   494→            'pausados': 0,
   495→            'concluidos': 0,
   496→        }
   497→
   498→    for card in todos_cards:
   499→        dept_code = card.get('departamento', '')
   500→        if dept_code in cards_por_setor:
   501→            cards_por_setor[dept_code]['cards'].append(card)
   502→            cards_por_setor[dept_code]['total'] += 1
   503→            etapa = card.get('etapa', '')
   504→            if etapa == 'em_producao':
   505→                cards_por_setor[dept_code]['em_producao'] += 1
   506→            elif etapa in ('aguardando', 'pendente'):
   507→                cards_por_setor[dept_code]['aguardando'] += 1
   508→            elif etapa == 'pausado':
   509→                cards_por_setor[dept_code]['pausados'] += 1
   510→            elif etapa == 'concluido':
   511→                cards_por_setor[dept_code]['concluidos'] += 1
   512→
   513→    # KPIs gerais
   514→    kpis = {
   515→        'setores_ativos': sum(1 for s in cards_por_setor.values() if s['total'] > 0),
   516→        'total_cards': len(todos_cards),
   517→        'em_producao': sum(1 for c in todos_cards if c.get('etapa') == 'em_producao'),
   518→        'concluidos_hoje': db.get_kanban_concluidos_hoje(),
   519→    }
   520→
   521→    return render_template('kanban/setores.html',
   522→                           cards_por_setor=cards_por_setor,
   523→                           kpis=kpis,
   524→                           usuario_role=session.get('usuario_role'))

   566→def pcp_importar_erp():
   567→    pg_ok   = False
   568→    pg_erro = ''
   569→    lotes_erp = []
   570→    lotes_ja_importados = set()
   571→
   572→    try:
   573→        pg = DatabasePostgreSQL()
   574→        # Busca lotes do ERP com produto final e maior ordem
   575→        lotes_raw = pg.get_lotes_agrupados_abertos()
   576→        # Para cada lote busca o produto final
   577→        for lote in lotes_raw:
   578→            prod = pg.get_produto_pronto_por_lote(lote['lote_codigo'])
   579→            lote['nome_produto'] = prod.get('nome_produto') if prod else None
   580→            # produto_final_codigo ja vem correto do ordem3, so sobrescreve se pronome foi encontrado
   581→            if prod and prod.get('nome_produto') and prod.get('codigo_produto'):
   582→                lote['produto_final_codigo'] = prod['codigo_produto']
   583→            lote['maior_ordem'] = prod.get('quantidade') if prod else None
   584→            # busca a maior ordem numerica do lote
   585→            ordens = pg.get_ordens_por_lote(lote['lote_codigo'])
   586→            if ordens:
   587→                lote['maior_ordem'] = max(int(o['ordem']) for o in ordens if o.get('ordem') and str(o['ordem']).isdigit())
   588→            lote['sob_medida'] = _is_sob_medida(lote.get('produto_final_codigo')) or _is_sob_medida(lote.get('lote_codigo'))
   589→        lotes_erp = lotes_raw
   590→        pg_ok = True
   591→    except Exception as e:
   592→        pg_erro = str(e)
   593→
   594→    # Lotes já importados no MariaDB
   595→    ja = db.query("SELECT DISTINCT lote_codigo FROM lotes_producao WHERE lote_codigo IS NOT NULL")
   596→    lotes_ja_importados = {r['lote_codigo'] for r in ja}
   597→
   598→    return render_template('pcp/importar_erp.html',
   599→                           pg_ok=pg_ok, pg_erro=pg_erro,
   600→                           lotes_erp=lotes_erp,

   606→@app.route('/pcp/importar_erp', methods=['POST'])
   607→@login_required
   608→@role_required('admin', 'gerente', 'pcp', 'diretor')
   609→def pcp_executar_importacao():
   610→    data  = request.json or {}
   611→    lotes = data.get('lotes', [])
   612→    if not lotes:
   613→        return jsonify({'status':'error','message':'Nenhum lote informado'}), 400
   614→
   615→    importados = 0
   616→    atualizados = 0
   617→    erros = []
   618→
   619→    try:
   620→        pg = DatabasePostgreSQL()
   621→    except Exception as e:
   622→        return jsonify({'status':'error','message': f'Erro PostgreSQL: {e}'}), 500
   623→
   624→    for item in lotes:
   625→        try:
   626→            lote_cod = str(item.get('lote_codigo',''))
   627→            ordem_num = item.get('ordem')
   628→            if not lote_cod:
   629→                continue
   630→
   631→            # Verifica se já existe
   632→            existente = db.query_one(
   633→                "SELECT ordem FROM lotes_producao WHERE lote_codigo=%s", (lote_cod,))
   634→            ja_existia = existente is not None
   635→
   636→            # Importa lote principal
   637→            db.importar_lote_erp({
   638→                'ordem':             int(ordem_num) if str(ordem_num).isdigit() else 0,
   639→                'lote_codigo':       lote_cod,
   640→                'codigo_produto':    item.get('produto',''),
   641→                'descricao_produto': item.get('nome','') or item.get('produto',''),
   642→                'qtde_ordem':        float(item.get('qtd', 0) or 0),
   643→                'unidade_medida':    'PC',
   644→                'status_erp':        'A',
   645→                'data_previsao_erp': item.get('previsao') or None,
   646→                'data_abertura_erp': item.get('abertura') or None,
   647→                'planejador':        session.get('usuario_nome'),
   648→                'observacoes':       item.get('observacoes') or '',
   649→            })
   650→
   651→            lote_ordem_local = int(ordem_num) if str(ordem_num).isdigit() else 0
   652→
   653→            # Busca ordens do lote no ERP e importa OFs
   654→            ordens_erp = pg.get_ordens_por_lote(lote_cod)
   655→            for of in ordens_erp:
   656→                of_num = of.get('ordem')
   657→                if not of_num:
   658→                    continue
   659→                db.importar_of({
   660→                    'lote_ordem':       lote_ordem_local,
   661→                    'of_numero':        str(of_num),
   662→                    'codigo_produto':   of.get('produto',''),
   663→                    'descricao_produto': item.get('nome','') or item.get('produto',''),
   664→                    'qtde_ordem':       float(of.get('quantidade') or 0),
   665→                    'unidade_medida':   'PC',
   666→                    'tipo':             'filho' if of.get('nivel_producao','') != '1' else 'pai',
   667→                    'data_previsao':    of.get('data_previsao') or None,
   668→                })
   669→
   670→                # Busca processos (operações) de cada OF
   671→                processos = pg.get_processos_ordem(str(of_num))
   672→                for proc in processos:
   673→                    dept_original = (proc.get('nome_departamento') or '').upper().strip()
   674→                    if not dept_original:
   675→                        continue
   676→                    # Limpa parênteses e sujeiras do nome do ERP (ex: "CORTE()" → "CORTE")
   677→                    dept_original = re.sub(r'\s*\([^)]*\)\s*$', '', dept_original).strip()
   678→                    # Mapeia nomes do ERP (fases) para nomes do sistema local
   679→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   680→                    # Correcao automatica: se a operacao envolve dobra, forca DOBRA
   681→                    desc_op = (proc.get('descricao_processo') or proc.get('operacao','')).upper()
   682→                    if 'DOBRAR' in desc_op and dept not in ('DOBRA',):
   683→                        dept = 'DOBRA'
   684→                    db.importar_operacao({
   685→                        'of_numero':          str(of_num),
   686→                        'lote_ordem':         lote_ordem_local,
   687→                        'sequencia':          proc.get('numero_sequencial', 0),
   688→                        'fase':               proc.get('fase',''),
   689→                        'descricao_operacao': proc.get('descricao_processo') or proc.get('operacao',''),
   690→                        'departamento':       dept,
   691→                        'codigo_barras':      None,
   692→                    })
   693→
   694→                # Sincroniza kanban
   695→                try:
   696→                    db.sincronizar_kanban_com_operacoes(lote_ordem_local)
   697→                except Exception:
   698→                    pass
   699→
   700→                # Atualiza departamento_atual baseado na primeira op pendente
   701→                try:
   702→                    db.atualizar_departamento_atual(lote_ordem_local)
   703→                except Exception:
   704→                    pass
   705→
   706→            db.add_log(session['usuario_id'], 'importar_erp',
   707→                       f'Lote {lote_cod} importado do ERP', 'lotes_producao', lote_ordem_local)
   708→
   709→            if ja_existia:
   710→                atualizados += 1
   711→            else:
   712→                importados += 1
   713→
   714→        except Exception as e:
   715→            erros.append(f"{item.get('lote_codigo')}: {e}")
   716→
   717→    return jsonify({
   718→        'status':     'success',
   719→        'importados':  importados,
   720→        'atualizados': atualizados,
   721→        'erros':       erros,
   722→    })

  1151→def _baixar_estoque_por_of(ordem):
