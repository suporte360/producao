   478→/* ─── Cronômetro de card ─── */
   479→function tempoDecorrido(isoStr) {
   480→  if (!isoStr) return '';
   481→  try {
   482→    const p = isoStr.replace('T',' ').split(/[- :]/).map(Number);
   483→    const inicio = new Date(p[0],p[1]-1,p[2],p[3]||0,p[4]||0,p[5]||0);
   484→    const diff = Math.max(0, Math.floor((Date.now() - inicio.getTime())/1000));
   485→    const h = String(Math.floor(diff/3600)).padStart(2,'0');
   486→    const m = String(Math.floor((diff%3600)/60)).padStart(2,'0');
   487→    const s = String(diff%60).padStart(2,'0');
   488→    return `${h}:${m}:${s}`;
   489→  } catch(e){ return ''; }
   490→}

   511→function buildCard(card) {
   512→  const prio = (card.prioridade||'media').toLowerCase();
   513→  const timer = card.etapa === 'em_producao' && card.data_inicio
   514→    ? '<span class="kc-timer" style="color:#4ade80" data-inicio="' + card.data_inicio + '">' + tempoDecorrido(card.data_inicio) + '</span>'
   515→    : '';
   516→  const pausaBadge = card.etapa === 'pausado'
   517→    ? '<span style="display:inline-flex;align-items:center;gap:3px;font-size:.5rem;font-weight:800;padding:1px 6px;border-radius:9999px;background:rgba(245,158,11,.2);color:#f59e0b"><i class="fa-solid fa-pause" style="font-size:.45rem"></i>PAUSADO</span>'
   518→    : '';
   519→  const badge = '<span class="badge-p ' + prio + '">' + ((card.prioridade||'MEDIA')).toUpperCase() + '</span>';
   520→  const pecaCode = card.cod_peca || '';
   521→  const pecaName = card.nome_peca || '';
   522→  const prod = pecaCode
   523→    ? '<span style="color:#60a5fa;font-weight:700">' + pecaCode + '</span>' + (card.sob_medida ? ' <span style="font-size:.5rem;font-weight:700;padding:1px 5px;border-radius:9999px;background:#e9d5ff;color:#a855f7">SOB MEDIDA</span>' : '') + (pecaName ? ' <span style="color:#ccc">' + pecaName.substring(0,35) + '</span>' : '')
   524→    : (card.descricao_operacao || 'Produto').substring(0,45);
   525→  const op2  = card.descricao_operacao
   526→    ? '<div class="kc-op2"><i class="fa-solid fa-screwdriver-wrench" style="font-size:.55rem;margin-right:3px"></i>' + card.descricao_operacao.substring(0,38) + '</div>' : '';
   527→  const qtd  = card.qtde_ordem ? '<span style="font-size:.6rem;color:var(--muted)">' + Math.round(card.qtde_ordem) + 'pc</span>' : '';
   528→  var prog = card.progresso || 0;
   529→  var progColor = prog >= 75 ? '#4ade80' : (prog >= 40 ? '#fbbf24' : '#f87171');
   530→  var progBar = card.etapa !== 'concluido'
   531→    ? '<div style="display:flex;align-items:center;gap:4px;margin-top:4px"><div style="flex:1;height:3px;background:rgba(255,255,255,.1);border-radius:2px;overflow:hidden"><div style="height:100%;width:' + prog + '%;background:' + progColor + ';border-radius:2px;transition:width .5s"></div></div><span style="font-size:.55rem;color:' + progColor + ';font-weight:700">' + prog + '%</span></div>'
   532→    : '';
   533→  return '<div class="k-card p-' + prio + '">'
   534→    + '<div class="kc-op">OP #' + card.lote_ordem + (card.lote_codigo ? ' &middot; ' + card.lote_codigo : '') + '</div>'
   535→    + '<div class="kc-prod">' + prod + '</div>'
   536→    + op2
   537→    + '<div class="kc-meta">' + badge + pausaBadge + qtd + timer + '</div>'
   538→    + progBar
   539→    + '</div>';
   540→}
