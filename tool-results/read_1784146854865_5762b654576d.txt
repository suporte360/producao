   289→  reqAtual = id;

   294→  document.getElementById('modalEntregarReq').classList.add('hidden');

   327→// ════════════════ RESUMO MATERIAL NA FABRICA ════════════════
   328→function carregarResumoEstoque() {
   329→  fetch('/api/estoque?filtro=baixo&tipo=fabrica').then(r => r.json()).then(d => {
   330→    if (d.status !== 'success') return;
   331→    const itens = d.itens || [];
   332→    const badge = document.getElementById('kpiPrecisaSep');
   333→    if (itens.length > 0) {
   334→      badge.textContent = itens.length + ' precisam separar';
   335→      badge.classList.remove('hidden');
   336→    } else {
   337→      badge.classList.add('hidden');
   338→    }
   339→    const el = document.getElementById('listaEstoqueResumo');
   340→    if (!itens.length) {
   341→      el.innerHTML = '<div class="prodsys-card p-4 text-center col-span-full text-sm text-green-600"><i class="fa-solid fa-circle-check mr-1"></i>Todos os materiais estao OK</div>';
   342→      return;
   343→    }
   344→    el.innerHTML = itens.slice(0, 6).map(item => {
   345→      const qtd = parseFloat(item.qtd_atual) || 0;
   346→      const qtdIni = parseFloat(item.qtd_inicial) || 0;
   347→      const sem = qtd <= 0;
   348→      const desc = (item.descricao||'—').replace(/'/g,"\\'");
   349→      const un = item.unidade || 'UN';
   350→      const pct = qtdIni > 0 ? Math.round((qtd / qtdIni) * 100) : 0;
   351→      return '<div class="prodsys-card border-l-4 ' + (sem ? 'border-red-500 bg-red-50/30' : 'border-amber-400 bg-amber-50/30') + ' px-3 py-2 flex items-center justify-between gap-2">' +
   352→        '<div class="min-w-0"><div class="font-bold text-sm text-gray-800 truncate">' + (item.descricao||'—') + '</div>' +
   353→        '<div class="text-[11px] text-gray-400">' + item.codigo + ' · ' + un + '</div></div>' +
   354→        '<div class="text-right flex-shrink-0"><div class="font-black text-lg ' + (sem?'text-red-600':'text-amber-600') + '">' + qtd.toFixed(1) + '</div>' +
   355→        '<div class="text-[10px] text-gray-400">de ' + qtdIni.toFixed(1) + (pct > 0 ? ' (' + pct + '%)' : '') + '</div></div></div>';
   356→    }).join('');
   357→  });
   358→}
