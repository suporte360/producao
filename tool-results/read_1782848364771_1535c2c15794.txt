    12→
    13→class DatabaseMySQL:
    14→    _initialized = False
    15→    _instance = None
    16→
    17→    def __new__(cls):
    18→        if cls._instance is None:
    19→            cls._instance = super(DatabaseMySQL, cls).__new__(cls)
    20→        return cls._instance
    21→
    22→    def __init__(self):
    23→        if self._initialized:
    24→            return
    25→        self._initialized = True
    26→        self.config = {
    27→            'host': Config.MYSQL_HOST,
    28→            'port': Config.MYSQL_PORT,
    29→            'user': Config.MYSQL_USERNAME,
    30→            'password': Config.MYSQL_PASSWORD,
    31→            'database': Config.MYSQL_DATABASE,
    32→            'charset': 'utf8mb4',
    33→            'cursorclass': DictCursor,
    34→            'autocommit': False,
    35→            'connect_timeout': 10,
    36→        }
    37→
    38→    @contextmanager
    39→    def get_connection(self):
    40→        connection = pymysql.connect(**self.config)
    41→        try:
    42→            yield connection
    43→            connection.commit()
    44→        except Exception as e:
    45→            connection.rollback()
    46→            raise e
    47→        finally:
    48→            connection.close()
    49→
    50→    def query(self, sql, params=None):
    51→        with self.get_connection() as conn:
    52→            with conn.cursor() as cursor:
    53→                cursor.execute(sql, params or ())
    54→                return cursor.fetchall()
    55→
    56→    def query_one(self, sql, params=None):
    57→        with self.get_connection() as conn:
    58→            with conn.cursor() as cursor:
    59→                cursor.execute(sql, params or ())
    60→                return cursor.fetchone()
    61→
    62→    def execute(self, sql, params=None):
    63→        with self.get_connection() as conn:
    64→            with conn.cursor() as cursor:
    65→                cursor.execute(sql, params or ())
    66→                return cursor.lastrowid or cursor.rowcount

    75→        if user:
    76→            if user['senha'].startswith('scrypt:') or user['senha'].startswith('pbkdf2:'):
    77→                if check_password_hash(user['senha'], senha):
    78→                    self.execute("UPDATE usuarios SET ultimo_acesso = NOW() WHERE id = %s", (user['id'],))
    79→                    return user
    80→            elif user['senha'] == senha:
    81→                self.execute("UPDATE usuarios SET ultimo_acesso = NOW() WHERE id = %s", (user['id'],))
    82→                return user
    83→        return None

   141→    def get_lotes(self, status=None, departamento=None, prioridade=None, pagina=1, por_pagina=30):
   142→        offset = (pagina - 1) * por_pagina
   143→        sql = """
   144→            SELECT l.*, 
   145→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem) AS total_operacoes,
   146→                   (SELECT COUNT(*) FROM operacoes_producao op WHERE op.lote_ordem = l.ordem AND op.status = 'concluido') AS operacoes_concluidas
   147→            FROM lotes_producao l
   148→            WHERE 1=1
   149→        """
   150→        params = []
   151→        if status:
   152→            if isinstance(status, list):
   153→                placeholders = ','.join(['%s'] * len(status))
   154→                sql += f" AND l.status IN ({placeholders})"
   155→                params.extend(status)
   156→            else:
   157→                sql += " AND l.status = %s"
   158→                params.append(status)
   159→        if departamento:
   160→            sql += " AND l.departamento_atual = %s"
   161→            params.append(departamento)
   162→        if prioridade:
   163→            sql += " AND l.prioridade = %s"
   164→            params.append(prioridade)
   165→        sql += """
   166→            ORDER BY
   167→                FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'),
   168→                l.data_previsao_erp ASC
   169→            LIMIT %s OFFSET %s
   170→        """
   171→        params.extend([por_pagina, offset])
   172→        return self.query(sql, params)

   184→    def get_lotes_por_departamento(self, departamento, status=None):
   185→        """
   186→        Retorna 1 linha por OP/LOTE para o departamento, com a próxima operação pendente.
   187→        Evita duplicação: pega só a operação mais antiga pendente/em_andamento do setor.
   188→        """
   189→        sql = """
   190→            SELECT l.*,
   191→                   op.id            AS op_id,
   192→                   op.status        AS op_status,
   193→                   op.data_inicio   AS data_inicio,
   194→                   op.descricao_operacao,
   195→                   op.qtde_produzida,
   196→                   op.sequencia,
   197→                   op.of_numero,
   198→                   (SELECT COUNT(*) FROM operacoes_producao op2
   199→                    WHERE op2.lote_ordem = l.ordem AND op2.departamento = %s
   200→                      AND op2.status != 'concluido') AS ops_pendentes
   201→            FROM lotes_producao l
   202→            INNER JOIN operacoes_producao op ON op.lote_ordem = l.ordem
   203→            WHERE op.departamento = %s
   204→              AND op.status IN ('pendente', 'em_andamento', 'pausado')
   205→              AND l.status NOT IN ('cancelado', 'finalizado')
   206→              AND op.id = (
   207→                  SELECT MIN(op3.id)
   208→                  FROM operacoes_producao op3
   209→                  WHERE op3.lote_ordem = l.ordem
   210→                    AND op3.departamento = %s
   211→                    AND op3.status IN ('pendente', 'em_andamento', 'pausado')
   212→              )
   213→        """
   214→        params = [departamento, departamento, departamento]
   215→        if status:
   216→            sql += " AND l.status = %s"
   217→            params.append(status)
   218→        sql += " ORDER BY FIELD(l.prioridade, 'urgente', 'alta', 'media', 'baixa'), l.data_previsao_erp ASC"
   219→        return self.query(sql, params)

   350→    def iniciar_operacao(self, operacao_id, usuario_id):
   351→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   352→        if not op:
   353→            return False
   354→        self.execute("""
   355→            UPDATE operacoes_producao 
   356→            SET status = 'em_andamento', data_inicio = NOW(), usuario_inicio_id = %s
   357→            WHERE id = %s AND status IN ('pendente', 'pausado')
   358→        """, (usuario_id, operacao_id))
   359→        self.atualizar_status_lote(op['lote_ordem'], 'em_producao', usuario_id)
   360→        self.registrar_apontamento_tempo(op['lote_ordem'], op['of_numero'], operacao_id, op['departamento'], usuario_id, 'inicio')
   361→        # Sincroniza kanban: card vai para 'em_producao'
   362→        self._sincronizar_kanban_operacao(operacao_id, 'em_producao', usuario_id)
   363→        return True

   389→    def _sincronizar_kanban_operacao(self, operacao_id, etapa, operador_id=None):
   390→        """Atualiza o card do kanban correspondente à operação.
   391→        Se o card não existir, cria automaticamente."""
   392→        try:
   393→            # Busca a operação e o lote
   394→            op = self.query_one("""
   395→                SELECT op.*, l.prioridade
   396→                FROM operacoes_producao op
   397→                LEFT JOIN lotes_producao l ON op.lote_ordem = l.ordem
   398→                WHERE op.id = %s
   399→            """, (operacao_id,))
   400→            if not op:
   401→                return
   402→
   403→            # Verifica se já existe card para esta operação
   404→            existente = self.query_one(
   405→                "SELECT id FROM kanban_cards WHERE operacao_id = %s",
   406→                (operacao_id,)
   407→            )
   408→
   409→            if existente:
   410→                # Atualiza card existente
   411→                self.execute("""
   412→                    UPDATE kanban_cards
   413→                    SET etapa = %s,
   414→                        operador_id = COALESCE(%s, operador_id),
   415→                        atualizado_em = NOW()
   416→                    WHERE operacao_id = %s
   417→                """, (etapa, operador_id, operacao_id))
   418→            else:
   419→                # Cria card novo (para operações importadas antes do fix)
   420→                self.execute("""
   421→                    INSERT INTO kanban_cards
   422→                        (lote_ordem, of_numero, operacao_id, etapa, departamento, prioridade, operador_id)
   423→                    VALUES (%s, %s, %s, %s, %s, %s, %s)
   424→                """, (
   425→                    op['lote_ordem'],
   426→                    op.get('of_numero'),
   427→                    operacao_id,
   428→                    etapa,
   429→                    op.get('departamento'),
   430→                    op.get('prioridade') or 'media',
   431→                    operador_id
   432→                ))
   433→        except Exception as e:
   434→            print(f"Erro ao sincronizar kanban: {e}")

   436→    def finalizar_operacao(self, operacao_id, usuario_id, qtde_produzida=None, qtde_refugada=0, observacoes=None, tipo='total'):
   437→        """
   438→        tipo='total'   → marca como concluido e fecha o lote se todas as ops concluídas
   439→        tipo='parcial' → registra a quantidade produzida até agora e reinicia a operação
   440→                         (status volta para 'pendente' com qtde_produzida acumulada)
   441→                         Assim o próximo setor já pode ver as peças disponíveis.
   442→        """
   443→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   444→        if not op:
   445→            return False
   446→
   447→        if tipo == 'parcial':
   448→            # Acumula a quantidade e volta para pendente (disponível para próximo setor)
   449→            self.execute("""
   450→                UPDATE operacoes_producao
   451→                SET status = 'pendente',
   452→                    data_inicio = NULL,
   453→                    qtde_produzida = COALESCE(qtde_produzida, 0) + COALESCE(%s, 0),
   454→                    qtde_refugada  = COALESCE(qtde_refugada, 0)  + COALESCE(%s, 0),
   455→                    observacoes    = COALESCE(%s, observacoes)
   456→                WHERE id = %s
   457→            """, (qtde_produzida, qtde_refugada, observacoes, operacao_id))
   458→            # Atualiza status do lote para 'importado' (disponível) se estava em_producao
   459→            self.execute("""
   460→                UPDATE lotes_producao SET status = 'importado'
   461→                WHERE ordem = %s AND status = 'em_producao'
   462→            """, (op['lote_ordem'],))
   463→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'parcial', qtde_produzida=qtde_produzida)
   464→            # Sincroniza kanban: card volta para 'aguardando' (próximo setor pega)
   465→            self._sincronizar_kanban_operacao(operacao_id, 'aguardando')
   466→        else:
   467→            self.execute("""
   468→                UPDATE operacoes_producao
   469→                SET status = 'concluido', data_fim = NOW(), usuario_fim_id = %s,
   470→                    qtde_produzida = COALESCE(%s, qtde_produzida),
   471→                    qtde_refugada = COALESCE(%s, 0),
   472→                    observacoes = COALESCE(%s, observacoes)
   473→                WHERE id = %s
   474→            """, (usuario_id, qtde_produzida, qtde_refugada, observacoes, operacao_id))
   475→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'fim', qtde_produzida=qtde_produzida)
   476→            # Sincroniza kanban: card vai para 'concluido'
   477→            self._sincronizar_kanban_operacao(operacao_id, 'concluido', usuario_id)
   478→            # Verifica se todas as operações do lote foram concluídas
   479→            self._verificar_conclusao_lote(op['lote_ordem'])
   480→        return True

   648→    def sincronizar_kanban_com_operacoes(self, lote_ordem):

   654→        for op in operacoes:

   656→            if not existente:
   657→                etapa = 'aguardando'
   658→                if op['status'] == 'em_andamento':
   659→                    etapa = 'em_producao'
   660→                elif op['status'] == 'concluido':
   661→                    etapa = 'concluido'
   662→                elif op['status'] == 'pausado':
   663→                    etapa = 'pausado'
   664→                self.add_kanban_card(
   665→                    lote_ordem=lote_ordem,
   666→                    etapa=etapa,
   667→                    prioridade=lote['prioridade'],
   668→                    departamento=op['departamento'],
   669→                    of_numero=op['of_numero'],

   963→    def get_estoque_interno(self, status_filtro=None):
   964→        """Retorna todos os itens do estoque interno. status_filtro: 'baixo' | 'normal' | None(todos)"""
   965→        sql = "SELECT * FROM estoque_interno WHERE 1=1"
   966→        params = []
   967→        if status_filtro == 'baixo':
   968→            sql += " AND (qtd_atual <= qtd_minima OR qtd_atual <= 0)"
   969→        elif status_filtro == 'normal':
   970→            sql += " AND qtd_atual > qtd_minima AND qtd_atual > 0"
   971→        sql += " ORDER BY (qtd_atual <= qtd_minima OR qtd_atual <= 0) DESC, descricao ASC"
   972→        return self.query(sql, params)

   974→    def get_estoque_kpis(self):
   975→        """KPIs do estoque interno para o dashboard do almoxarife."""
   976→        sql = """
   977→            SELECT
   978→                COUNT(*) AS total_itens,
   979→                SUM(CASE WHEN qtd_atual <= 0 THEN 1 ELSE 0 END) AS sem_estoque,
   980→                SUM(CASE WHEN qtd_atual > 0 AND qtd_atual <= qtd_minima THEN 1 ELSE 0 END) AS estoque_baixo,
   981→                SUM(CASE WHEN qtd_atual > qtd_minima THEN 1 ELSE 0 END) AS estoque_ok
   982→            FROM estoque_interno
   983→        """
   984→        return self.query_one(sql)

  1036→    def atualizar_estoque(self, item_id, dados):
  1037→        """Atualiza dados de um item do estoque."""
  1038→        campos = []
  1039→        params = []
  1040→        for campo in ('descricao', 'unidade', 'qtd_minima'):
  1041→            if campo in dados:
  1042→                campos.append(f"{campo} = %s")
  1043→                params.append(dados[campo])
  1044→        if not campos:
  1045→            return False
  1046→        params.append(item_id)
  1047→        sql = f"UPDATE estoque_interno SET {', '.join(campos)}, updated_at = NOW() WHERE id = %s"
  1048→        return self.execute(sql, params)
