   519→function buscarProducaoPeriodo() {
   520→  var di = document.getElementById('filtroDtInicio').value;
   521→  var df = document.getElementById('filtroDtFim').value;
   522→  if (!di || !df) return;
   523→
   524→  document.getElementById('periodoLoader').classList.remove('hidden');
   525→  document.getElementById('periodoTabelaWrap').innerHTML = '';
   526→
   527→  fetch('/api/relatorios/producao_periodo?data_inicio=' + di + '&data_fim=' + df)
   528→    .then(function(r) { return r.json(); })
   529→    .then(function(data) {
   530→      document.getElementById('periodoLoader').classList.add('hidden');
   531→      if (data.error) {
   532→        document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-red-600 text-center py-3 text-sm">Erro: ' + data.error + '</p>';
   533→        return;
   534→      }
   535→      var kp = data.kpis;
   536→      document.getElementById('pKpiTotal').textContent = kp.total_lotes || 0;
   537→      document.getElementById('pKpiEmProducao').textContent = kp.total_em_producao || 0;
   538→      document.getElementById('pKpiFinalizados').textContent = kp.total_finalizados || 0;
   539→      document.getElementById('pKpiPecas').textContent = Math.round(Number(kp.total_pecas) || 0);
   540→      document.getElementById('pKpiAtrasados').textContent = kp.atrasados_agora || 0;
   541→      document.getElementById('pKpiTempo').textContent = Math.round(kp.tempo_medio_dias || 0) + 'd';
   542→
   543→      if (data.lotes.length === 0) {
   544→        document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-gray-400 text-center py-8">Nenhum lote encontrado neste periodo.</p>';
   545→        return;
   546→      }
   547→      var statusColors = {'em_producao':'bg-orange-100 text-orange-700','pausado':'bg-red-100 text-red-700','finalizado':'bg-green-100 text-green-700','liberado':'bg-blue-100 text-blue-700','importado':'bg-gray-100 text-gray-600'};
   548→      var statusLabels = {'em_producao':'Em Producao','pausado':'Pausado','finalizado':'Finalizado','liberado':'Liberado','importado':'Importado'};
   549→      var prioColors = {'urgente':'text-red-600 font-black','alta':'text-orange-600 font-bold','media':'text-yellow-600','baixa':'text-gray-400'};
   550→      var html = '<div class="overflow-x-auto"><table class="w-full text-sm"><thead class="text-xs uppercase tracking-wider text-blue-700 bg-blue-50 border-b border-blue-200"><tr>' +
   551→        '<th class="px-3 py-2.5 text-left">Lote</th><th class="px-3 py-2.5 text-left">Descricao</th><th class="px-3 py-2.5 text-center">Qtd</th><th class="px-3 py-2.5 text-center">Status</th><th class="px-3 py-2.5 text-left">Setor</th><th class="px-3 py-2.5 text-center">Previsao</th><th class="px-3 py-2.5 text-center">Dias</th></tr></thead><tbody class="divide-y divide-gray-50">';
   552→      for (var i = 0; i < data.lotes.length; i++) {
   553→        var l = data.lotes[i];
   554→        var sc = statusColors[l.status] || 'bg-gray-100 text-gray-600';
   555→        var sl = statusLabels[l.status] || l.status || '—';
   556→        var pc = prioColors[l.prioridade] || '';
   557→        var prevStr = (l.data_previsao_erp || '').substring(0, 10);
   558→        html += '<tr class="hover:bg-gray-50"><td class="px-3 py-2.5 font-bold text-blue-700">' + (l.lote_codigo || l.ordem || '') + '</td>' +
   559→          '<td class="px-3 py-2.5 text-gray-700 max-w-[200px] truncate">' + (l.descricao_produto || '—') + '</td>' +
   560→          '<td class="px-3 py-2.5 text-center font-bold">' + (Math.round(Number(l.qtde_ordem) || 0)) + '</td>' +
   561→          '<td class="px-3 py-2.5 text-center"><span class="inline-block px-2 py-0.5 rounded-full text-[10px] font-bold ' + sc + '">' + sl + '</span></td>' +
   562→          '<td class="px-3 py-2.5 text-gray-600 text-xs">' + (l.departamento_atual || '—') + '</td>' +
   563→          '<td class="px-3 py-2.5 text-center text-xs ' + pc + '">' + (prevStr || '—') + '</td>' +
   564→          '<td class="px-3 py-2.5 text-center font-bold">' + (l.dias_producao || '—') + '</td></tr>';
   565→      }
   566→      html += '</tbody></table></div>';
   567→      document.getElementById('periodoTabelaWrap').innerHTML = html;
   568→    })
   569→    .catch(function(e) {
   570→      document.getElementById('periodoLoader').classList.add('hidden');
   571→      document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-red-600 text-center py-3 text-sm">Erro de conexao</p>';
   572→    });
