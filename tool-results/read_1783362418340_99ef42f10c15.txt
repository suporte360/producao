   111→def _pg_conn():
   112→    import psycopg2
   113→    return psycopg2.connect(
   114→        host=PG_HOST, port=PG_PORT, dbname=PG_DB,

   119→def _pg_query(sql, params=None):
   120→    """Run query and return list of dicts."""
   121→    import psycopg2
   122→    from psycopg2.extras import RealDictCursor
   123→    conn = _pg_conn()
   124→    try:
   125→        with conn.cursor(cursor_factory=RealDictCursor) as cur:
   126→            cur.execute(sql, params or ())
   127→            rows = cur.fetchall()
   128→            return [dict(r) for r in rows]
   129→    finally:
   130→        conn.close()
   131→
   132→def _pg_query_one(sql, params=None):
   133→    """Run query and return one dict or None."""
   134→    import psycopg2
   135→    from psycopg2.extras import RealDictCursor
   136→    conn = _pg_conn()
   137→    try:
   138→        with conn.cursor(cursor_factory=RealDictCursor) as cur:
   139→            cur.execute(sql, params or ())
   140→            r = cur.fetchone()
   141→            return dict(r) if r else None
   142→    finally:
   143→        conn.close()

   147→def _int(val):
   148→    """Safely convert to int (handles Decimal, None, str)."""
   149→    if val is None:
   150→        return 0
   151→    return int(float(val))
   152→
   153→def _str(val):
   154→    """Safely convert to stripped string."""
   155→    if val is None:
   156→        return ''
   157→    return str(val).strip()
   158→
   159→def _extract_s_code(codigo_produto):
   160→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   161→    if not codigo_produto:
   162→        return ''
   163→    m = re.search(r'S-\d+', str(codigo_produto))
   164→    return m.group(0) if m else str(codigo_produto).strip()

   169→def _get_produto_nome(codigo_produto):
   170→    """Get real product name from in-memory cache. No MariaDB table needed."""
   171→    if not codigo_produto:

   173→    key = str(codigo_produto).strip()

   177→def _load_produto_cache():
   178→    """Load all product names from PostgreSQL into memory."""
   179→    global _produto_cache
   180→    try:
   181→        rows = _pg_query(
   182→            "SELECT produto, pronome FROM public.produto "
   183→            "WHERE pronome IS NOT NULL AND pronome != ''"
   184→        )
   185→        _produto_cache = {}
   186→        for r in rows:
   187→            codigo = r.get('produto', '')
   188→            nome = r.get('pronome', '')
   189→            if codigo:
   190→                _produto_cache[str(codigo).strip()] = str(nome).strip()[:200]
   191→        app.logger.info('Cache produtos carregado: %d produtos do ERP', len(_produto_cache))
   192→    except Exception as e:
   193→        app.logger.warning('Cache produtos falhou: %s', e)

   196→# ── ERP Sync: PostgreSQL -> MariaDB ─────────────────────────────────
   197→
   198→def _sync_erp_to_mariadb():
   199→    """
   200→    Sincroniza dados do ERP (PostgreSQL) para o MariaDB.
   201→    - Lotes abertos (EP, ES) do loteprod -> lotes_producao
   202→    - OFs de cada lote (ordem) -> ordens_fabricacao
   203→    - Processos de cada OF (respror + fases) -> operacoes_producao
   204→    - Lotes que nao existem mais no ERP sao marcados 'cancelado_erp'
   205→    """
   206→    t0 = time.time()
   207→    lotes_synced = 0
   208→    ofs_synced = 0
   209→    ops_synced = 0
   210→
   211→    try:
   212→        # 1. Buscar lotes abertos no ERP
   213→        lotes_erp = _pg_query("""
   214→            SELECT
   215→                lp.lotcod AS lote_codigo,
   216→                lp.lotdes AS lote_descricao,
   217→                lp.lotdtini AS data_abertura,
   218→                lp.lotdtpre AS data_previsao,
   219→                lp.lotstatus AS status_erp
   220→            FROM public.loteprod lp
   221→            WHERE lp.lotcod IS NOT NULL
   222→              AND lp.lotcod <> ''
   223→              AND (lp.lotstatus IS NULL OR lp.lotstatus IN ('EP', 'ES', ''))
   224→            ORDER BY lp.lotcod DESC
   225→        """)
   226→
   227→        # Coletar codigos dos lotes ativos no ERP
   228→        lotes_erp_codigos = set()
   229→        for lote_erp in lotes_erp:
   230→            lotcod = _str(lote_erp['lote_codigo'])
   231→            if not lotcod:
   232→                continue
   233→            lotes_erp_codigos.add(lotcod)
   234→
   235→            # Buscar OFs deste lote no ERP (tabela ordem)
   236→            ordens_erp = _pg_query("""
   237→                SELECT
   238→                    o.ordem,
   239→                    o.tipoord AS tipo_ordem,
   240→                    o.ordquanti AS quantidade,
   241→                    o.ordproduto AS produto,
   242→                    o.lotcod AS lote_codigo,
   243→                    o.ordnivprod AS nivel_producao
   244→                FROM public.ordem o
   245→                WHERE o.lotcod = %s
   246→                ORDER BY o.ordem
   247→            """, (lotcod,))
   248→
   249→            # Maior ordem numerica do lote (usado como 'ordem' no sistema local)
   250→            maior_ordem = 0
   251→            for o in ordens_erp:
   252→                ordem_val = o.get('ordem')
   253→                if ordem_val is not None:
   254→                    try:
   255→                        num = int(float(ordem_val))
   256→                        if num > maior_ordem:
   257→                            maior_ordem = num
   258→                    except (ValueError, TypeError):
   259→                        pass
   260→
   261→            # Produto final (OF pai via ordem3)
   262→            produto_final = ''
   263→            descricao_final = ''
   264→            qtd_final = 0
   265→            try:
   266→                pai = _pg_query_one("""
   267→                    SELECT
   268→                        o3.ordproof AS codigo_produto,
   269→                        p.pronome AS nome_produto,
   270→                        (SELECT o2.ordquanti FROM public.ordem o2
   271→                         WHERE o2.ordem = o3.ordem LIMIT 1) AS quantidade
   272→                    FROM public.ordem3 o3
   273→                    LEFT JOIN public.produto p ON o3.ordproof::TEXT = p.produto::TEXT
   274→                    WHERE o3.ordoflote = %s AND o3.ordem != 0
   275→                    LIMIT 1
   276→                """, (lotcod,))
   277→                if pai:
   278→                    produto_final = _str(pai.get('codigo_produto'))
   279→                    descricao_final = _str(pai.get('nome_produto'))
   280→                    qtd_final = _int(pai.get('quantidade'))
   281→            except Exception:
   282→                pass
   283→
   284→            if not produto_final and ordens_erp:
   285→                # Fallback: usar a primeira OF
   286→                produto_final = _str(ordens_erp[0].get('produto'))
   287→
   288→            # Upsert lote no MariaDB
   289→            lote_desc = _str(lote_erp['lote_descricao'])
   290→            if not lote_desc and descricao_final:
   291→                lote_desc = descricao_final
   292→            if not lote_desc:
   293→                lote_desc = produto_final
   294→
   295→            # Descobrir quais colunas existem (sem ALTER TABLE)
   296→            try:
   297→                db_execute("""
   298→                    INSERT INTO lotes_producao
   299→                        (ordem, lote_codigo, codigo_produto, descricao_produto, qtde_ordem)
   300→                    VALUES (%s, %s, %s, %s, %s)
   301→                    ON DUPLICATE KEY UPDATE
   302→                        descricao_produto = VALUES(descricao_produto),
   303→                        codigo_produto = VALUES(codigo_produto),
   304→                        qtde_ordem = VALUES(qtde_ordem)
   305→                """, (
   306→                    maior_ordem, lotcod, produto_final, lote_desc,
   307→                    qtd_final if qtd_final else 0,
   308→                ))
   309→                lotes_synced += 1
   310→            except Exception as e:
   311→                app.logger.warning('Sync lote %s falhou: %s', lotcod, e)
   312→
   313→            # Sincronizar OFs deste lote
   314→            for o in ordens_erp:
   315→                of_num = o.get('ordem')
   316→                if of_num is None:
   317→                    continue
   318→                of_num_str = str(int(float(of_num)))
   319→                cod_produto = _str(o.get('produto'))
   320→                nivel = _str(o.get('nivel_producao'))
   321→                tipo = 'pai' if nivel == '1' else 'filho'
   322→                qtd = _int(o.get('quantidade'))
   323→
   324→                # Descricao: usar nome do produto do cache se disponivel
   325→                of_desc = _get_produto_nome(cod_produto) or cod_produto
   326→
   327→                try:
   328→                    db_execute("""
   329→                        INSERT INTO ordens_fabricacao
   330→                            (lote_ordem, of_numero, codigo_produto, descricao_produto,
   331→                             qtde_ordem, unidade_medida, tipo, data_previsao)
   332→                        VALUES (%s, %s, %s, %s, %s, 'PC', %s, NULL)
   333→                        ON DUPLICATE KEY UPDATE
   334→                            codigo_produto = VALUES(codigo_produto),
   335→                            descricao_produto = VALUES(descricao_produto),
   336→                            qtde_ordem = VALUES(qtde_ordem),
   337→                            tipo = VALUES(tipo)
   338→                    """, (maior_ordem, of_num_str, cod_produto, of_desc, qtd, tipo))
   339→                    ofs_synced += 1
   340→                except Exception as e:
   341→                    app.logger.warning('Sync OF %s falhou: %s', of_num_str, e)
   342→
   343→                # Buscar processos (roteiro) desta OF
   344→                try:
   345→                    processos = _pg_query("""
   346→                        SELECT
   347→                            rpr.rprnumero AS numero_sequencial,
   348→                            rpr.rprordem AS ordem,
   349→                            rpr.rprproce AS processo,
   350→                            rpr.rproper AS operacao,
   351→                            rpr.rproped AS descricao_processo,
   352→                            rpr.rprfase AS fase,
   353→                            f.fasnome AS nome_departamento
   354→                        FROM public.respror rpr
   355→                        INNER JOIN public.fases f ON rpr.rprfase::TEXT = f.fase::TEXT
   356→                        WHERE rpr.rprordem::TEXT = %s::TEXT
   357→                        ORDER BY rpr.rprproce
   358→                    """, (of_num_str,))
   359→                except Exception:
   360→                    # Fallback sem JOIN com fases
   361→                    try:
   362→                        processos = _pg_query("""
   363→                            SELECT
   364→                                rpr.rprnumero AS numero_sequencial,
   365→                                rpr.rprordem AS ordem,
   366→                                rpr.rprproce AS processo,
   367→                                rpr.rproper AS operacao,
   368→                                rpr.rproped AS descricao_processo,
   369→                                rpr.rprfase AS fase,
   370→                                NULL AS nome_departamento
   371→                            FROM public.respror rpr
   372→                            WHERE rpr.rprordem::TEXT = %s::TEXT
   373→                            ORDER BY rpr.rprproce
   374→                        """, (of_num_str,))
   375→                    except Exception as e2:
   376→                        app.logger.warning('Sync processos OF %s falhou: %s', of_num_str, e2)
   377→                        processos = []
   378→
   379→                for proc in processos:
   380→                    dept_original = (_str(proc.get('nome_departamento'))).upper().strip()
   381→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   382→                    if not dept:
   383→                        continue
   384→                    desc_op = _str(proc.get('descricao_processo')) or _str(proc.get('operacao'))
   385→
   386→                    try:
   387→                        db_execute("""
   388→                            INSERT INTO operacoes_producao
   389→                                (of_numero, lote_ordem, sequencia, fase,
   390→                                 descricao_operacao, departamento, codigo_barras)
   391→                            VALUES (%s, %s, %s, %s, %s, %s, NULL)
   392→                            ON DUPLICATE KEY UPDATE
   393→                                descricao_operacao = VALUES(descricao_operacao),
   394→                                departamento = VALUES(departamento),
   395→                                fase = VALUES(fase)
   396→                        """, (
   397→                            of_num_str, maior_ordem,
   398→                            _int(proc.get('numero_sequencial')),
   399→                            _str(proc.get('fase')),
   400→                            desc_op, dept
   401→                        ))
   402→                        ops_synced += 1
   403→                    except Exception as e:
   404→                        app.logger.warning('Sync op OF %s falhou: %s', of_num_str, e)
   405→
   406→        # 2. Salvar lista de lotes ativos no ERP (para filtrar na API)
   407→        global _lotes_erp_ativos
   408→        _lotes_erp_ativos = lotes_erp_codigos
   409→
   410→        # 3. Reload product names cache
   411→        _load_produto_cache()
   412→
   413→        elapsed = time.time() - t0
   414→        app.logger.info(
   415→            'Sync ERP completo em %.1fs: %d lotes, %d OFs, %d ops',
   416→            elapsed, lotes_synced, ofs_synced, ops_synced
   417→        )
   418→
   419→    except Exception as e:
   420→        app.logger.error('Sync ERP falhou: %s', e)
   421→
   422→
   423→def _sync_loop():
   424→    """Background thread: sync ERP a cada SYNC_INTERVAL segundos."""
   425→    while True:
   426→        try:
   427→            _sync_erp_to_mariadb()
   428→        except Exception as e:
   429→            app.logger.error('Sync loop erro: %s', e)
   430→        time.sleep(SYNC_INTERVAL)
   431→

   433→# ── Init DB ──────────────────────────────────────────────────────────
   434→
   435→def init_db():
   436→    """Create serralheria tables if not exist, seed serra1-9, sync ERP."""
   437→    db_execute("""
   438→        CREATE TABLE IF NOT EXISTS serralheria_usuarios (
   439→            id INT AUTO_INCREMENT PRIMARY KEY,
   440→            nome VARCHAR(100) NOT NULL,
   441→            setor VARCHAR(100) DEFAULT 'Serralheria',
   442→            ativo TINYINT(1) DEFAULT 1
   443→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
   444→    """)
   445→    db_execute("""
   446→        CREATE TABLE IF NOT EXISTS serralheria_producao (
   447→            id INT AUTO_INCREMENT PRIMARY KEY,
   448→            ordem_id INT DEFAULT NULL,
   449→            lote VARCHAR(50) NOT NULL,
   450→            produto VARCHAR(50) NOT NULL,
   451→            of_numero INT DEFAULT NULL,
   452→            usuario_id INT DEFAULT NULL,
   453→            usuario_nome VARCHAR(100) DEFAULT NULL,
   454→            setor VARCHAR(50) DEFAULT '',
   455→            status VARCHAR(20) DEFAULT 'em_producao',
   456→            data_inicio DATETIME DEFAULT NULL,
   457→            data_fim DATETIME DEFAULT NULL,
   458→            INDEX idx_sp_lote (lote),
   459→            INDEX idx_sp_status (status)
   460→        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
   461→    """)
   462→    # Seed serra1-9
   463→    for i in range(1, 10):
   464→        nome = 'serra' + str(i)
   465→        existing = db_query_one("SELECT id FROM serralheria_usuarios WHERE nome = %s", (nome,))
   466→        if not existing:
   467→            db_execute(
   468→                "INSERT INTO serralheria_usuarios (nome, setor, ativo) VALUES (%s, 'Serralheria', 1)",
   469→                (nome,)
   470→            )
   471→    # Migration: add missing columns if table already existed
   472→    for col, col_def in [
   473→        ('setor', "VARCHAR(50) DEFAULT ''"),
   474→        ('of_numero', 'INT DEFAULT NULL'),
   475→        ('ordem_id', 'INT DEFAULT NULL'),
   476→    ]:
   477→        try:
   478→            db_query_one('SELECT ' + col + ' FROM serralheria_producao LIMIT 1')
   479→        except Exception:
   480→            db_execute('ALTER TABLE serralheria_producao ADD COLUMN ' + col + ' ' + col_def)
   481→
   482→    # NOTE: Nao fazemos ALTER TABLE em lotes_producao (disco cheio).
   483→    # O sync usa apenas os valores que o ENUM de 'status' ja aceita.
   484→    # Se faltar alguma coluna, o sync vai falhar silenciosamente por lote/OF.
   485→
   486→    # Initial ERP sync (loads lotes, OFs, operacoes, and product names)
   487→    print('[SYNC] Sincronizando com ERP...')
   488→    _sync_erp_to_mariadb()
   489→
   490→    # Start background sync thread
   491→    t = threading.Thread(target=_sync_loop, daemon=True)
   492→    t.start()
   493→    print('[SYNC] Thread de sync ativa (a cada %d segundos)' % SYNC_INTERVAL)
   494→
   495→    print('[OK] Tabelas serralheria verificadas no MariaDB')
   496→

   506→def api_kpis():
   507→    """KPIs da serralheria: lotes ativos, pecas, total qtd, producoes."""
   508→    try:
   509→        total_lotes = _int(db_query_one(
   510→            'SELECT COUNT(*) as t FROM lotes_producao WHERE status != %s', ('cancelado_erp',)
   511→        )['t'])
   512→
   513→        total_pecas = _int(db_query_one("""
   514→            SELECT COUNT(*) as t
   515→            FROM ordens_fabricacao of2
   516→            INNER JOIN lotes_producao lp ON of2.lote_ordem = lp.ordem
   517→            WHERE of2.tipo = 'filho' AND lp.status != %s
   518→        """, ('cancelado_erp',))['t'])
   519→
   520→        total_qtd = _int(db_query_one("""
   521→            SELECT COALESCE(SUM(qtde_ordem), 0) as t
   522→            FROM lotes_producao WHERE status != %s
   523→        """, ('cancelado_erp',))['t'])
   524→
   525→        em_andamento = _int(db_query_one(
   526→            "SELECT COUNT(*) as t FROM serralheria_producao WHERE status = 'em_producao'"
   527→        )['t'])
   528→

   766→@app.route('/api/lotes/<lotcod>/pecas')
   767→def api_pecas(lotcod):
   768→    """Pecas (OFs filhas) de um lote com operacoes por setor."""
   769→    try:
   770→        # Lote info
   771→        lote = db_query_one(
   772→            "SELECT ordem, lote_codigo, descricao_produto, qtde_ordem "
   773→            "FROM lotes_producao WHERE lote_codigo = %s",
   774→            (lotcod,)
   775→        )
   776→        if not lote:
   777→            return jsonify({'error': 'Lote nao encontrado'}), 404
   778→
   779→        # Child OFs (pecas para fabricar - PC-/CJ-)
   780→        pecas = db_query("""
   781→            SELECT of.of_numero, of.codigo_produto as codigo, of.descricao_produto as descricao,
   782→                   of.qtde_ordem as quantidade, of.tipo
   783→            FROM ordens_fabricacao of
   784→            WHERE of.lote_ordem = %s AND of.tipo = 'filho'
   785→            ORDER BY of.codigo_produto
   786→        """, (lote['ordem'],))
   787→
   788→        # Parent OF (produto final S-)
   789→        pai = db_query_one("""
   790→            SELECT codigo_produto, descricao_produto, qtde_ordem
   791→            FROM ordens_fabricacao WHERE lote_ordem = %s AND tipo = 'pai' LIMIT 1
   792→        """, (lote['ordem'],))
   793→
   794→        produto_final = None
   795→        if pai:
   796→            pai_nome = _get_produto_nome(pai.get('codigo_produto'))
   797→            produto_final = {
   798→                'codigo': _str(pai['codigo_produto']),
   799→                'descricao': pai_nome or _str(pai['descricao_produto']),
   800→                'quantidade': _int(pai['qtde_ordem']),
   801→            }
   802→
   803→        # Build pecas with operations info
   804→        pecas_result = []
   805→        for p in pecas:
   806→            ops = db_query("""
   807→                SELECT id, departamento, descricao_operacao, status, sequencia
   808→                FROM operacoes_producao
   809→                WHERE of_numero = %s
   810→                ORDER BY sequencia
   811→            """, (p['of_numero'],))
   812→
   813→            # Unique departments (setores) for this OF
   814→            setores = list(set(
   815→                op['departamento'] for op in ops if op.get('departamento')
   816→            ))
   817→
   818→            # Use real product name from ERP cache instead of "PRODUCAO DE S-XXX"
   819→            nome_real = _get_produto_nome(p.get('codigo'))
   820→            produto_nome = nome_real if nome_real else _str(p['descricao'])
   821→
   822→            pecas_result.append({
   823→                'of_numero': p['of_numero'],
   824→                'ordem': p['of_numero'],
   825→                'codigo': _str(p['codigo']),
   826→                'quantidade': _int(p['quantidade']),
   827→                'produto_nome': produto_nome,
   828→                'setores': setores,
   829→                'operacoes': ops,
   830→            })
   831→
   832→        # Active productions in serralheria_producao
   833→        ativas_raw = db_query(
   834→            "SELECT produto, of_numero, setor, usuario_nome "
   835→            "FROM serralheria_producao WHERE lote = %s AND status = 'em_producao'",
   836→            (lotcod,)
   837→        )
   838→        pecas_ativas = {}
   839→        for row in ativas_raw:
   840→            pecas_ativas[row['produto']] = {
   841→                'setor': row['setor'],
   842→                'usuario': row['usuario_nome'],
   843→            }
   844→
   845→        return jsonify({
   846→            'lote': lotcod,
   847→            'descricao': _str(lote['descricao_produto']),
   848→            'produto_final': produto_final,
   849→            'pecas': pecas_result,
   850→            'pecas_ativas': pecas_ativas,
   851→        })
   852→    except Exception as e:
   853→        app.logger.error('Erro pecas: %s', e)
   854→        return jsonify({'error': str(e)}), 500
