   656→function _bindPecaClicks(){
   657→  var container=document.getElementById('pecasContent');
   658→  if(!container)return;
   659→  container.onclick=function(e){
   660→    var card=_findCard(e.target);
   661→    if(!card){_debugToast('CLIQUE FORA DO CARD');return;}
   662→    var codigo=card.getAttribute('data-codigo');
   663→    if(!codigo){_debugToast('CODIGO VAZIO: '+codigo);return;}
   664→    /* Usa estado global em vez de data-action para evitar condicao de corrida */
   665→    var isActive=currentPecasAtivas[codigo];
   666→    if(finalizarMode){
   667→      if(isActive)toggleFinalizarPeca(codigo);
   668→    }else if(isActive){
   669→      return;
   670→    }else if(selectedSetor){
   671→      /* Se setor selecionado mas nenhuma peca marcada, abrir popup para esta peca */
   672→      if(selectedPecas.length===0){
   673→        selectedSetor=null;
   674→        try{var setores=JSON.parse(decodeURIComponent(card.getAttribute('data-setores')||'[]'));}catch(er){setores=[];}
   675→        abrirSetor(codigo,setores);
   676→      }else{
   677→        togglePecaSelection(codigo);
   678→      }
   679→    }else{
   680→      try{var setores=JSON.parse(decodeURIComponent(card.getAttribute('data-setores')||'[]'));}catch(er){setores=[];}
   681→      abrirSetor(codigo,setores);
