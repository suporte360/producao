   735→      if(ativasCheck&&ativasCheck.length>0)toggleFinalizarPeca(codigo);
   736→      return;
   737→    }

   739→    if(selectedSetor){selectedSetor=null;selectedPecas=[];}
   740→    try{var setores=JSON.parse(decodeURIComponent(card.getAttribute('data-setores')||'[]'));}catch(er){setores=[];}
   741→    try{var ativos=JSON.parse(decodeURIComponent(card.getAttribute('data-setores-ativos')||'[]'));}catch(er){ativos=[];}
   742→    abrirSetor(codigo,setores,ativos);
   743→  };
