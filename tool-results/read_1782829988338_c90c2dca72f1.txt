   403→   403→   403→            # Verifica se já existe card para esta operação

   436→   436→   436→    def finalizar_operacao(self, operacao_id, usuario_id, qtde_produzida=None, qtde_refugada=0, observacoes=None, tipo='total'):
   437→   437→   437→        """
   438→   438→   438→        tipo='total'   → marca como concluido e fecha o lote se todas as ops concluídas
   439→   439→   439→        tipo='parcial' → registra a quantidade produzida até agora e reinicia a operação
   440→   440→   440→                         (status volta para 'pendente' com qtde_produzida acumulada)
   441→   441→   441→                         Assim o próximo setor já pode ver as peças disponíveis.
   442→   442→   442→        """
   443→   443→   443→        op = self.query_one("SELECT * FROM operacoes_producao WHERE id = %s", (operacao_id,))
   444→   444→   444→        if not op:
   445→   445→   445→            return False
   446→   446→   446→
   447→   447→   447→        if tipo == 'parcial':
   448→   448→   448→            # Acumula a quantidade e volta para pendente (disponível para próximo setor)
   449→   449→   449→            self.execute("""
   450→   450→   450→                UPDATE operacoes_producao
   451→   451→   451→                SET status = 'pendente',
   452→   452→   452→                    data_inicio = NULL,
   453→   453→   453→                    qtde_produzida = COALESCE(qtde_produzida, 0) + COALESCE(%s, 0),
   454→   454→   454→                    qtde_refugada  = COALESCE(qtde_refugada, 0)  + COALESCE(%s, 0),
   455→   455→   455→                    observacoes    = COALESCE(%s, observacoes)
   456→   456→   456→                WHERE id = %s
   457→   457→   457→            """, (qtde_produzida, qtde_refugada, observacoes, operacao_id))
   458→   458→   458→            # Atualiza status do lote para 'importado' (disponível) se estava em_producao
   459→   459→   459→            self.execute("""
   460→   460→   460→                UPDATE lotes_producao SET status = 'importado'
   461→   461→   461→                WHERE ordem = %s AND status = 'em_producao'
   462→   462→   462→            """, (op['lote_ordem'],))
   463→   463→   463→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'parcial', qtde_produzida=qtde_produzida)
   464→   464→   464→            # Sincroniza kanban: card volta para 'aguardando' (próximo setor pega)
   465→   465→   465→            self._sincronizar_kanban_operacao(operacao_id, 'aguardando')
   466→   466→   466→        else:
   467→   467→   467→            self.execute("""
   468→   468→   468→                UPDATE operacoes_producao
   469→   469→   469→                SET status = 'concluido', data_fim = NOW(), usuario_fim_id = %s,
   470→   470→   470→                    qtde_produzida = COALESCE(%s, qtde_produzida),
   471→   471→   471→                    qtde_refugada = COALESCE(%s, 0),
   472→   472→   472→                    observacoes = COALESCE(%s, observacoes)
   473→   473→   473→                WHERE id = %s
   474→   474→   474→            """, (usuario_id, qtde_produzida, qtde_refugada, observacoes, operacao_id))
   475→   475→   475→            self.registrar_apontamento_tempo(op['lote_ordem'], op.get('of_numero'), operacao_id, op['departamento'], usuario_id, 'fim', qtde_produzida=qtde_produzida)
   476→   476→   476→            # Sincroniza kanban: card vai para 'concluido'
   477→   477→   477→            self._sincronizar_kanban_operacao(operacao_id, 'concluido', usuario_id)
   478→   478→   478→            # Verifica se todas as operações do lote foram concluídas
   479→   479→   479→            self._verificar_conclusao_lote(op['lote_ordem'])
   480→   480→   480→        return True

   862→   862→   862→                        AND status NOT IN ('finalizado','cancelado')) AS atrasados

   876→   876→   876→                    SUM(op.status = 'pausado')      AS pausados,
