    59→@app.template_filter('formatar_tempo')
    60→def formatar_tempo(segundos):
    61→    """Formata segundos em HH:MM:SS"""
    62→    try:
    63→        seg = int(segundos or 0)
    64→        h = seg // 3600
    65→        m = (seg % 3600) // 60
    66→        s = seg % 60
    67→        return f'{h:02d}:{m:02d}:{s:02d}'
    68→    except Exception:
    69→        return '00:00:00'

    71→@app.template_filter('formatar_data_br')
    72→def formatar_data_br(value):
    73→    """Formata datetime/date para dd/mm/yyyy HH:MM"""
    74→    if not value:
    75→        return '-'
    76→    try:
    77→        if hasattr(value, 'strftime'):
    78→            if hasattr(value, 'hour'):
    79→                return value.strftime('%d/%m/%Y %H:%M')
    80→            return value.strftime('%d/%m/%Y')
    81→        return str(value)
    82→    except Exception:
    83→        return str(value)

    85→# =============================================
    86→# Proteção contra força bruta no login
    87→# Armazena em memória: {ip: {'tentativas': N, 'bloqueado_ate': timestamp}}
    88→# =============================================
    89→
    90→_login_attempts = defaultdict(lambda: {'tentativas': 0, 'bloqueado_ate': 0})
    91→MAX_TENTATIVAS  = 5
    92→BLOQUEIO_SEGUNDOS = 600  # 10 minutos
    93→
    94→def _get_ip():
    95→    return request.headers.get('X-Forwarded-For', request.remote_addr).split(',')[0].strip()
    96→
    97→def _checar_bloqueio(ip):
    98→    dados = _login_attempts[ip]
    99→    if time.time() < dados['bloqueado_ate']:
   100→        restante = int(dados['bloqueado_ate'] - time.time())
   101→        return True, restante
   102→    return False, 0
   103→
   104→def _registrar_falha(ip):
   105→    dados = _login_attempts[ip]
   106→    dados['tentativas'] += 1
   107→    if dados['tentativas'] >= MAX_TENTATIVAS:
   108→        dados['bloqueado_ate'] = time.time() + BLOQUEIO_SEGUNDOS
   109→        dados['tentativas'] = 0
   110→
   111→def _limpar_tentativas(ip):
   112→    _login_attempts[ip] = {'tentativas': 0, 'bloqueado_ate': 0}

   118→def login_required(f):
   119→    @wraps(f)
   120→    def decorated(*args, **kwargs):
   121→        if 'usuario_id' not in session:
   122→            flash('Faça login para continuar.', 'warning')
   123→            return redirect(url_for('login'))
   124→        return f(*args, **kwargs)
   125→    return decorated
   126→
   127→def role_required(*roles):
   128→    def decorator(f):
   129→        @wraps(f)
   130→        def decorated(*args, **kwargs):
   131→            if session.get('usuario_role') not in roles:
   132→                flash('Acesso não autorizado.', 'error')
   133→                return redirect(url_for('dashboard'))
   134→            return f(*args, **kwargs)
   135→        return decorated
   136→    return decorator

   142→@app.route('/login', methods=['GET', 'POST'])
   143→def login():
   144→    if 'usuario_id' in session:
   145→        return redirect(url_for('dashboard'))
   146→
   147→    if request.method == 'POST':
   148→        ip = _get_ip()
   149→
   150→        bloqueado, restante = _checar_bloqueio(ip)
   151→        if bloqueado:
   152→            mins = restante // 60
   153→            secs = restante % 60
   154→            flash(f'Muitas tentativas. Tente novamente em {mins}m {secs}s.', 'error')
   155→            return render_template('login.html')
   156→
   157→        usuario = request.form.get('usuario', '').strip()
   158→        senha   = request.form.get('senha', '')
   159→
   160→        if not usuario or not senha:
   161→            flash('Informe usuário e senha.', 'error')
   162→            return render_template('login.html')
   163→
   164→        user = db.verificar_usuario(usuario, senha)
   165→
   166→        if user:
   167→            _limpar_tentativas(ip)
   168→            session.permanent = True
   169→            session['usuario_id']          = user['id']
   170→            session['usuario_nome']        = user['nome']
   171→            session['usuario_role']        = user['role']
   172→            session['usuario_departamento']= user.get('nome_departamento', '')
   173→            flash(f'Bem-vindo, {user["nome"]}!', 'success')
   174→            return redirect(url_for('dashboard'))
   175→        else:
   176→            _registrar_falha(ip)
   177→            dados   = _login_attempts[ip]
   178→            restantes = MAX_TENTATIVAS - dados['tentativas']
   179→            if restantes > 0:
   180→                flash(f'Usuário ou senha inválidos. {restantes} tentativa(s) restante(s).', 'error')
   181→            else:
   182→                flash('Conta bloqueada por 10 minutos devido a múltiplas tentativas.', 'error')
   183→
   184→    return render_template('login.html')

   318→@app.route('/pcp/importar')
   319→@login_required
   320→@role_required('admin', 'gerente', 'pcp')
   321→def pcp_importar_erp():
   322→    pg_ok   = False
   323→    pg_erro = ''
   324→    lotes_erp = []
   325→    lotes_ja_importados = set()
   326→
   327→    try:
   328→        pg = DatabasePostgreSQL()
   329→        # Busca lotes do ERP com produto final e maior ordem
   330→        lotes_raw = pg.get_lotes_agrupados_abertos()
   331→        # Para cada lote busca o produto final
   332→        for lote in lotes_raw:
   333→            prod = pg.get_produto_pronto_por_lote(lote['lote_codigo'])
   334→            lote['nome_produto']        = prod['nome_produto']   if prod else None
   335→            lote['produto_final_codigo']= prod['codigo_produto'] if prod else lote.get('produto_final_codigo')
   336→            lote['maior_ordem']         = prod['quantidade']     if prod else None
   337→            # busca a maior ordem numerica do lote
   338→            ordens = pg.get_ordens_por_lote(lote['lote_codigo'])
   339→            if ordens:
   340→                lote['maior_ordem'] = max(int(o['ordem']) for o in ordens if o.get('ordem') and str(o['ordem']).isdigit())
   341→        lotes_erp = lotes_raw
   342→        pg_ok = True
   343→    except Exception as e:
   344→        pg_erro = str(e)
   345→
   346→    # Lotes já importados no MariaDB
   347→    ja = db.query("SELECT DISTINCT lote_codigo FROM lotes_producao WHERE lote_codigo IS NOT NULL")
   348→    lotes_ja_importados = {r['lote_codigo'] for r in ja}
   349→
   350→    return render_template('pcp/importar_erp.html',
   351→                           pg_ok=pg_ok, pg_erro=pg_erro,
   352→                           lotes_erp=lotes_erp,
   353→                           lotes_ja_importados=lotes_ja_importados,
   354→                           now=datetime.now(),
   355→                           usuario_nome=session['usuario_nome'])

   358→@app.route('/pcp/importar_erp', methods=['POST'])
   359→@login_required
   360→@role_required('admin', 'gerente', 'pcp')
   361→def pcp_executar_importacao():
   362→    data  = request.json or {}
   363→    lotes = data.get('lotes', [])
   364→    if not lotes:
   365→        return jsonify({'status':'error','message':'Nenhum lote informado'}), 400
   366→
   367→    importados = 0
   368→    atualizados = 0
   369→    erros = []
   370→
   371→    try:
   372→        pg = DatabasePostgreSQL()
   373→    except Exception as e:
   374→        return jsonify({'status':'error','message': f'Erro PostgreSQL: {e}'}), 500
   375→
   376→    for item in lotes:
   377→        try:
   378→            lote_cod = str(item.get('lote_codigo',''))
   379→            ordem_num = item.get('ordem')
   380→            if not lote_cod:
   381→                continue
   382→
   383→            # Verifica se já existe
   384→            existente = db.query_one(
   385→                "SELECT ordem FROM lotes_producao WHERE lote_codigo=%s", (lote_cod,))
   386→            ja_existia = existente is not None
   387→
   388→            # Importa lote principal
   389→            db.importar_lote_erp({
   390→                'ordem':             int(ordem_num) if str(ordem_num).isdigit() else 0,
   391→                'lote_codigo':       lote_cod,
   392→                'codigo_produto':    item.get('produto',''),
   393→                'descricao_produto': item.get('produto',''),  # agora envia lote_descricao do template
   394→                'qtde_ordem':        float(item.get('qtd', 0) or 0),
   395→                'unidade_medida':    'PC',
   396→                'status_erp':        'A',
   397→                'data_previsao_erp': item.get('previsao') or None,
   398→                'data_abertura_erp': item.get('abertura') or None,
   399→                'planejador':        session.get('usuario_nome'),
   400→            })
   401→
   402→            lote_ordem_local = int(ordem_num) if str(ordem_num).isdigit() else 0
   403→
   404→            # Busca ordens do lote no ERP e importa OFs
   405→            ordens_erp = pg.get_ordens_por_lote(lote_cod)
   406→            for of in ordens_erp:
   407→                of_num = of.get('ordem')
   408→                if not of_num:
   409→                    continue
   410→                db.importar_of({
   411→                    'lote_ordem':       lote_ordem_local,
   412→                    'of_numero':        str(of_num),
   413→                    'codigo_produto':   of.get('produto',''),
   414→                    'descricao_produto':item.get('produto',''),
   415→                    'qtde_ordem':       float(of.get('quantidade') or 0),
   416→                    'unidade_medida':   'PC',
   417→                    'tipo':             'filho' if of.get('nivel_producao','') != '1' else 'pai',
   418→                    'data_previsao':    of.get('data_previsao') or None,
   419→                })
   420→
   421→                # Busca processos (operações) de cada OF
   422→                processos = pg.get_processos_ordem(str(of_num))
   423→                for proc in processos:
   424→                    dept_original = (proc.get('nome_departamento') or '').upper().strip()
   425→                    if not dept_original:
   426→                        continue
   427→                    # Mapeia nomes do ERP (fases) para nomes do sistema local
   428→                    dept = MAPEAMENTO_SETORES.get(dept_original, dept_original)
   429→                    db.importar_operacao({
   430→                        'of_numero':          str(of_num),
   431→                        'lote_ordem':         lote_ordem_local,
   432→                        'sequencia':          proc.get('numero_sequencial', 0),
   433→                        'fase':               proc.get('fase',''),
   434→                        'descricao_operacao': proc.get('descricao_processo') or proc.get('operacao',''),
   435→                        'departamento':       dept,
   436→                        'codigo_barras':      None,
   437→                    })
   438→
   439→                # Sincroniza kanban
   440→                try:
   441→                    db.sincronizar_kanban_com_operacoes(lote_ordem_local)
   442→                except Exception:
   443→                    pass
   444→
   445→            db.add_log(session['usuario_id'], 'importar_erp',
   446→                       f'Lote {lote_cod} importado do ERP', 'lotes_producao', lote_ordem_local)
   447→
   448→            if ja_existia:
   449→                atualizados += 1
   450→            else:
   451→                importados += 1
   452→
   453→        except Exception as e:
   454→            erros.append(f"{item.get('lote_codigo')}: {e}")
   455→
   456→    return jsonify({
   457→        'status':     'success',
   458→        'importados':  importados,
   459→        'atualizados': atualizados,
   460→        'erros':       erros,
   461→    })

   711→# =============================================
   712→# APIs — ALMOXARIFADO (separação de OPs)

   715→def _baixar_estoque_por_of(ordem):
   716→    """Busca materiais da OF no ERP e dá baixa automática no estoque interno."""
   717→    try:
   718→        pg = DatabasePostgreSQL()
   719→        materiais = pg.get_requisicoes_ordem(ordem)
   720→        if not materiais:
   721→            return
   722→        for mat in materiais:
   723→            codigo = str(mat.get('produto') or mat.get('reqproduto') or '')
   724→            if not codigo:
   725→                continue
   726→            qtd = float(mat.get('quantidade') or mat.get('rqoquanti') or 0)
   727→            if qtd <= 0:
   728→                continue
   729→            # Verifica se o material existe no estoque interno
   730→            item = db.get_estoque_por_codigo(codigo)
   731→            if item and item['qtd_atual'] > 0:
   732→                db.registrar_baixa_estoque(codigo, qtd, ordem, 'liberacao_of')
   733→                print(f"[ESTOQUE] Baixa automática: {codigo} -{qtd} (OP #{ordem})")
   734→    except Exception as e:
   735→        print(f"[ESTOQUE] Erro ao baixar estoque da OP #{ordem}: {e}")
