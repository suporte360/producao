   109→def _int(val):
   110→    """Safely convert to int (handles Decimal, None, str)."""
   111→    if val is None:
   112→        return 0
   113→    return int(float(val))
   114→
   115→def _str(val):
   116→    """Safely convert to stripped string."""
   117→    if val is None:
   118→        return ''
   119→    return str(val).strip()
   120→
   121→def _is_sob_medida(codigo):
   122→    """Verifica se o produto codigo termina com -X (sob medida)."""
   123→    if not codigo:
   124→        return False
   125→    c = str(codigo).strip().upper()
   126→    return c.endswith('-X')
   127→
   128→def _extract_s_code(codigo_produto):
   129→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   130→    if not codigo_produto:
   131→        return ''
   132→    m = re.search(r'S-\d+', str(codigo_produto))
   133→    return m.group(0) if m else str(codigo_produto).strip()

   138→def _get_produto_nome(codigo_produto):
   139→    """Get real product name from in-memory cache (PostgreSQL produto.pronome)."""
   140→    if not codigo_produto:
   141→        return ''
   142→    key = str(codigo_produto).strip()
   143→    return _produto_cache.get(key, '')

   146→def _pg_query_one(sql, params=None):
   147→    """Query PostgreSQL and return one row as dict."""
   148→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   149→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   150→    conn.autocommit = True
   151→    try:
   152→        with conn.cursor() as cur:
   153→            cur.execute(sql, params or ())
   154→            cols = [d[0] for d in cur.description] if cur.description else []
   155→            row = cur.fetchone()
   156→            if row:
   157→                return dict(zip(cols, row))
   158→            return None
   159→    finally:
   160→        conn.close()

   163→def _load_produto_cache():
   164→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   165→    global _produto_cache
   166→    _produto_cache = {}
   167→    # 1) PostgreSQL: tabela produto (pronome real)
   168→    try:
   169→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   170→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   171→        conn.autocommit = True
   172→        try:
   173→            with conn.cursor() as cur:
   174→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   175→                cols = [d[0] for d in cur.description]
   176→                for row in cur.fetchall():
   177→                    r = dict(zip(cols, row))
   178→                    cod = str(r.get('produto', '')).strip()
   179→                    nome = str(r.get('pronome', '')).strip()
   180→                    if cod and nome:
   181→                        _produto_cache[cod] = nome[:200]
   182→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   183→        finally:
   184→            conn.close()
   185→    except Exception as e:
   186→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   187→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   188→    try:
   189→        rows = db_query(
   190→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   191→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   192→        )
   193→        added = 0
   194→        for r in rows:
   195→            cod = str(r.get('codigo_produto', '')).strip()
   196→            nome = str(r.get('descricao_produto', '')).strip()
   197→            if cod and cod not in _produto_cache and nome:
   198→                _produto_cache[cod] = nome[:200]
   199→                added += 1
   200→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   201→    except Exception as e:
   202→        print('[WARN] Cache MariaDB falhou: %s' % e)
   203→

   207→def _refresh_lotes_ativos():
   208→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""
   209→    global _lotes_ativos
   210→    try:
   211→        rows = db_query(
   212→            "SELECT DISTINCT lote_codigo FROM lotes_producao "
   213→            "WHERE status IN ('liberado','em_producao','pausado')"
   214→        )
   215→        _lotes_ativos = set()
   216→        for r in rows:
   217→            if r.get('lote_codigo'):
   218→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   219→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   220→    except Exception as e:
   221→        print('[WARN] Lotes ativos falhou: %s' % e)
   222→
   223→
   224→def _build_lotes_filter():
   225→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   226→    if _lotes_ativos:
   227→        ph = ','.join(['%s'] * len(_lotes_ativos))
   228→        where = 'WHERE lote_codigo IN (' + ph + ')'
   229→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   230→        params = tuple(_lotes_ativos)
   231→    else:
   232→        where = 'WHERE 1=0'
   233→        join = ' AND 1=0'
   234→        params = ()
   235→    return where, join, params

   783→def _atualizar_departamento_lote(lote_ordem):
   784→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   785→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   786→    if not lote_ordem:
   787→        return
   788→    # Total de departamentos unicos do lote
   789→    total = db_query_one(
   790→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   791→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   792→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   793→        (lote_ordem,)
   794→    )
   795→    total_setores = int(total['t'] or 0) if total else 0
   796→
   797→    # Primeira operacao pendente (nao concluida) por sequencia
   798→    primeira = db_query_one(
   799→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   800→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   801→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   802→        "ORDER BY op.sequencia ASC LIMIT 1",
   803→        (lote_ordem,)
   804→    )
   805→    dept = primeira['departamento'] if primeira else None
   806→    seq = int(primeira['sequencia']) if primeira else 0
   807→
   808→    db_execute(
   809→        "UPDATE lotes_producao "
   810→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   811→        "WHERE ordem = %s",
   812→        (dept, seq, total_setores, lote_ordem)
   813→    )

   816→def _sync_lote_status(of_numero, novo_status):
   817→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   818→    if not of_numero:
   819→        return
   820→    of_row = db_query_one(
   821→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",
   822→        (of_numero,)
   823→    )
   824→    if not of_row or not of_row.get('lote_ordem'):
   825→        return
   826→    lote_ordem = of_row['lote_ordem']
   827→
   828→    if novo_status == 'em_andamento':
   829→        # Muda status do lote para em_producao se ainda estiver liberado
   830→        db_execute(
   831→            "UPDATE lotes_producao SET status = 'em_producao' "
   832→            "WHERE ordem = %s AND status = 'liberado'",
   833→            (lote_ordem,)
   834→        )
   835→    elif novo_status == 'concluido':
   836→        # Verifica se TODAS as ops do lote estao concluidas
   837→        total_pend = db_query_one(
   838→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   839→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   840→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   841→            (lote_ordem,)
   842→        )
   843→        if total_pend and int(total_pend['t']) == 0:
   844→            db_execute(
   845→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   846→                (lote_ordem,)
   847→            )

   850→def _sync_op_status(of_numero, departamento, novo_status):
   851→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   852→    if not of_numero or not departamento:
   853→        return
   854→    try:
   855→        # 1. Atualiza operacoes_producao
   856→        db_execute(
   857→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   858→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   859→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   860→            (novo_status, novo_status, of_numero, departamento)
   861→        )
   862→        # 2. Atualiza kanban_cards
   863→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   864→        nova_etapa = etapa_map.get(novo_status)
   865→        if nova_etapa:
   866→            db_execute(
   867→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   868→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   869→                (nova_etapa, of_numero, departamento)
   870→            )
   871→        # 3. Sincroniza lotes_producao (status + departamento_atual)
   872→        _sync_lote_status(of_numero, novo_status)
   873→        _atualizar_departamento_lote_from_of(of_numero)
   874→    except Exception as e:
   875→        app.logger.warning('Sync op status falhou: %s', e)

   890→@app.route('/api/finalizar', methods=['POST'])
   891→def api_finalizar():
   892→    """Finalizar producao de pecas especificas e sincroniza com operacoes_producao."""
   893→    try:
   894→        data = request.get_json()
   895→        lotcod = data.get('lote')
   896→        pecas = data.get('pecas', [])
   897→        setor = data.get('setor', '')
   898→        qtd_informada = data.get('qtd_produzida', 0)
   899→
   900→        if not lotcod:
   901→            return jsonify({'error': 'Lote obrigatorio'}), 400
   902→
   903→        setor_para_dept = {
   904→            'CORTE': 'CORTE', 'DOBRA': 'DOBRA', 'SOLDA': 'SOLDA',
   905→            'ACABAMENTO': 'ACABAMENTO', 'MONTAGEM': 'MONTAGEM',
   906→            'FUNILARIA': 'FUNILARIA', 'PINTURA': 'PINTURA',
   907→        }
   908→
   909→        if pecas and setor:
   910→            # Finalizar pecas em um setor especifico (com suporte a quantidade parcial)
   911→            for pcod in pecas:
   912→                row = db_query_one(
   913→                    "SELECT sp.id, sp.of_numero, sp.qtd_produzida, "
   914→                    "COALESCE(of2.qtde_ordem, 0) as qtde_ordem "
   915→                    "FROM serralheria_producao sp "
   916→                    "LEFT JOIN ordens_fabricacao of2 ON sp.of_numero = of2.of_numero "
   917→                    "WHERE sp.lote = %s AND sp.produto = %s AND sp.setor = %s "
   918→                    "AND sp.status = 'em_producao'",
   919→                    (lotcod, pcod, setor)
   920→                )
   921→                if not row:
   922→                    app.logger.warning('Finalizar: peca %s setor %s nao encontrada em producao', pcod, setor)
   923→                    continue
   924→
   925→                qtd_total = _int(row['qtde_ordem'])
   926→                qtd_ja_feita = _int(row['qtd_produzida'])
   927→                restante = qtd_total - qtd_ja_feita
   928→
   929→                if qtd_total <= 0:
   930→                    # Sem qtde_ordem, finaliza direto
   931→                    db_execute(
   932→                        "UPDATE serralheria_producao SET status='finalizado', data_fim=NOW() "
   933→                        "WHERE id=%s", (row['id'],)
   934→                    )
   935→                    dept = setor_para_dept.get(setor, setor)
   936→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   937→                    continue
   938→
   939→                # Validacao: quantidade nao pode exceder o restante
   940→                if qtd_informada > restante:
   941→                    return jsonify({
   942→                        'error': 'Quantidade maxima permitida: %d (restam %d de %d)' % (restante, restante, qtd_total),
   943→                        'max_qtd': restante
   944→                    }), 400
   945→
   946→                if qtd_informada <= 0:
   947→                    return jsonify({'error': 'Informe uma quantidade maior que zero'}), 400
   948→
   949→                nova_qtd = qtd_ja_feita + qtd_informada
   950→
   951→                if nova_qtd >= qtd_total:
   952→                    # Finalizacao completa
   953→                    db_execute(
   954→                        "UPDATE serralheria_producao "
   955→                        "SET status='finalizado', data_fim=NOW(), qtd_produzida=%s "
   956→                        "WHERE id=%s",
