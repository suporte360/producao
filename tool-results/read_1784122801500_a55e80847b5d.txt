   147→   147→def _pg_query_one(sql, params=None):
   148→   148→    """Query PostgreSQL and return one row as dict."""
   149→   149→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   150→   150→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   151→   151→    conn.autocommit = True
   152→   152→    try:
   153→   153→        with conn.cursor() as cur:
   154→   154→            cur.execute(sql, params or ())
   155→   155→            cols = [d[0] for d in cur.description] if cur.description else []
   156→   156→            row = cur.fetchone()
   157→   157→            if row:
   158→   158→                return dict(zip(cols, row))
   159→   159→            return None
   160→   160→    finally:
   161→   161→        conn.close()

   164→   164→def _load_produto_cache():
   165→   165→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   166→   166→    global _produto_cache
   167→   167→    _produto_cache = {}
   168→   168→    # 1) PostgreSQL: tabela produto (pronome real)
   169→   169→    try:
   170→   170→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   171→   171→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   172→   172→        conn.autocommit = True
   173→   173→        try:
   174→   174→            with conn.cursor() as cur:
   175→   175→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   176→   176→                cols = [d[0] for d in cur.description]
   177→   177→                for row in cur.fetchall():
   178→   178→                    r = dict(zip(cols, row))
   179→   179→                    cod = str(r.get('produto', '')).strip()
   180→   180→                    nome = str(r.get('pronome', '')).strip()
   181→   181→                    if cod and nome:
   182→   182→                        _produto_cache[cod] = nome[:200]
   183→   183→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   184→   184→        finally:
   185→   185→            conn.close()
   186→   186→    except Exception as e:
   187→   187→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   188→   188→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   189→   189→    try:
   190→   190→        rows = db_query(
   191→   191→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   192→   192→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   193→   193→        )
   194→   194→        added = 0
   195→   195→        for r in rows:
   196→   196→            cod = str(r.get('codigo_produto', '')).strip()
   197→   197→            nome = str(r.get('descricao_produto', '')).strip()
   198→   198→            if cod and cod not in _produto_cache and nome:
   199→   199→                _produto_cache[cod] = nome[:200]
   200→   200→                added += 1
   201→   201→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   202→   202→    except Exception as e:
   203→   203→        print('[WARN] Cache MariaDB falhou: %s' % e)

   225→   225→def _build_lotes_filter():
   226→   226→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   227→   227→    if _lotes_ativos:
   228→   228→        ph = ','.join(['%s'] * len(_lotes_ativos))
   229→   229→        where = 'WHERE lote_codigo IN (' + ph + ')'
   230→   230→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   231→   231→        params = tuple(_lotes_ativos)
   232→   232→    else:
   233→   233→        where = 'WHERE 1=0'
   234→   234→        join = ' AND 1=0'
   235→   235→        params = ()
   236→   236→    return where, join, params

   828→   828→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.

   830→   830→    if not lote_ordem:

   860→   860→def _sync_lote_status(of_numero, novo_status):
   861→   861→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   862→   862→    if not of_numero:
   863→   863→        return
   864→   864→    of_row = db_query_one(
   865→   865→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",

   868→   868→    if not of_row or not of_row.get('lote_ordem'):
   869→   869→        return
   870→   870→    lote_ordem = of_row['lote_ordem']
   871→   871→
   872→   872→    if novo_status == 'em_andamento':
   873→   873→        # Muda status do lote para em_producao se ainda estiver liberado
   874→   874→        db_execute(
   875→   875→            "UPDATE lotes_producao SET status = 'em_producao' "
   876→   876→            "WHERE ordem = %s AND status = 'liberado'",
   877→   877→            (lote_ordem,)
   878→   878→        )
   879→   879→    elif novo_status == 'concluido':
   880→   880→        # Verifica se TODAS as ops do lote estao concluidas
   881→   881→        total_pend = db_query_one(
   882→   882→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   883→   883→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   884→   884→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   885→   885→            (lote_ordem,)
   886→   886→        )
   887→   887→        if total_pend and int(total_pend['t']) == 0:
   888→   888→            db_execute(
   889→   889→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   890→   890→                (lote_ordem,)
   891→   891→            )

   895→   895→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   896→   896→    if not of_numero or not departamento:

   902→   902→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   903→   903→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   904→   904→            (novo_status, novo_status, of_numero, departamento)

   906→   906→        # 2. Atualiza kanban_cards
