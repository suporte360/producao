   109→def _int(val):
   110→    """Safely convert to int (handles Decimal, None, str)."""
   111→    if val is None:
   112→        return 0
   113→    return int(float(val))
   114→
   115→def _str(val):
   116→    """Safely convert to stripped string."""
   117→    if val is None:
   118→        return ''
   119→    return str(val).strip()
   120→
   121→def _is_sob_medida(codigo):
   122→    """Verifica se o produto codigo termina com -X (sob medida)."""
   123→    if not codigo:
   124→        return False
   125→    c = str(codigo).strip().upper()
   126→    return c.endswith('-X')
   127→
   128→def _extract_s_code(codigo_produto):
   129→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   130→    if not codigo_produto:
   131→        return ''
   132→    m = re.search(r'S-\d+', str(codigo_produto))
   133→    return m.group(0) if m else str(codigo_produto).strip()

   138→def _get_produto_nome(codigo_produto):
   139→    """Get real product name from in-memory cache (PostgreSQL produto.pronome)."""
   140→    if not codigo_produto:
   141→        return ''
   142→    key = str(codigo_produto).strip()
   143→    return _produto_cache.get(key, '')

   146→def _pg_query_one(sql, params=None):
   147→    """Query PostgreSQL and return one row as dict."""
   148→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   149→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   150→    conn.autocommit = True
   151→    try:
   152→        with conn.cursor() as cur:
   153→            cur.execute(sql, params or ())
   154→            cols = [d[0] for d in cur.description] if cur.description else []
   155→            row = cur.fetchone()
   156→            if row:
   157→                return dict(zip(cols, row))
   158→            return None
   159→    finally:
   160→        conn.close()

   163→def _load_produto_cache():
   164→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   165→    global _produto_cache
   166→    _produto_cache = {}
   167→    # 1) PostgreSQL: tabela produto (pronome real)
   168→    try:
   169→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   170→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   171→        conn.autocommit = True
   172→        try:
   173→            with conn.cursor() as cur:
   174→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   175→                cols = [d[0] for d in cur.description]
   176→                for row in cur.fetchall():
   177→                    r = dict(zip(cols, row))
   178→                    cod = str(r.get('produto', '')).strip()
   179→                    nome = str(r.get('pronome', '')).strip()
   180→                    if cod and nome:
   181→                        _produto_cache[cod] = nome[:200]
   182→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   183→        finally:
   184→            conn.close()
   185→    except Exception as e:
   186→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   187→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   188→    try:
   189→        rows = db_query(
   190→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   191→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   192→        )
   193→        added = 0
   194→        for r in rows:
   195→            cod = str(r.get('codigo_produto', '')).strip()
   196→            nome = str(r.get('descricao_produto', '')).strip()
   197→            if cod and cod not in _produto_cache and nome:
   198→                _produto_cache[cod] = nome[:200]
   199→                added += 1
   200→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   201→    except Exception as e:
   202→        print('[WARN] Cache MariaDB falhou: %s' % e)
   203→

   207→def _refresh_lotes_ativos():
   208→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""
   209→    global _lotes_ativos
   210→    try:
   211→        rows = db_query(

   213→            "WHERE status IN ('liberado','em_producao','pausado')"
   214→        )
   215→        _lotes_ativos = set()
   216→        for r in rows:
   217→            if r.get('lote_codigo'):
   218→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   219→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   220→    except Exception as e:
   221→        print('[WARN] Lotes ativos falhou: %s' % e)

   223→
   224→def _build_lotes_filter():
   225→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   226→    if _lotes_ativos:
   227→        ph = ','.join(['%s'] * len(_lotes_ativos))
   228→        where = 'WHERE lote_codigo IN (' + ph + ')'
   229→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   230→        params = tuple(_lotes_ativos)
   231→    else:
   232→        where = 'WHERE 1=0'
   233→        join = ' AND 1=0'
   234→        params = ()
   235→    return where, join, params

   778→def _atualizar_departamento_lote(lote_ordem):
   779→    """Recalcula departamento_atual, setor_atual_seq e total_setores de um lote.
   780→    Replica a logica de mysql_db.atualizar_departamento_atual para o totem."""
   781→    if not lote_ordem:
   782→        return
   783→    # Total de departamentos unicos do lote
   784→    total = db_query_one(
   785→        "SELECT COUNT(DISTINCT op.departamento) as t FROM operacoes_producao op "
   786→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   787→        "WHERE of2.lote_ordem = %s AND op.departamento IS NOT NULL AND op.departamento != ''",
   788→        (lote_ordem,)
   789→    )
   790→    total_setores = int(total['t'] or 0) if total else 0
   791→
   792→    # Primeira operacao pendente (nao concluida) por sequencia
   793→    primeira = db_query_one(
   794→        "SELECT op.departamento, op.sequencia FROM operacoes_producao op "
   795→        "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   796→        "WHERE of2.lote_ordem = %s AND op.status != 'concluido' "
   797→        "ORDER BY op.sequencia ASC LIMIT 1",
   798→        (lote_ordem,)
   799→    )
   800→    dept = primeira['departamento'] if primeira else None
   801→    seq = int(primeira['sequencia']) if primeira else 0
   802→
   803→    db_execute(
   804→        "UPDATE lotes_producao "
   805→        "SET departamento_atual = %s, setor_atual_seq = %s, total_setores = %s "
   806→        "WHERE ordem = %s",
   807→        (dept, seq, total_setores, lote_ordem)
   808→    )

   811→def _sync_lote_status(of_numero, novo_status):
   812→    """Atualiza status do lote em lotes_producao conforme operacoes."""
   813→    if not of_numero:
   814→        return
   815→    of_row = db_query_one(
   816→        "SELECT lote_ordem FROM ordens_fabricacao WHERE of_numero = %s LIMIT 1",
   817→        (of_numero,)
   818→    )
   819→    if not of_row or not of_row.get('lote_ordem'):
   820→        return
   821→    lote_ordem = of_row['lote_ordem']
   822→
   823→    if novo_status == 'em_andamento':
   824→        # Muda status do lote para em_producao se ainda estiver liberado
   825→        db_execute(
   826→            "UPDATE lotes_producao SET status = 'em_producao' "
   827→            "WHERE ordem = %s AND status = 'liberado'",
   828→            (lote_ordem,)
   829→        )
   830→    elif novo_status == 'concluido':
   831→        # Verifica se TODAS as ops do lote estao concluidas
   832→        total_pend = db_query_one(
   833→            "SELECT COUNT(*) as t FROM operacoes_producao op "
   834→            "INNER JOIN ordens_fabricacao of2 ON op.of_numero = of2.of_numero "
   835→            "WHERE of2.lote_ordem = %s AND op.status != 'concluido'",
   836→            (lote_ordem,)
   837→        )
   838→        if total_pend and int(total_pend['t']) == 0:
   839→            db_execute(
   840→                "UPDATE lotes_producao SET status = 'finalizado' WHERE ordem = %s AND status != 'finalizado'",
   841→                (lote_ordem,)
   842→            )

   845→def _sync_op_status(of_numero, departamento, novo_status):
   846→    """Sincroniza operacoes_producao, kanban_cards e lotes_producao quando totem inicia/finaliza."""
   847→    if not of_numero or not departamento:
   848→        return
   849→    try:
   850→        # 1. Atualiza operacoes_producao
   851→        db_execute(
   852→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   853→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   854→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   855→            (novo_status, novo_status, of_numero, departamento)
   856→        )
   857→        # 2. Atualiza kanban_cards
   858→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   859→        nova_etapa = etapa_map.get(novo_status)
   860→        if nova_etapa:
   861→            db_execute(
   862→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   863→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   864→                (nova_etapa, of_numero, departamento)
   865→            )
   866→        # 3. Sincroniza lotes_producao (status + departamento_atual)
   867→        _sync_lote_status(of_numero, novo_status)
   868→        _atualizar_departamento_lote_from_of(of_numero)
   869→    except Exception as e:
   870→        app.logger.warning('Sync op status falhou: %s', e)
