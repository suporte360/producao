   297→   297→// ════════════════ RESUMO MATERIAL NA FABRICA ════════════════
   298→   298→function carregarResumoEstoque() {
   299→   299→  fetch('/api/estoque?filtro=baixo&tipo=fabrica').then(r => r.json()).then(d => {
   300→   300→    if (d.status !== 'success') return;
   301→   301→    const itens = d.itens || [];
   302→   302→    const badge = document.getElementById('kpiPrecisaSep');
   303→   303→    if (itens.length > 0) {
   304→   304→      badge.textContent = itens.length + ' precisam separar';
   305→   305→      badge.classList.remove('hidden');
   306→   306→    } else {
   307→   307→      badge.classList.add('hidden');
   308→   308→    }
   309→   309→    const el = document.getElementById('listaEstoqueResumo');
   310→   310→    if (!itens.length) {
   311→   311→      el.innerHTML = '<div class="prodsys-card p-4 text-center col-span-full text-sm text-green-600"><i class="fa-solid fa-circle-check mr-1"></i>Todos os materiais estao OK</div>';
   312→   312→      return;
   313→   313→    }
   314→   314→    el.innerHTML = itens.slice(0, 6).map(item => {
   315→   315→      const qtd = parseFloat(item.qtd_atual) || 0;
   316→   316→      const qtdIni = parseFloat(item.qtd_inicial) || 0;
   317→   317→      const sem = qtd <= 0;
   318→   318→      const desc = (item.descricao||'—').replace(/'/g,"\\'");
   319→   319→      const un = item.unidade || 'UN';
   320→   320→      const pct = qtdIni > 0 ? Math.round((qtd / qtdIni) * 100) : 0;
   321→   321→      return '<div class="prodsys-card border-l-4 ' + (sem ? 'border-red-500 bg-red-50/30' : 'border-amber-400 bg-amber-50/30') + ' px-3 py-2 flex items-center justify-between gap-2">' +
   322→   322→        '<div class="min-w-0"><div class="font-bold text-sm text-gray-800 truncate">' + (item.descricao||'—') + '</div>' +
   323→   323→        '<div class="text-[11px] text-gray-400">' + item.codigo + ' · ' + un + '</div></div>' +
   324→   324→        '<div class="text-right flex-shrink-0"><div class="font-black text-lg ' + (sem?'text-red-600':'text-amber-600') + '">' + qtd.toFixed(1) + '</div>' +
   325→   325→        '<div class="text-[10px] text-gray-400">de ' + qtdIni.toFixed(1) + (pct > 0 ? ' (' + pct + '%)' : '') + '</div></div></div>';
   326→   326→    }).join('');
   327→   327→  });
   328→   328→}
