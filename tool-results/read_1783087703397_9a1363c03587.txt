   411→function esc(s) {
   412→  if (!s) return '';
   413→  var d = document.createElement('div');
   414→  d.appendChild(document.createTextNode(String(s)));
   415→  return d.innerHTML;

   418→/* ── Animate Number ── */
   419→function animateValue(el, end) {
   420→  var startVal = parseInt(el.textContent) || 0;
   421→  if (startVal === end) return;
   422→  var dur = 600;
   423→  var st = null;
   424→  function step(ts) {
   425→    if (!st) st = ts;
   426→    var p = Math.min((ts - st) / dur, 1);
   427→    var eased = 1 - Math.pow(1 - p, 3);
   428→    el.textContent = Math.round(startVal + (end - startVal) * eased);
   429→    if (p < 1) requestAnimationFrame(step);
   430→    else el.style.animation = 'countPulse .3s ease';
   431→  }
   432→  el.style.animation = '';
   433→  requestAnimationFrame(step);
   434→}

   467→function updatePipeline(ativos) {

   471→    document.getElementById('node-' + i).classList.remove('active');

   473→  }
   474→
   475→  /* Assign active workers to sectors cyclically */
   476→  if (ativos.length === 0) return;
   477→
   478→  /* Group by unique worker (one entry per worker) */
   479→  var seen = {};
   480→  var workers = [];
   481→  for (i = 0; i < ativos.length; i++) {

   489→  for (i = 0; i < workers.length && i < 5; i++) {

   491→    node.classList.add('active');
   492→    document.getElementById('nw-' + i).textContent = workers[i].usuario_nome;
   493→  }
   494→
   495→  /* If there are active workers, keep all pipes active */
   496→  var arrows = document.querySelectorAll('.p-arrow');
   497→  var hasActive = workers.length > 0;
   498→  for (i = 0; i < arrows.length; i++) {
   499→    if (hasActive) {
   500→      arrows[i].classList.remove('inactive');
   501→    } else {
   502→      arrows[i].classList.add('inactive');

   561→function renderWorkers(ativos, data) {

   584→        lote_desc: a.lote_desc || '',
   585→        pecas: [],
   586→        data_inicio: a.data_inicio,
   587→        pecas_count: a.pecas_count || 1

   592→      produto_nome: a.produto_nome || ''

   730→      html += '<div class="peca-card">'
   731→        + '<div><div class="peca-codigo">' + esc(p.codigo) + '</div>'
   732→        + '<div class="peca-tipo">' + tipoLabel + ' | OF ' + p.ordem + '</div></div>'
   733→        + '<div class="peca-qtd">x' + p.quantidade + '</div>'
   734→        + '</div>';
   735→    }
   736→
   737→    if (currentPecas.length === 0) {
   738→      html += '<div class="loading" style="color:#f59e0b">Nenhuma peca para fabricar neste lote</div>';
   739→    }
   740→
   741→    document.getElementById('pecasContent').innerHTML = html;
   742→
   743→    var actions = '<button class="btn-back" onclick="showScreen(2)">&#8592; Voltar aos Lotes</button>';
   744→    if (data.producao_ativa) {
   745→      actions = '<div class="status-bar">Em producao por: ' + esc(data.producao_ativa.usuario_nome) + '</div>'
   746→        + '<button class="btn-touch btn-red" onclick="confirmarFinalizar(\'' + esc(lotcod) + '\')">FINALIZAR PRODUCAO</button>'
   747→        + '<button class="btn-back" onclick="showScreen(2)">&#8592; Voltar aos Lotes</button>';
   748→    } else if (currentPecas.length > 0) {
   749→      actions = '<button class="btn-touch btn-green" onclick="iniciarProducao()">INICIAR PRODUCAO</button>'
   750→        + '<button class="btn-back" onclick="showScreen(2)">&#8592; Voltar aos Lotes</button>';
   751→    }
   752→    document.getElementById('pecasActions').innerHTML = actions;
   753→  }).catch(function(e) {
   754→    console.error('Pecas error:', e);
   755→    document.getElementById('pecasContent').innerHTML = '<div class="loading" style="color:#ef5350">Erro ao carregar</div>';
   756→  });
   757→}

   781→function loadUsuarios() {
   782→  fetch('/api/usuarios').then(function(r) { return r.json(); }).then(function(data) {
   783→    if (!data.usuarios || data.usuarios.length === 0) {
   784→      document.getElementById('usuarioGrid').innerHTML = '<div class="loading">Nenhum usuario cadastrado</div>';
   785→      return;

   788→    for (var i = 0; i < data.usuarios.length; i++) {
   789→      var u = data.usuarios[i];

   791→        + '<div class="user-icon">&#9881;</div>'
   792→        + '<div>' + esc(u.nome) + '</div>'
   793→        + '<div class="user-setor">' + esc(u.setor) + '</div>'
   794→        + '</div>';
   795→    }

   797→  }).catch(function(e) { console.error('Usuarios error:', e); });

   800→/* ── Selecionar Usuario → Iniciar → RETURN TO DASHBOARD ── */
   801→function selecionarUsuario(id, nome) {
   802→  var pecas = [];
   803→  for (var i = 0; i < currentPecas.length; i++) {
   804→    pecas.push({ ordem: currentPecas[i].ordem, codigo: currentPecas[i].codigo });
   805→  }
   806→  fetch('/api/iniciar', {
   807→    method: 'POST',
   808→    headers: { 'Content-Type': 'application/json' },
   809→    body: JSON.stringify({ lote: currentLote, usuario_id: id, usuario_nome: nome, pecas: pecas })
   810→  }).then(function(r) { return r.json(); }).then(function(data) {
   811→    if (data.success) {
   812→      /* Return to dashboard as requested */
   813→      showScreen(1);
   814→    } else {
   815→      alert('Erro: ' + (data.error || 'Erro desconhecido'));
   816→    }
   817→  }).catch(function(e) { alert('Erro de conexao'); });
   818→}
