   411→   411→/* ─── Auto-scroll suave (fractional accumulator) ── */
   412→   412→var scrollCol = null;
   413→   413→var scrollAccum = 0;
   414→   414→var scrollSpeed = 0.4;
   415→   415→var scrollDir = 1;
   416→   416→
   417→   417→function autoScrollTick() {
   418→   418→  if (!scrollCol) return;
   419→   419→  var maxScroll = scrollCol.scrollHeight - scrollCol.clientHeight;
   420→   420→  if (maxScroll <= 0) return;
   421→   421→  scrollAccum += scrollSpeed * scrollDir;
   422→   422→  if (Math.abs(scrollAccum) >= 1) {
   423→   423→    scrollCol.scrollTop += Math.round(scrollAccum);
   424→   424→    scrollAccum = 0;
   425→   425→  }
   426→   426→  if (scrollCol.scrollTop >= maxScroll - 2) { scrollDir = -1; }
   427→   427→  else if (scrollCol.scrollTop <= 0) { scrollDir = 1; }
   428→   428→}

   431→   431→function renderLotes(lotes) {
   432→   432→  var panel = document.getElementById('painelLotes');
   433→   433→  document.getElementById('cntLotes').textContent = lotes.length;
   434→   434→  if (!lotes.length) {
   435→   435→    panel.innerHTML = '<div class="empty-panel"><i class="fa-solid fa-circle-check" style="color:#14532d"></i><span>Nenhuma OP aguardando separacao</span></div>';
   436→   436→    return;
   437→   437→  }
   438→   438→  var html = '';
   439→   439→  for (var i = 0; i < lotes.length; i++) {
   440→   440→    var l = lotes[i];
   441→   441→    var sepStatus = l.separacao_status || 'pendente';
   442→   442→    var prio = (l.prioridade||'media').toLowerCase();
   443→   443→    var urgente = prio === 'urgente' || prio === 'alta';
   444→   444→    var atrasada = (l.dias_atraso||0) > 0;
   445→   445→    var cls = 'op-card';
   446→   446→    if (urgente) cls += ' urgente';
   447→   447→    if (atrasada) cls += ' atrasada';
   448→   448→    if (sepStatus === 'separando') cls += ' separando';
   449→   449→    else if (sepStatus === 'separado') cls += ' separado';
   450→   450→    else if (sepStatus === 'entregue') cls += ' entregue';
   451→   451→
   452→   452→    /* OFs horizontais - mostra todas, com cor por status */
   453→   453→    var ofs = l.ordens_do_lote || [];
   454→   454→    var ofsHtml = '';
   455→   455→    var sepCount = 0;
   456→   456→    var entCount = 0;
   457→   457→    for (var k = 0; k < ofs.length; k++) {
   458→   458→      var o = ofs[k];
   459→   459→      if (o.separacao_status === 'separado') { sepCount++; }
   460→   460→      if (o.separacao_status === 'entregue') { entCount++; }
   461→   461→      if (ofsHtml) ofsHtml += '<span class="lote-of-sep">-</span>';
   462→   462→      var ofCls = 'lote-of-item';
   463→   463→      var ofPrio = (o.prioridade||'media').toLowerCase();
   464→   464→      var ofAtraso = (o.dias_atraso||0) > 0;
   465→   465→      if (ofPrio === 'urgente') ofCls += ' urgente-of';
   466→   466→      else if (ofAtraso) ofCls += ' atraso';
   467→   467→      if (o.separacao_status === 'separando') ofCls += ' separando';
   468→   468→      else if (o.separacao_status === 'separado') ofCls += ' separado';
   469→   469→      else if (o.separacao_status === 'entregue') ofCls += ' entregue';
   470→   470→      ofsHtml += '<span class="' + ofCls + '">#' + o.ordem + '</span>';
   471→   471→    }
   472→   472→    var sepInfo = '';
   473→   473→    if (sepCount > 0) {
   474→   474→      sepInfo = '<span class="lote-sep-info" style="color:#fbbf24">' + sepCount + ' OF' + (sepCount > 1 ? 's' : '') + ' separada' + (sepCount > 1 ? 's' : '') + '</span>';
   475→   475→    }
   476→   476→    if (entCount > 0) {
   477→   477→      sepInfo += '<span class="lote-sep-info" style="color:#64748b">' + entCount + ' entregue' + (entCount > 1 ? 's' : '') + '</span>';
   478→   478→    }
   479→   479→
   480→   480→    /* componentes */
   481→   481→    var matsHtml = '';
   482→   482→    var comps = l.componentes || [];
   483→   483→    var totalComps = l.total_componentes || comps.length;
   484→   484→    if (comps.length > 0) {
   485→   485→      matsHtml = '<div class="op-mats"><div class="op-mats-title"><i class="fa-solid fa-screwdriver-wrench" style="margin-right:4px"></i>' + totalComps + ' componente' + (totalComps !== 1 ? 's' : '') + ' para separar</div>';
   486→   486→      for (var j = 0; j < comps.length; j++) {
   487→   487→        var c = comps[j];
   488→   488→        matsHtml += '<div class="op-mat-row">' +
   489→   489→          '<span class="mat-cod">' + c.codigo + '</span>' +
   490→   490→          '<span class="mat-desc">' + c.descricao + '</span>' +
   491→   491→          '<span class="mat-qtd">' + c.qtd_necessaria + '</span>' +
   492→   492→          '<span class="mat-und">' + c.unidade + '</span>' +
   493→   493→        '</div>';
   494→   494→      }
   495→   495→      matsHtml += '</div>';
   496→   496→    } else {
   497→   497→      matsHtml = '<div class="op-comp-count" style="margin-top:4px"><i class="fa-solid fa-info-circle" style="margin-right:3px"></i>Sem componentes brutos (so pecas fabricadas)</div>';
   498→   498→    }
   499→   499→
   500→   500→    html += '<div class="' + cls + '">' +
   501→   501→      '<div class="lote-header">' +
   502→   502→        '<div class="lote-id">' +
   503→   503→          '<span class="lote-lp">LP ' + (l.lote_codigo || '---') + '</span>' +
   504→   504→          '<span class="lote-of-label">OF</span>' +
   505→   505→          '<span class="lote-ofs">' + ofsHtml + '</span>' +
   506→   506→          sepInfo +
   507→   507→        '</div>' +
   508→   508→        '<div class="lote-badges">' +
   509→   509→          badgePrio(l.prioridade) +
   510→   510→          badgeSep(sepStatus) +
   511→   511→        '</div>' +
   512→   512→      '</div>' +
   513→   513→      '<div class="op-info">' +
   514→   514→        '<div class="op-prod">' + (l.descricao_produto || '\u2014').substring(0, 50) + '</div>' +
   515→   515→        '<div class="op-meta">' +
   516→   516→          '<span class="op-qtd"><i class="fa-solid fa-boxes-stacked" style="font-size:.6rem;margin-right:2px"></i>' + Math.round(l.qtde_ordem || 0) + ' pc</span>' +
   517→   517→          (l.departamento_atual ? '<span class="op-dept">' + l.departamento_atual + '</span>' : '') +
   518→   518→          (atrasada ? '<span class="op-qtd" style="color:#f87171"><i class="fa-solid fa-circle-exclamation" style="font-size:.6rem;margin-right:2px"></i>' + l.dias_atraso + 'd atraso</span>' : '') +

   521→   521→        matsHtml +
   522→   522→      '</div>' +
   523→   523→    '</div>';
   524→   524→  }
   525→   525→  panel.innerHTML = html;
   526→   526→}
