   562→/* ─── Producao por Periodo (AJAX) ─── */
   563→function buscarProducaoPeriodo() {
   564→  var di = document.getElementById('filtroDtInicio').value;
   565→  var df = document.getElementById('filtroDtFim').value;
   566→  if (!di || !df) return;
   567→
   568→  document.getElementById('periodoLoader').classList.remove('hidden');
   569→  document.getElementById('periodoTabelaWrap').innerHTML = '';
   570→
   571→  fetch('/api/relatorios/producao_periodo?data_inicio=' + di + '&data_fim=' + df)
   572→    .then(function(r) { return r.json(); })
   573→    .then(function(data) {
   574→      document.getElementById('periodoLoader').classList.add('hidden');
   575→      if (data.error) {
   576→        document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-red-600 text-center py-3 text-sm">Erro: ' + data.error + '</p>';
   577→        return;
   578→      }
   579→      var kp = data.kpis;
   580→      document.getElementById('pKpiTotal').textContent = kp.total_lotes || 0;
   581→      document.getElementById('pKpiEmProducao').textContent = kp.total_em_producao || 0;
   582→      document.getElementById('pKpiFinalizados').textContent = kp.total_finalizados || 0;
   583→      document.getElementById('pKpiPecas').textContent = Math.round(Number(kp.total_pecas) || 0);
   584→      document.getElementById('pKpiAtrasados').textContent = kp.atrasados_agora || 0;
   585→      document.getElementById('pKpiTempo').textContent = Math.round(kp.tempo_medio_dias || 0) + 'd';
   586→
   587→      if (data.lotes.length === 0) {
   588→        document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-gray-400 text-center py-8">Nenhum lote encontrado neste periodo.</p>';
   589→        return;
   590→      }
   591→      var statusColors = {'em_producao':'bg-orange-100 text-orange-700','pausado':'bg-red-100 text-red-700','finalizado':'bg-green-100 text-green-700','liberado':'bg-blue-100 text-blue-700','importado':'bg-gray-100 text-gray-600'};
   592→      var statusLabels = {'em_producao':'Em Producao','pausado':'Pausado','finalizado':'Finalizado','liberado':'Liberado','importado':'Importado'};
   593→      var prioColors = {'urgente':'text-red-600 font-black','alta':'text-orange-600 font-bold','media':'text-yellow-600','baixa':'text-gray-400'};
   594→      var html = '<div class="overflow-x-auto"><table class="w-full text-sm"><thead class="text-xs uppercase tracking-wider text-blue-700 bg-blue-50 border-b border-blue-200"><tr>' +
   595→        '<th class="px-3 py-2.5 text-left">Lote</th><th class="px-3 py-2.5 text-left">Descricao</th><th class="px-3 py-2.5 text-center">Qtd</th><th class="px-3 py-2.5 text-center">Status</th><th class="px-3 py-2.5 text-left">Setor</th><th class="px-3 py-2.5 text-center">Previsao</th><th class="px-3 py-2.5 text-center">Dias</th></tr></thead><tbody class="divide-y divide-gray-50">';
   596→      for (var i = 0; i < data.lotes.length; i++) {
   597→        var l = data.lotes[i];
   598→        var sc = statusColors[l.status] || 'bg-gray-100 text-gray-600';
   599→        var sl = statusLabels[l.status] || l.status || '—';
   600→        var pc = prioColors[l.prioridade] || '';
   601→        var prevStr = (l.data_previsao_erp || '').substring(0, 10);
   602→        html += '<tr class="hover:bg-gray-50"><td class="px-3 py-2.5 font-bold text-blue-700">' + (l.lote_codigo || l.ordem || '') + '</td>' +
   603→          '<td class="px-3 py-2.5 text-gray-700 max-w-[200px] truncate">' + (l.descricao_produto || '—') + '</td>' +
   604→          '<td class="px-3 py-2.5 text-center font-bold">' + (Math.round(Number(l.qtde_ordem) || 0)) + '</td>' +
   605→          '<td class="px-3 py-2.5 text-center"><span class="inline-block px-2 py-0.5 rounded-full text-[10px] font-bold ' + sc + '">' + sl + '</span></td>' +
   606→          '<td class="px-3 py-2.5 text-gray-600 text-xs">' + (l.departamento_atual || '—') + '</td>' +
   607→          '<td class="px-3 py-2.5 text-center text-xs ' + pc + '">' + (prevStr || '—') + '</td>' +
   608→          '<td class="px-3 py-2.5 text-center font-bold">' + (l.dias_producao || '—') + '</td></tr>';
   609→      }
   610→      html += '</tbody></table></div>';
   611→      document.getElementById('periodoTabelaWrap').innerHTML = html;
   612→    })
   613→    .catch(function(e) {
   614→      document.getElementById('periodoLoader').classList.add('hidden');
   615→      document.getElementById('periodoTabelaWrap').innerHTML = '<p class="text-red-600 text-center py-3 text-sm">Erro de conexao</p>';
   616→    });
   617→}
