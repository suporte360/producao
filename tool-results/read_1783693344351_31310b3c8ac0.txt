   109→def _int(val):
   110→    """Safely convert to int (handles Decimal, None, str)."""
   111→    if val is None:
   112→        return 0
   113→    return int(float(val))
   114→
   115→def _str(val):
   116→    """Safely convert to stripped string."""
   117→    if val is None:
   118→        return ''
   119→    return str(val).strip()
   120→
   121→def _is_sob_medida(codigo):
   122→    """Verifica se o produto codigo termina com -X (sob medida)."""
   123→    if not codigo:
   124→        return False
   125→    c = str(codigo).strip().upper()
   126→    return c.endswith('-X')
   127→
   128→def _extract_s_code(codigo_produto):
   129→    """Extract S-XXXX from strings like 'PRODUCAO DE S-0240'."""
   130→    if not codigo_produto:
   131→        return ''
   132→    m = re.search(r'S-\d+', str(codigo_produto))
   133→    return m.group(0) if m else str(codigo_produto).strip()

   138→def _get_produto_nome(codigo_produto):
   139→    """Get real product name from in-memory cache (PostgreSQL produto.pronome)."""
   140→    if not codigo_produto:
   141→        return ''
   142→    key = str(codigo_produto).strip()
   143→    return _produto_cache.get(key, '')

   146→def _pg_query_one(sql, params=None):
   147→    """Query PostgreSQL and return one row as dict."""
   148→    conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   149→                            user=PG_USER, password=PG_PASS, connect_timeout=5)
   150→    conn.autocommit = True
   151→    try:
   152→        with conn.cursor() as cur:
   153→            cur.execute(sql, params or ())
   154→            cols = [d[0] for d in cur.description] if cur.description else []
   155→            row = cur.fetchone()
   156→            if row:
   157→                return dict(zip(cols, row))
   158→            return None
   159→    finally:
   160→        conn.close()
   161→
   162→
   163→def _load_produto_cache():
   164→    """Load product names from PostgreSQL produto.pronome + MariaDB fallback."""
   165→    global _produto_cache
   166→    _produto_cache = {}
   167→    # 1) PostgreSQL: tabela produto (pronome real)
   168→    try:
   169→        conn = psycopg2.connect(host=PG_HOST, port=PG_PORT, dbname=PG_DB,
   170→                                user=PG_USER, password=PG_PASS, connect_timeout=5)
   171→        conn.autocommit = True
   172→        try:
   173→            with conn.cursor() as cur:
   174→                cur.execute("SELECT produto, pronome FROM public.produto WHERE produto IS NOT NULL AND pronome IS NOT NULL")
   175→                cols = [d[0] for d in cur.description]
   176→                for row in cur.fetchall():
   177→                    r = dict(zip(cols, row))
   178→                    cod = str(r.get('produto', '')).strip()
   179→                    nome = str(r.get('pronome', '')).strip()
   180→                    if cod and nome:
   181→                        _produto_cache[cod] = nome[:200]
   182→                print('[OK] Cache de produtos (PostgreSQL): %d' % len(_produto_cache))
   183→        finally:
   184→            conn.close()
   185→    except Exception as e:
   186→        print('[WARN] Cache PostgreSQL falhou: %s' % e)
   187→    # 2) Fallback MariaDB: lotes_producao (preenche codigos que o PG nao tem)
   188→    try:
   189→        rows = db_query(
   190→            "SELECT codigo_produto, descricao_produto FROM lotes_producao "
   191→            "WHERE codigo_produto IS NOT NULL AND descricao_produto IS NOT NULL"
   192→        )
   193→        added = 0
   194→        for r in rows:
   195→            cod = str(r.get('codigo_produto', '')).strip()
   196→            nome = str(r.get('descricao_produto', '')).strip()
   197→            if cod and cod not in _produto_cache and nome:
   198→                _produto_cache[cod] = nome[:200]
   199→                added += 1
   200→        print('[OK] Cache produtos (MariaDB fallback): +%d' % added)
   201→    except Exception as e:
   202→        print('[WARN] Cache MariaDB falhou: %s' % e)
   203→

   207→def _refresh_lotes_ativos():
   208→    """Carrega lotes ativos do MariaDB (liberado, em_producao, pausado)."""
   209→    global _lotes_ativos
   210→    try:
   211→        rows = db_query(
   212→            "SELECT DISTINCT lote_codigo FROM lotes_producao "
   213→            "WHERE status IN ('liberado','em_producao','pausado')"
   214→        )
   215→        _lotes_ativos = set()
   216→        for r in rows:
   217→            if r.get('lote_codigo'):
   218→                _lotes_ativos.add(str(r['lote_codigo']).strip())
   219→        print('[OK] Lotes ativos: %d (MariaDB)' % len(_lotes_ativos))
   220→    except Exception as e:
   221→        print('[WARN] Lotes ativos falhou: %s' % e)
   222→
   223→
   224→def _build_lotes_filter():
   225→    """Retorna (where_clause, join_clause, params) para filtrar lotes ativos."""
   226→    if _lotes_ativos:
   227→        ph = ','.join(['%s'] * len(_lotes_ativos))
   228→        where = 'WHERE lote_codigo IN (' + ph + ')'
   229→        join = ' AND lp.lote_codigo IN (' + ph + ')'
   230→        params = tuple(_lotes_ativos)
   231→    else:
   232→        where = 'WHERE 1=0'
   233→        join = ' AND 1=0'
   234→        params = ()
   235→    return where, join, params

   596→@app.route('/api/lotes/<lotcod>/pecas')
   597→def api_pecas(lotcod):
   598→    """Pecas (OFs filhas) de um lote com operacoes por setor."""
   599→    try:
   600→        lote = db_query_one(
   601→            "SELECT ordem, lote_codigo, descricao_produto, qtde_ordem "
   602→            "FROM lotes_producao WHERE lote_codigo = %s",
   603→            (lotcod,)
   604→        )
   605→        if not lote:
   606→            return jsonify({'error': 'Lote nao encontrado'}), 404
   607→
   608→        pecas = db_query("""
   609→            SELECT of.of_numero, of.codigo_produto as codigo, of.descricao_produto as descricao,
   610→                   of.qtde_ordem as quantidade, of.tipo
   611→            FROM ordens_fabricacao of
   612→            WHERE of.lote_ordem = %s AND of.tipo = 'filho'
   613→            ORDER BY of.codigo_produto
   614→        """, (lote['ordem'],))
   615→
   616→        pai = db_query_one("""
   617→            SELECT codigo_produto, descricao_produto, qtde_ordem
   618→            FROM ordens_fabricacao WHERE lote_ordem = %s AND tipo = 'pai' LIMIT 1
   619→        """, (lote['ordem'],))
   620→
   621→        produto_final = None
   622→        if pai:
   623→            pai_nome = _get_produto_nome(pai.get('codigo_produto'))
   624→            produto_final = {
   625→                'codigo': _str(pai['codigo_produto']),
   626→                'descricao': pai_nome or _str(pai['descricao_produto']),
   627→                'quantidade': _int(pai['qtde_ordem']),
   628→            }
   629→
   630→        pecas_result = []
   631→        for p in pecas:
   632→            ops = db_query("""
   633→                SELECT id, departamento, descricao_operacao, status, sequencia
   634→                FROM operacoes_producao
   635→                WHERE of_numero = %s
   636→                ORDER BY sequencia
   637→            """, (p['of_numero'],))
   638→
   639→            setores = list(set(
   640→                op['departamento'] for op in ops
   641→                if op.get('departamento') and op['departamento'] in SETORES_SERRALHERIA
   642→            ))
   643→
   644→            peca_cod = str(p.get('codigo', '')).strip()
   645→            nome_real = _get_produto_nome(peca_cod)
   646→            # Fallback: consulta direta ao PG se cache falhou
   647→            if not nome_real and peca_cod:
   648→                try:
   649→                    pg_row = _pg_query_one(
   650→                        "SELECT pronome FROM public.produto WHERE TRIM(produto) = %s LIMIT 1",
   651→                        (peca_cod,)
   652→                    )
   653→                    if pg_row and pg_row.get('pronome'):
   654→                        nome_real = str(pg_row['pronome']).strip()[:200]
   655→                        _produto_cache[peca_cod] = nome_real
   656→                except Exception:
   657→                    pass
   658→            produto_nome = nome_real if nome_real else _str(p['descricao'])
   659→
   660→            pecas_result.append({
   661→                'of_numero': p['of_numero'],
   662→                'ordem': p['of_numero'],
   663→                'codigo': _str(p['codigo']),
   664→                'quantidade': _int(p['quantidade']),
   665→                'produto_nome': produto_nome,
   666→                'setores': setores,
   667→                'operacoes': ops,
   668→            })
   669→
   670→        ativas_raw = db_query(
   671→            "SELECT produto, of_numero, setor, usuario_nome "
   672→            "FROM serralheria_producao WHERE lote = %s AND status = 'em_producao'",
   673→            (lotcod,)
   674→        )
   675→        pecas_ativas = {}
   676→        for row in ativas_raw:
   677→            peca = row['produto']
   678→            if peca not in pecas_ativas:
   679→                pecas_ativas[peca] = []
   680→            pecas_ativas[peca].append({
   681→                'setor': row['setor'],
   682→                'usuario': row['usuario_nome'],
   683→            })
   684→
   685→        return jsonify({
   686→            'lote': lotcod,
   687→            'descricao': _str(lote['descricao_produto']),
   688→            'produto_final': produto_final,
   689→            'pecas': pecas_result,
   690→            'pecas_ativas': pecas_ativas,
   691→        })
   692→    except Exception as e:
   693→        app.logger.error('Erro pecas: %s', e)
   694→        return jsonify({'error': str(e)}), 500

   711→@app.route('/api/iniciar', methods=['POST'])
   712→def api_iniciar():
   713→    """Iniciar producao: salva no MariaDB e sincroniza operacoes_producao + kanban."""
   714→    try:
   715→        data = request.get_json()
   716→        lotcod = data.get('lote')
   717→        usuario_id = data.get('usuario_id')
   718→        usuario_nome = data.get('usuario_nome')
   719→        setor = data.get('setor', '')
   720→        pecas = data.get('pecas', [])
   721→
   722→        if not lotcod or not usuario_id:
   723→            return jsonify({'error': 'Dados obrigatorios: lote, usuario_id'}), 400
   724→
   725→        # Mapeia setor totem -> departamento operacoes_producao
   726→        setor_para_dept = {
   727→            'CORTE': 'CORTE', 'DOBRA': 'DOBRA', 'SOLDA': 'SOLDA',
   728→            'ACABAMENTO': 'ACABAMENTO', 'MONTAGEM': 'MONTAGEM',
   729→            'FUNILARIA': 'FUNILARIA', 'PINTURA': 'PINTURA',
   730→        }
   731→        dept = setor_para_dept.get(setor, setor)
   732→
   733→        for peca in pecas:
   734→            db_execute(
   735→                "INSERT INTO serralheria_producao "
   736→                "(ordem_id, lote, produto, of_numero, usuario_id, usuario_nome, setor, status, data_inicio) "
   737→                "VALUES (%s, %s, %s, %s, %s, %s, %s, 'em_producao', NOW())",
   738→                (peca.get('ordem'), lotcod, peca['codigo'], peca.get('ordem'),
   739→                 usuario_id, usuario_nome, setor)
   740→            )
   741→            # Sincroniza operacoes_producao -> em_andamento
   742→            _sync_op_status(peca.get('ordem'), dept, 'em_andamento')
   743→
   744→        app.logger.info(
   745→            'Producao iniciada: lote=%s usuario=%s setor=%s pecas=%d',
   746→            lotcod, usuario_nome, setor, len(pecas)
   747→        )
   748→        return jsonify({'success': True, 'mensagem': 'Producao iniciada!'})
   749→    except Exception as e:
   750→        app.logger.error('Erro iniciar: %s', e)
   751→        return jsonify({'error': str(e)}), 500

   754→def _sync_op_status(of_numero, departamento, novo_status):
   755→    """Sincroniza operacoes_producao e kanban_cards quando totem inicia/finaliza."""
   756→    if not of_numero or not departamento:
   757→        return
   758→    try:
   759→        # Atualiza operacoes_producao
   760→        db_execute(
   761→            "UPDATE operacoes_producao SET status = %s, data_inicio = COALESCE(data_inicio, NOW()), "
   762→            "data_fim = CASE WHEN %s = 'concluido' THEN NOW() ELSE data_fim END "
   763→            "WHERE of_numero = %s AND departamento = %s AND status != 'concluido'",
   764→            (novo_status, novo_status, of_numero, departamento)
   765→        )
   766→        # Atualiza kanban_cards
   767→        etapa_map = {'em_andamento': 'em_producao', 'concluido': 'concluido', 'pendente': 'aguardando'}
   768→        nova_etapa = etapa_map.get(novo_status)
   769→        if nova_etapa:
   770→            db_execute(
   771→                "UPDATE kanban_cards SET etapa = %s, atualizado_em = NOW() "
   772→                "WHERE of_numero = %s AND departamento = %s AND etapa != 'concluido'",
   773→                (nova_etapa, of_numero, departamento)
   774→            )
   775→    except Exception as e:
   776→        app.logger.warning('Sync op status falhou: %s', e)

   779→@app.route('/api/finalizar', methods=['POST'])
   780→def api_finalizar():
   781→    """Finalizar producao de pecas especificas e sincroniza com operacoes_producao."""
   782→    try:
   783→        data = request.get_json()
   784→        lotcod = data.get('lote')
   785→        pecas = data.get('pecas', [])
   786→        setor = data.get('setor', '')
   787→
   788→        if not lotcod:
   789→            return jsonify({'error': 'Lote obrigatorio'}), 400
   790→
   791→        # Busca setor e of_numero das pecas ativas para sincronizar
   792→        setor_para_dept = {
   793→            'CORTE': 'CORTE', 'DOBRA': 'DOBRA', 'SOLDA': 'SOLDA',
   794→            'ACABAMENTO': 'ACABAMENTO', 'MONTAGEM': 'MONTAGEM',
   795→            'FUNILARIA': 'FUNILARIA', 'PINTURA': 'PINTURA',
   796→        }
   797→
   798→        if pecas and setor:
   799→            # Finalizar pecas em um setor especifico
   800→            placeholders = ','.join(['%s'] * len(pecas))
   801→            db_execute(
   802→                f"UPDATE serralheria_producao "
   803→                f"SET status = 'finalizado', data_fim = NOW() "
   804→                f"WHERE lote = %s AND produto IN ({placeholders}) AND setor = %s AND status = 'em_producao'",
   805→                [lotcod] + pecas + [setor]
   806→            )
   807→            for pcod in pecas:
   808→                row = db_query_one(
   809→                    "SELECT of_numero FROM serralheria_producao "
   810→                    "WHERE lote = %s AND produto = %s AND setor = %s AND status = 'finalizado' "
   811→                    "ORDER BY data_fim DESC LIMIT 1",
   812→                    (lotcod, pcod, setor)
   813→                )
   814→                if row:
   815→                    dept = setor_para_dept.get(setor, setor)
   816→                    _sync_op_status(row['of_numero'], dept, 'concluido')
   817→            app.logger.info('Lote %s: %d peca(s) finalizada(s) no setor %s', lotcod, len(pecas), setor)
   818→        elif pecas:
   819→            # Finalizar todas as producoes das pecas selecionadas (todos os setores)
   820→            placeholders = ','.join(['%s'] * len(pecas))
   821→            rows = db_query(
   822→                f"SELECT of_numero, setor FROM serralheria_producao "
   823→                f"WHERE lote = %s AND produto IN ({placeholders}) AND status = 'em_producao'",
   824→                [lotcod] + pecas
   825→            )
   826→            db_execute(
   827→                f"UPDATE serralheria_producao "
   828→                f"SET status = 'finalizado', data_fim = NOW() "
   829→                f"WHERE lote = %s AND produto IN ({placeholders}) AND status = 'em_producao'",
   830→                [lotcod] + pecas
   831→            )
   832→            for row in rows:
   833→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
   834→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
   835→            app.logger.info('Lote %s: %d peca(s) finalizada(s) no totem', lotcod, len(pecas))
   836→        else:
   837→            # Finalizar todas as pecas do lote
   838→            rows = db_query(
   839→                "SELECT of_numero, setor FROM serralheria_producao "
   840→                "WHERE lote = %s AND status = 'em_producao'", (lotcod,)
   841→            )
   842→            db_execute(
   843→                "UPDATE serralheria_producao "
   844→                "SET status = 'finalizado', data_fim = NOW() "
   845→                "WHERE lote = %s AND status = 'em_producao'",
   846→                (lotcod,)
   847→            )
   848→            for row in rows:
   849→                dept = setor_para_dept.get(row.get('setor',''), row.get('setor',''))
   850→                _sync_op_status(row.get('of_numero'), dept, 'concluido')
   851→            app.logger.info('Lote %s finalizado no totem', lotcod)
   852→
   853→        return jsonify({'success': True, 'mensagem': 'Producao finalizada!'})
   854→    except Exception as e:
   855→        app.logger.error('Erro finalizar: %s', e)
   856→        return jsonify({'error': str(e)}), 500
