   472→   472→function tempoDecorrido(isoStr) {
   473→   473→  if (!isoStr) return '';
   474→   474→  try {
   475→   475→    const p = isoStr.replace('T',' ').split(/[- :]/).map(Number);
   476→   476→    const inicio = new Date(p[0],p[1]-1,p[2],p[3]||0,p[4]||0,p[5]||0);
   477→   477→    const diff = Math.max(0, Math.floor((Date.now() - inicio.getTime())/1000));
   478→   478→    const h = String(Math.floor(diff/3600)).padStart(2,'0');
   479→   479→    const m = String(Math.floor((diff%3600)/60)).padStart(2,'0');
   480→   480→    const s = String(diff%60).padStart(2,'0');
   481→   481→    return `${h}:${m}:${s}`;
   482→   482→  } catch(e){ return ''; }
   483→   483→}
